## פרק יג — שבירת הסדרים והופעת משיח השקר

בַּיָּמִים הָהֵם יִתְיַקְּרוּ הָאִיִּים בַּיָּם, וְיִפָּסְקוּ מַעַבְרֵי הַיַּמִּים וְיוֹשְׁבֵי הָאִיִּים.

וְאָז יִשְׁלְמוּ הַזְּמַנִּים, וְיִקְרְבוּ הַיָּמִים הָאַחֲרוֹנִים, וְיִהְיוּ הָאוֹתוֹת הָאֵלֶּה: הַשֶּׁמֶשׁ תֹּאבַד מֵאוֹרָהּ, וְהַיָּרֵחַ יִכָּלֵא מִמֶּהֲלָכוֹ, אַף הָרָקִיעַ אֶת פָּנָיו יִכַּסֶּה — כַּאֲשֶׁר יַחֵלּוּ לְהַרְבּוֹת הַיָּמִים הָאַחֲרוֹנִים. יַחֵלּוּ הָאוֹתוֹת הַגְּדוֹלוֹת לְהֵרָאוֹת מִתַּחַת הַשֶּׁקֶט: וְרַעַשׁ גָּדוֹל יִהְיֶה בָּאָרֶץ, וְקוֹל חָזָק וְנִשְׁמָע בְּרָקִיעַ הַשָּׁמַיִם, וּגְנִיחָה גְדוֹלָה בָּעוֹלָם; וְיִשָּׁמַע מִיָּם לַיָּם, וּמִקְצוֹת הַשָּׁמַיִם וְעַד קְצוֹת הָאָרֶץ. וְעַנְנֵי הָרָקִיעַ יִסְרְקוּ, וְאֵשׁ עַזָּה בֵּינֵיהֶם בְּאַרְבַּע רוּחוֹת הַשָּׁמַיִם תִּשְׁכֹּן, וְרָאשֵׁי הֶהָרִים תּוּקַדְנָה אַרְבַּע פְּעָמִים בְּיוֹם אֶחָד.

וּמַלְאָכִים יֵרָאוּ בְּצִיּוֹן, וּקְדוֹשִׁים בִּירוּשָׁלַיִם; אַף מַחֲנוֹת מַלְאָכִים עַל גַּלֵּי הַיָּם יֵרָאוּ. אָז פַּחַד בַּיָּם יִשְׁכֹּן, וּרְתֵת בָּאִיִּים יִפֹּל, וָרַעַשׁ גָּדוֹל בִּמְדוֹרֵי הָעוֹלָם. וְאָז מַלְאֲכֵי הַשָּׁמַיִם כָּאֵשׁ יֵרָאוּ, וּבִבְנֵי אָדָם רַבִּים בִּשְׁבִילֵיהֶם יִשָּׂרֵפוּ. אָז תִּשְׁתַּגֵּשׁ הָאָרֶץ וְהַיָּם, וְיָשֹׁבּוּ רוּחוֹת חֲזָקוֹת וְעַזּוֹת.

בָּהֶם בַּיָּמִים הָהֵם הַשֶּׁמֶשׁ כַּשַּׂק תִּתְעַטֵּף, וְהַיָּרֵחַ כְּדָם יִלְבַּשׁ, וְהַכּוֹכָבִים יִפְּלוּ כַּעָלִים מִן הָאִילָן, וּכְנִיצוֹצוֹת אֵשׁ; וְחִצִּים וּרְמָחִים עַל פְּנֵי הָאָרֶץ יִשָּׁפֵכוּ.

וְאָז כָּל הָעַמִּים וְכָל הַלְּשׁוֹנוֹת יֵחָרְדוּ וְיִתְרַעֲשׁוּ, וּמִן הַשֶּׁקֶט — גְּנִיחוֹת גְּדוֹלוֹת, וּפַחִים יִהְיוּ בְּקֶרֶב הָאָרֶץ, וְהָאָרֶץ מִמְּקוֹמָהּ תִּסָּחֵף כְּנַחַת עַד תְּהוֹם רַבָּה. וּבְנֵי אָדָם רַבִּים, בְּחַיֵּיהֶם עוֹד, יִבָּלְעוּ בְּקֶרֶב הָאָרֶץ.

וְאָז יִתְרַעֲשׁוּ הֶהָרִים מִמְּקוֹמָם, אַף תִּשָּׁאֶינָה הָרָמוֹת מִמְּקוֹמָן. וְאָז יֵרָאוּ עַמּוּדֵי אֵשׁ מִן הַשָּׁמַיִם, וְתַנּוּר שַׁלְהֶבֶת מִתּוֹךְ הֶעָנָן; וְיֵרָאוּ בָּרָקִיעַ הַשָּׁמַיִם כְּסוּסֵי אֵשׁ וּכְמֶרְכְּבוֹת מִלְחָמָה, אֲחוּזִים חֶרֶב בַּרְזֶל וָרֹמַח קְרָב.

יִהְיֶה בַּיָּמִים הָהֵם: תֵּלֵד אִשָּׁה זָכָר מִשֵּׁבֶט בֵּית לֵוִי, וְיֵרָאוּ בוֹ הָאוֹתוֹת הָאֵלֶּה — וְהָיוּ צַמְרִים(?) עַל בְּשָׂרוֹ כִּכְלֵי מִלְחָמָה, וְשָׁלָל — תַּנּוּר וְקֶשֶׁת וְחֶרֶב וְרֹמַח וְחֶרֶב בַּרְזֶל וּמֶרְכְּבוֹת קְרָב. פָּנָיו כִּפְנֵי תַנּוּר בּוֹעֵר, עֵינָיו כְּגַחֲלֵי שַׁלְהֶבֶת, קֶרֶן אַחַת לוֹ בֵּין עֵינָיו, וְרֹאשׁוֹ שָׁבוּר, וְיוֹצֵא מִמֶּנּוּ כִּדְמוּת נָחָשׁ.

וּכְשֶׁיַּחֵלּוּ הָאוֹתוֹת הָאֵלֶּה לִהְיוֹת, אָז תָּחֵל בִּיאָתוֹ שֶׁל מוֹשֵׁךְ־הַלֵּב, וְהַנָּחָשׁ הָעֲקַלְקַל, מְשִׁיחַ הַשֶּׁקֶר — לְהֵרָאוֹת.
