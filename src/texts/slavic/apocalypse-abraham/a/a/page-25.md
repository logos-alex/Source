וּבְעוֹדֶנּוּ מְדַבֵּר, וָאֶמָּצֵא עַל הָאָרֶץ, וָאֹמַר: קַדְמוֹן, אֵינֶנִּי עוֹד בַּכָּבוֹד אֲשֶׁר הָיִיתִי בּוֹ לְמַעְלָה, וַאֲשֶׁר בְּלִבִּי חָפַצְתִּי לְהָבִין לֹא הֵבַנְתִּי. וַיֹּאמֶר אֵלַי: אֶת מַאֲוַיֵּי לִבְּךָ אַגִּיד לְךָ. הִגַּדְתָּ לָדַעַת אֶת הַנֶּגַע הָעֲשִׂירִי אֲשֶׁר עַל הַגּוֹיִם, הֲכִינֹתִי וְהִזְמַנְתִּי, אַחֲרֵי כְלוֹת שְׁנֵים עָשָׂר שָׁעוֹת עַל הָאָרֶץ. שְׁמַע אֶת אֲשֶׁר אַגִּיד לְךָ, יִהְיֶה כֵן: הָרִאשׁוֹנָה — צָרָה מִדֹּחַק רָב. הַשְּׁנִיָּה — שְׂרֵפוֹת אֵשׁ בֶּעָרִים. הַשְּׁלִישִׁית — אֲבַדּוֹן הַיָּם בַּבְּהֵמָה. הָרְבִיעִית — רָעָב לְתוֹשְׁבֵי מִשְׁפַּחְתָּם. הַחֲמִישִׁית — בְּרַעַשׁ וּבְחֶרֶב אֲבַדּוֹן. הַשִּׁשִּׁית — בָּרָד וְרֹב שֶׁלֶג. הַשְּׁבִיעִית — חַיַּת שָׂדֶה תִּהְיֶה קִבְרָם. הַשְּׁמִינִית — רָעָב וָדֶבֶר יָבִיאוּ כִלָּיוֹן. הַתְּשִׁיעִית — עֹנֶשׁ חֶרֶב וּמָנוֹס בְּצָרוֹת. הָעֲשִׂירִית — רַעַם וְקוֹלוֹת וּרְעָשִׁים לְהַשְׁחָתָה. אָז אֶתְקַע בַּשּׁוֹפָר מִן הָאֲוִיר וְאֶשְׁלַח אֶת בְּחִירִי, וּמִלֵּאתִי בוֹ מִדַּת כָּל כֹּחִי אַחַת, וְהוּא יִקְרָא לְעַמִּי הַנֶּאֱסָף מִן הַגּוֹיִם. וַאֲנִי אֶשְׂרֹף בָּאֵשׁ אֶת הַמְבַזִּים לָהֶם, הַשּׁוֹלְטִים בָּהֶם לְעוֹלָם, וְאֶתֵּן אֶת הַמְבַזִּים אוֹתִי לְחֶרְפָּה מֵאֵת עוֹלָם הַזֶּה, כִּי אַתָּה הֲכִינוֹתָ לָהֶם מַאֲכַל אֵשׁ שְׁאוֹל, וּמְעוֹף בְּלִי הֶפְסֵק בָּאֲוִיר, תַּחַת הָאָרֶץ, בַּתַּחְתִּיּוֹת.

*[הַטֶּקְסְט הַסְּלָאבִי שֶׁהוּעֲלָה מִסְתַּיֵּם כָּאן (עמ' 53), בְּתוֹךְ נְאוּם הַמַּלְאָךְ עַל הַמַּכּוֹת וְקֵץ הַיָּמִים]*

וַיְהִי בִּבְקָעִי אֶת הָעֵצִים וָאֶמְצָא בָהֶם אֵל קָטָן, נִכְנָס בְּתוֹךְ סַל הָעֵצִים כַּאֲשֶׁר בְּקַעְתִּיו, וְכָתוּב הָיָה עַל מִצְחוֹ: אֵל בָּרִיסָת. וְלֹא הִגַּדְתִּי לְאָבִי כִּי אֵל עֵץ מָצָאתִי בְּתוֹךְ שְׁבָבֵי בָרִיסָת. וַיְהִי כְּשׂוּמִי אֶת הָעֵצִים עַל הָאֵשׁ לַעֲשׂוֹת אֳכֶל לְאָבִי, וּבְצֵאתוֹ הַחוּצָה לִשְׁאֹל עַל הָאֳכֶל, שַׂמְתִּי אֶת בָּרִיסָת אֵצֶל הַבְעָרַת הָאֵשׁ, וָאֹמַר אֵלָיו בְּאִיּוּם: בָּרִיסָת, שְׁמֹר פֶּן תִּכְבֶּה הָאֵשׁ עַד בּוֹאִי, וְאִם תִּכְבֶּה נְפַח בָּהּ וְתִבְעַר. וָאֵצֵא וָאֶעֱשֶׂה עֲבוֹדָתִי, וּבְשׁוּבִי מָצָאתִי אֶת בָּרִיסָת נָפוּל בְּתוֹךְ הַלַּהַב, רַגְלָיו אֲחוּזוֹת אֵשׁ וְנִשְׂרָפוֹת קָשׁוֹת. וָאֶצְחַק מְאֹד בְּלִבִּי וָאֹמַר: אָכֵן בָּרִיסָת יוֹדֵעַ לְהַבְעִיר אֵשׁ וּלְבַשֵּׁל אֳכֶל. וַיְהִי כְּדַבְּרִי כֵן בְּשִׂחְקִי, וְהוּא מְעַט מְעַט מִן הָאֵשׁ נִשְׂרַף וַיְהִי לְאֵפֶר. וָאָבִיא אֶת הָאֳכֶל אֶל אָבִי לֶאֱכֹל, וָאֶתֵּן לוֹ יַיִן וְחָלָב, וַיֹּאכַל וַיִּשְׁתְּ וַיִּמְתַּק לוֹ, וַיְבָרֶךְ אֶת מָרוּמָת אֱלֹהָיו.
