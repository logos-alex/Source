וַיֹּאמֶר אֵלַי: כֵּן קָרוֹב לָעַמִּים בַּעֲבוּרְךָ, וְאַחֲרֶיךָ בַּעֲבוּר הָעָם הַמֻּבְדָּל, וְעִם זַרְעֲךָ, כִּי תִרְאֶה בַתְּמוּנָה אֶת הַכָּבֵד עֲלֵיהֶם, וַאֲנִי אַגִּיד לְךָ מָה יִהְיֶה וְכַמָּה יִהְיֶה בְּאַחֲרִית הַיָּמִים. הַבֵּט עַתָּה בְּךָ אֶל הַתְּמוּנָה. וָאַבִּיט וָאֵרֶא שָׁם אֵת אֲשֶׁר הָיָה לְפָנַי בַּבְּרִיאָה. וָאֵרֶא אֶת אָדָם וְאֶת חַוָּה אֲשֶׁר עִמּוֹ, וְעִמָּהֶם הָאוֹיֵב הֶעָרוּם, וְאֶת קַיִן פֹּעֵל עָוֶל בְּיַד הָאוֹיֵב, וְאֶת הֶבֶל הַנֶּהֱרָג, וְאֶת הַכִּלָּיוֹן הַמּוּבָא עַל הָרָשָׁע וְהַנָּתוּן לוֹ. וָאֵרֶא שָׁם אֶת הַזְּנוּת וְאֶת הַחוֹמְדִים אוֹתָהּ וְאֶת הֶסְתֵּרָהּ, וְאֶת קִנְאָתָם וְאֵשׁ כִּלְיוֹנָם בְּתַחְתִּיּוֹת הָאָרֶץ. וָאֵרֶא שָׁם אֶת הַגְּנֵבָה וְאֶת הַנֶּחְפָּזִים אֵלֶיהָ וְאֶת מְזִמּוֹתָם. וָאֵרֶא שָׁם אֲנָשִׁים עֲרֻמִּים, מִצְחָם זֶה מוּל זֶה, וְעֶרְוָתָם, וְנֵזֶק אִישׁ בְּרֵעֵהוּ וּגְמוּלָם. וָאֵרֶא שָׁם אֶת הַתַּאֲוָה, וּבְיָדָהּ רֹאשׁ כָּל עַוְלָה. וָאֵרֶא שָׁם אֶת דְּמוּת פֶּסֶל הַקִּנְאָה, כְּמַעֲשֵׂה חָרָשׁ עֵצִים כַּאֲשֶׁר עָשָׂה אָבִי, גּוּפוֹ כִּנְחֹשֶׁת, וּלְפָנָיו אִישׁ מִשְׁתַּחֲוֶה לוֹ, וּמִזְבֵּחַ נֶגְדּוֹ, וּנְעָרִים נִשְׁחָטִים עָלָיו לִפְנֵי פְנֵי הַפֶּסֶל. וָאֹמַר: מַה זֶּה הַפֶּסֶל, אוֹ מַה הַמִּזְבֵּחַ, אוֹ מִי הַנִּזְבָּחִים, אוֹ מִי הַזֹּבֵחַ, אוֹ מַה זֶּה הַהֵיכָל אֲשֶׁר אֲנִי רוֹאֶה, נָאֶה בְּחָכְמַת מַעֲשֵׂהוּ וּבְחִנּוֹ, תַּחַת כִּסֵּא כְבוֹדְךָ.
