וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִפְתְּחוּ הָרְקִיעִים מִתַּחְתַּי, וָאֵרֶא אֶת הָרָקִיעַ הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו, מוּטָל תַּחְתָּיו, וְאוֹר וָטַל וַהֲמוֹן מַלְאָכִים וּגְבוּרַת כָּבוֹד נִסְתָּר מִלְּמַעְלָה. וָאֵרֶא חַיּוֹת, וְלֹא רָאִיתִי שָׁם אַחֵר. וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הָרָקִיעַ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים מִצְוַת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כַּאֲשֶׁר עָמַדְתִּי עַל הָעוֹמְדִים מֵעָלָם. וְלֹא הָיָה בָּרָקִיעַ הַהוּא צוּרָה אַחֶרֶת שֶׁל גְּבוּרָתוֹ, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הָרָקִיעַ הַשְּׁבִיעִי, הַכֹּחַ אֲשֶׁר רָאִיתִי צֻוָּה לָנוּעַ אֶל הָרָקִיעַ הַשְּׁלִישִׁי. וָאֵרֶא שָׁם, בָּרָקִיעַ הַחֲמִישִׁי, אֶת כֹּחוֹת הַכּוֹכָבִים עֹשִׂים אֶת הַמִּצְוֹת הַמֻּטָּלוֹת עֲלֵיהֶם, וִיסוֹדוֹת הָאָרֶץ שׁוֹמְעִים לָהֶם. וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם, וָאֹמַר הִנֵּנִי. רְאֵה מִלְּמַעְלָה אֶת אֲשֶׁר תַּחְתֶּיךָ, אֶת הַכּוֹכָבִים, וּסְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם. וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם. וַיֹּאמֶר אֵלַי: כְּמִסְפַּר הַכּוֹכָבִים וּכְחֵילָם אֶתֵּן אוֹתָם עַם וְעַמִּים מִזַּרְעֲךָ, וְעָם מֻבְדָּל לִי בְגוֹרָלִי עִם עֲזָאזֵל. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יְדַבֵּר נָא עַבְדְּךָ לְפָנֶיךָ, וְאַל יִחַר אַפְּךָ עַל בְּחִירְךָ: הִנֵּה בְּטֶרֶם הֲרִימוֹתַנִי חֵרְפַנִי עֲזָאזֵל, וְאֵיךְ עַתָּה בְּאֵין הוּא לְפָנֶיךָ תְּכוֹנֵן עִמָּם. וַיֹּאמֶר אֵלַי: רְאֵה עַתָּה תַּחַת רַגְלֶיךָ אֶת הָרָקִיעַ, וְהָבֵן אֶת הַבְּרִיאָה הַמְצֻיֶּרֶת מֵאָז עַל הָרָקִיעַ הַזֶּה, וְאֶת הַבְּרוּאִים אֲשֶׁר בּוֹ, וְאַחֲרָיו אֶת הָעוֹלָם הַמּוּכָן. וָאַבִּיט תַּחַת רְקִיעַ רַגְלַי, וָאֵרֶא תַחַת שְׁלֹשָׁה שָׁמַיִם אֶת אֲשֶׁר בּוֹ: אֶת הָאָרֶץ וּפְרִיָּהּ וְאֶת רֶמֶשׂ...
