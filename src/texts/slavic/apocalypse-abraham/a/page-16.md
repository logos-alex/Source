כַמָּה זְמַן שְׁעַת עוֹלָם. וַיֹּאמֶר: שְׁתֵּים עֶשְׂרֵה שָׁנָה שַׂמְתִּי לָעוֹלָם הַזֶּה שֶׁלֹּא יִשְׂבַּע, לְהַחֲזִיק בָּעַמִּים וּבְזַרְעֲךָ, וְעַד קֵץ הַזְּמַן יִהְיֶה כְּכֹל אֲשֶׁר רָאִיתָ; סְפֹר וְדַע וְהַבֵּט בַּתְּמוּנָה. וָאֵרֶא אִישׁ יוֹצֵא מִצַּד שְׂמֹאל הַגּוֹיִם. וַיֵּצְאוּ אֲנָשִׁים וְנָשִׁים וִילָדִים מִצַּד הַגּוֹיִם, עַמִּים רַבִּים, וַיִּשְׁתַּחֲווּ לוֹ. וְעוֹד אֲנִי מַבִּיט, וַיֵּצְאוּ מִצַּד יָמִין, וּמִקְצָתָם בּוֹשִׁים מֵאוֹתוֹ אִישׁ, וּמִקְצָתָם מַכִּים אוֹתוֹ, וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ. וָאֵרֶא כִּי הִשְׁתַּחֲווּ לוֹ, וַיָּרָץ עֲזָאזֵל וַיִּשְׁתַּחוּ, וַיִּשַּׁק אֶת פָּנָיו וַיָּשָׁב וַיַּעֲמֹד אַחֲרָיו. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מִי הָאִישׁ הַנִּכְלָם וְהַמֻּכֶּה מִן הַגּוֹיִם, וְעִם עֲזָאזֵל מִשְׁתַּחֲוֶה לוֹ. וַיַּעַן וַיֹּאמֶר: שְׁמַע אַבְרָהָם, הָאִישׁ אֲשֶׁר רָאִיתָ נִכְלָם וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוֶה לוֹ, זֶה מְנוּחַת הַגּוֹיִם, מִזַּרְעֲךָ יִהְיֶה לְעַם בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל הָעוֹלָם הַזֶּה בְּגוּפוֹ. וּבַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל קֵץ עוֹלָמִי אָקִים אֶת הָאִישׁ הַזֶּה מִזַּרְעֲךָ...

אֲשֶׁר רָאִיתָ מֵעַמִּי, לוֹ יִדַּמּוּ כֻלָּם, וּמְשָׁלִים כְּקוֹרְאִים מִמֶּנִּי, עוֹבְרִים בְּאוֹרוֹתָם. וַאֲשֶׁר רָאִיתָ מִצַּד שְׂמֹאל בַּתְּמוּנָה יוֹצְאִים וּמִשְׁתַּחֲוִים לוֹ, מֵהַגּוֹיִם רַבִּים יִבְטְחוּ בוֹ. וַאֲשֶׁר רָאִיתָ מִזַּרְעֲךָ מִצַּד יָמִין, מִקְצָתָם בּוֹשִׁים וּמַכִּים וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ, רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ, וְהוּא יְנַסֶּה אֶת הַמִּשְׁתַּחֲוִים לוֹ מִזַּרְעֲךָ, בְּקֵץ הַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה, בְּקֵץ הָעוֹלָם. בְּטֶרֶם יָחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, יָבוֹא מִשְׁפָּטִי עַל הָעַמִּים הָעַרְמוּמִים, עַל הָעָם הַמֻּבְדָּל לִי מִזַּרְעֲךָ. וּבַיָּמִים הָהֵם אָבִיא עַל כָּל הַתֵּבֵל
