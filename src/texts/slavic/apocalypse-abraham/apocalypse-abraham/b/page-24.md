וַיַּרְאֵנִי אֶת הֲמוֹן עַמּוֹ, וַיֹּאמֶר אֵלַי: לָכֵן בַּעֲבוּר אַרְבַּעַת הַמּוֹרָדוֹת אֲשֶׁר רָאִיתָ, יִהְיֶה לִי מֵהֶם קֶצֶף, וּבָהֶם גְּמוּל מַעֲשֵׂיהֶם מֵאִתִּי.

וּבָרְבִיעִי — מוֹרָד שֶׁל מֵאָה שָׁנָה — וּשְׁעַת עוֹלָם אַחַת, וְהִיא גַּם כֵּן מֵאָה שָׁנָה, תִּהְיֶינָה בְרָעָה בָּעַמִּים, בְּשָׁעָה שֶׁל תּוֹכֵחָה וְשֶׁל חֶרְפָּה בַּגּוֹיִם.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, כַּמָּה זְמַן שְׁעַת עוֹלָם.

וַיֹּאמֶר: עֶשְׂרִים וְאַחַת שָׁנָה שַׂמְתִּי לָעוֹלָם הַזֶּה שֶׁלֹּא יִשְׂבַּע, לְהַחֲזִיק בָּעַמִּים וּבְזַרְעֲךָ, וְעַד קֵץ הַזְּמַן יִהְיֶה כְּכֹל אֲשֶׁר רָאִיתָ; סְפֹר וְדַע וְהַבֵּט בַּתְּמוּנָה.

וָאַבִּיט וָאֵרֶא אִישׁ יוֹצֵא מִצַּד שְׂמֹאל הַגּוֹיִם, וַיֵּצְאוּ אֲנָשִׁים וְנָשִׁים וִילָדִים מִצַּד הַגּוֹיִם הָרַבִּים, וַיִּשְׁתַּחֲווּ לוֹ.

וְעוֹד אֲנִי מַבִּיט, וַיֵּצְאוּ מִצַּד יָמִין, וּמִקְצָתָם בּוֹשִׁים מֵאוֹתוֹ אִישׁ, וּמִקְצָתָם מַכִּים אוֹתוֹ, וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ.

וָאֵרֶא כִּי הִשְׁתַּחֲווּ לוֹ, וַיָּרָץ עֲזָאזֵל וַיִּשְׁתַּחוּ, וַיִּשַּׁק אֶת פָּנָיו וַיָּשָׁב וַיַּעֲמֹד אַחֲרָיו.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מִי הָאִישׁ הַנִּכְלָם וְהַמֻּכֶּה מִן הַגּוֹיִם, וְעִם עֲזָאזֵל מִשְׁתַּחֲוֶה לוֹ.

וַיַּעַן וַיֹּאמֶר: שְׁמַע אַבְרָהָם, הָאִישׁ אֲשֶׁר רָאִיתָ נִכְלָם וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוֶה לוֹ, זֶה מְנוּחַת הַגּוֹיִם, מִזַּרְעֲךָ יִהְיֶה לְעַם בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל הָעוֹלָם הַזֶּה בְּגוּפוֹ.

וּבַשָּׁנָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל קֵץ עוֹלָמִי אָקִים אֶת הָאִישׁ הַזֶּה מִזֶּרַע עַמִּי — אֲשֶׁר רָאִיתָ — לוֹ יִדַּמּוּ כֻלָּם, וּבִמְשָׁלִים כְּקוֹרְאִים מִמֶּנִּי, עוֹבְרִים בְּאוֹרוֹתָם.

וַאֲשֶׁר רָאִיתָ מִצַּד שְׂמֹאל בַּתְּמוּנָה יוֹצְאִים וּמִשְׁתַּחֲוִים לוֹ, מֵהַגּוֹיִם רַבִּים יִבְטְחוּ בּוֹ.

וַאֲשֶׁר רָאִיתָ מִזַּרְעֲךָ מִצַּד יָמִין, מִקְצָתָם בּוֹשִׁים וּמַכִּים וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ — רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ.

וְהוּא יְנַסֶּה אֶת הַמִּשְׁתַּחֲוִים לוֹ מִזַּרְעֲךָ, בְּקֵץ הַשָּׁעָה הָעֶשְׂרִים וְאַחַת, בְּקֵץ הָעוֹלָם הַבָּלוּי.

בְּטֶרֶם יָחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, יָבוֹא מִשְׁפָּטִי עַל הָעַמִּים הָעַרְמוּמִים, עַל זֶרַע עַמְּךָ הַמֻּבְדָּל לִי.
