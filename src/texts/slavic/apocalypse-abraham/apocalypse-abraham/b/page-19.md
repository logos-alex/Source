וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִפְתְּחוּ הָרְקִיעִים מִתַּחְתַּי, וָאֵרֶא עַל הָרָקִיעַ הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו אֵשׁ פְּרוּסָה, וְאוֹר וָטַל, וַהֲמוֹן מַלְאָכִים, וּגְבוּרַת כָּבוֹד בִּלְתִּי נִרְאֶה מִלְּמַעְלָה — מִלְּבַד הַחַיּוֹת אֲשֶׁר רָאִיתִי, לֹא רָאִיתִי שָׁם אַחֵר.

וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הָרָקִיעַ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים אֶת מִצְוַת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כְּפִי אֲשֶׁר הָיִיתִי עוֹמֵד עַל הַתְּלוּיִים בּוֹ.

וְלֹא הָיָה בָּרָקִיעַ הַהוּא כֹּחַ בְּשׁוּם צוּרָה אַחֶרֶת, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הָרָקִיעַ הַשְּׁבִיעִי — הַכֹּחַ אֲשֶׁר רָאִיתִי — צֻוָּה לָנוּעַ.

וָאֵרֶא שָׁם, בָּרָקִיעַ הַחֲמִישִׁי, אֶת כֹּחוֹת הַכּוֹכָבִים מְקַיְּמִים אֶת הַמִּצְוֹת הַמֻּטָּלוֹת עֲלֵיהֶם, וִיסוֹדוֹת הָאָרֶץ שׁוֹמְעִים לָהֶם.

וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם, אַבְרָהָם.

וָאֹמַר הִנֵּנִי.

וַיֹּאמֶר: הַבֵּט מִלְּמַעְלָה, וְהִנֵּה תַּחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם.

וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם — עָפָר וָאֵפֶר.

וַיֹּאמֶר לִי: כְּמִסְפַּר הַכּוֹכָבִים וְכֹחָם אֶתֵּן לְזַרְעֲךָ, גּוֹיִם וְעַמִּים מֻבְדָּלִים לִי בְגוֹרָלִי, עִם עֲזָאזֵל.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יְדַבֵּר נָא עַבְדְּךָ לְפָנֶיךָ, וְאַל יִחַר אַפְּךָ עַל בְּחִירְךָ: הִנֵּה בְּטֶרֶם הֲרִימוֹתַנִי, חֵרְפַנִי עֲזָאזֵל; וְעַתָּה, כְּשֶׁאֵינֶנּוּ לְפָנֶיךָ, תְּכוֹנֵן עִמּוֹ.

הַבֵּט עוֹד בַּתְּמוּנָה — מִי הוּא הַמַּתְעֶה אֶת חַוָּה, וְאֵי זֶה פְּרִי הָעֵץ, וְתֵדַע מָה יִהְיֶה וְכַמָּה יִהְיֶה לְזַרְעֲךָ נַחֲלַת הַיָּמִים, וְאֶת אֲשֶׁר לֹא תוּכַל לְהָבִין אֲנִי אַגִּיד לְךָ, כִּי מָצָאתָ חֵן לְפָנַי; לְךָ אֲגַלֶּה אֶת אֲשֶׁר צָפַנְתִּי בְלִבִּי.

וָאַבִּיט בַּתְּמוּנָה, וַתָּרֻצְנָה עֵינַי אֶל צַד גַּן עֵדֶן, וָאֵרֶא שָׁם אִישׁ גָּדוֹל מְאֹד בְּקוֹמָתוֹ וְנוֹרָא בְרָחְבּוֹ, אֵין דָּמֵהוּ בְמַרְאֵהוּ, מְלֻפָּף בְּאִשָּׁה הַשָּׁוָה לוֹ בְּמַרְאֵה זָכָר וּבְקוֹמָה.

וְהֵם עֹמְדִים תַּחַת עֵץ אֶחָד בְּעֵדֶן, וּפְרִי הָעֵץ הַהוּא כְּמַרְאֵה אֶשְׁכּוֹל תְּמָרִים, וְאַחֲרֵי הָעֵץ עֹמֵד כְּדִמְיוֹן נָחָשׁ, וְיָדַיִם וְרַגְלַיִם לוֹ כְּאָדָם, וְעַל כְּתֵפָיו שֵׁשׁ כְּנָפַיִם, שָׁלֹשׁ מִימִינוֹ וְשָׁלֹשׁ מִשְּׂמֹאלוֹ, וּבְיָדוֹ אֶשְׁכּוֹל הָעֵץ וּמַאֲכִיל אֶת הַשְּׁנַיִם אֲשֶׁר רָאִיתִי מְלֻפָּפִים.

וָאֹמַר: מִי הֵם הַמְלֻפָּפִים זֶה בָזֶה, אוֹ מִי הָעֹמֵד בֵּינֵיהֶם, אוֹ מַה פִּרְיוֹ אֲשֶׁר יֹאכְלוּ, קַדְמוֹן וְגִבּוֹר.

וַיֹּאמֶר: זֶה מַחְשֶׁבֶת בְּנֵי אָדָם, זֶה אָדָם, וְזֹאת תְּשׁוּקָתָם בָּאָרֶץ, זֹאת חַוָּה; וַאֲשֶׁר בֵּינֵיהֶם, זֶה קְלוֹן מַעֲשֵׂיהֶם לְאָבְדָּן — הוּא עֲזָאזֵל בְּעַצְמוֹ.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, לָמָּה נָתַתָּ לוֹ כֹחַ וְשִׁלְטוֹן כָּזֶה לְהַשְׁחִית אֶת בְּנֵי הָאָדָם בְּמַעֲשֵׂיהֶם עַל הָאָרֶץ.

וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, אֲשֶׁר יַחְפְּצוּ בָרַע וַאֲשֶׁר שָׂנֵאתִי בְעֹשֵׂיהֶם, עֲלֵיהֶם נָתַתִּי לוֹ שָׁלְטָן, וְלִהְיוֹת נֶאֱהָב.

וָאַעַן וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מַדּוּעַ חָפַצְתָּ לָתֵת אֶת הָרָע נִחְמָד בְּלֵב אָדָם, וְקָצַפְתָּ עַל אֲשֶׁר בָּרְצוֹנְךָ — וּבְעֵצָתְךָ הָיָה מַעֲשֵׂהוּ לְלֹא תוֹעֶלֶת.
