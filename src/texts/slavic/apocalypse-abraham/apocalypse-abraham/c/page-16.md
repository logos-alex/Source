וָאֶשְׁמַע אֶת קוֹל קְדֻשָּׁתָם כְּקוֹל אִישׁ אֶחָד, וַיָּבֹא אֵלַי קוֹל מִתּוֹךְ הָאֵשׁ, אֹמֵר: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי.

וַיֹּאמֶר: הַבֵּט אֶל הַמִּפְרָשִׂים אֲשֶׁר תַּחְתֶּיךָ, אֲשֶׁר עֲלֵיהֶם נִצְעָדְתָּ, וּרְאֵה כִּי בְּשׁוּם מִפְרָשׂ אֵין אַחֵר זוּלָתִי אֲשֶׁר בִּקַּשְׁתָּ אוֹ אֲשֶׁר אֲהֵבְךָ.

וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִבְקְעוּ הַמִּפְרָשִׂים, וְמִתַּחְתַּי הַשָּׁמַיִם.

וָאֵרֶא עַל הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו, אֵשׁ פְּרוּסָה, וְאוֹר רַב, וָטַל, וַהֲמוֹן מַלְאָכִים, וּגְבוּרַת כָּבוֹד בִּלְתִּי-נִרְאֶה מִלְּמַעְלָה — מִלְּבַד הַחַיּוֹת אֲשֶׁר רָאִיתִי, לֹא רָאִיתִי שָׁם אַחֵר.

וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הַמִּפְרָשׂ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים אֶת מִצְוֹת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כַּאֲשֶׁר עָמַדְתִּי עַל הָעוֹמְדִים עָלָיו.

וְלֹא הָיָה עַל הַמִּפְרָשׂ הַהוּא בְּשׁוּם צוּרָה אַחֶרֶת כֹּחַ, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הַמִּפְרָשׂ הַשְּׁבִיעִי, הַכֹּחַ אֲשֶׁר רָאִיתִי, פָּקַד עַל הַמִּפְרָשׂ הַשִּׁשִּׁי.

וְנִשָּׂא, וָאֵרֶא בַּמִּפְרָשׂ הַחֲמִישִׁי כֹּחוֹת הַכּוֹכָבִים, וַיְצַוּוּ לְקַיֵּם אֶת הַמִּצְוֹת, וַיִּשְׁמְעוּ לָהֶם יְסוֹדוֹת הָאָרֶץ.

וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם אַבְרָהָם.

וָאֹמַר הִנֵּנִי.

וַיֹּאמֶר: הַבֵּט מִלְּמַעְלָה, וְהִנֵּה תַּחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם.

וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם — עָפָר וָאֵפֶר.

וַיֹּאמֶר לִי: כְּמִסְפַּר הַכּוֹכָבִים וְכֹחָם אֶתֵּן לְזַרְעֲךָ, גּוֹיִם וְעַמִּים מֻבְדָּלִים לִי בְגוֹרָלִי, עִם עֲזָאזֵל.

וּבְעוֹדֶנִּי אוֹמֵר אֶת הַשִּׁיר, וּפִי הָאֵשׁ אֲשֶׁר עַל הַמִּפְרָשׂ עָלָה לְמַעְלָה, וָאֶשְׁמַע קוֹל כִּמְבוּכַת הַיָּם, וְלֹא חָדַל מִבְּעֵרַת הָאֵשׁ הָרַבָּה.

וְכַאֲשֶׁר עָלְתָה הָאֵשׁ לְמַעְלָה מִתְנוֹשְׂאָה, וָאֵרֶא תַחַת הָאֵשׁ כִּסֵּא, וּמִן הָאֵשׁ רַבֵּי הָעֵינַיִם סָבִיב, אֹמְרִים שִׁיר יָפֶה וְנָעִים, וְתַחַת הַכִּסֵּא אַרְבַּע חַיּוֹת אֵשׁ — וּמַרְאֵיהֶן וְקוֹלָן אֶחָד, כָּל אַחַת אַרְבַּעַת פָּנִים, וְעַל גּוּפָן אַרְבָּעָה רָאשִׁים.

מַרְאֵה פְנֵיהֶן: אַרְיֵה, שׁוֹר, אָדָם, נֶשֶׁר, כֵּן פְּנֵי הָאַרְבַּע חַיּוֹת, וּלְכָל אַחַת אַרְבַּע כְּנָפַיִם מִכְּתֵפֵיהֶן וּמֵחֶצְיָן וּמִמָּתְנֵיהֶן.

וּבִכְנָפַיִם אֲשֶׁר מִכְּתֵפֵיהֶן יְכַסּוּ פְּנֵיהֶן, וּבִשְׁתֵּי כְנָפַיִם מִמָּתְנֵיהֶן יְכַסּוּ רַגְלֵיהֶן, וּבִכְנָפַיִם הָאֶמְצָעִיּוֹת יִפְרְשׂוּ וְעָפוּ יָשָׁר.

וַיְהִי כְּכַלּוֹתָם אֶת הַשִּׁיר וַיַּבִּיטוּ אִישׁ אֶל רֵעֵהוּ וַיְאַיְּמוּ אִישׁ בְּרֵעֵהוּ.

וַיְהִי כִּרְאוֹת הַמַּלְאָךְ אֲשֶׁר עִמִּי כִּי יְאַיְּמוּ, וַיַּעַזְבֵנִי בַּמָּקוֹם וַיָּרָץ אֲלֵיהֶם מַהֵר, וַיַּסֵּב אֶת פְּנֵי כָּל חַיָּה מִנֶּגֶד פְּנֵי רֵעוּתָהּ, לְבִלְתִּי רְאוֹת אִישׁ אֶת פְּנֵי רֵעֵהוּ מְאַיְּמוֹת, וַיְלַמְּדֵם שִׁיר הַשָּׁלוֹם אֲשֶׁר בּוֹ עִם הַקַּדְמוֹן.

וַאֲנִי עוֹד עוֹמֵד וּמַבִּיט, וָאֵרֶא בִּיסוֹד הַחַיּוֹת מֶרְכָּבָה, גַּלְגַּלֵּי אֵשׁ, וְכָל גַּלְגַּל מָלֵא עֵינַיִם סָבִיב, וּמֵעַל הַגַּלְגַּלִּים הַכִּסֵּא אֲשֶׁר רָאִיתִי, וְהוּא נִסְתָּר בָּאֵשׁ, וְאֵשׁ סוֹבֵב אוֹתוֹ סָבִיב, וְאוֹר אֲשֶׁר לֹא יְסֻפָּר סוֹבֵב אֶת מַחְנֵה הָאֵשׁ.
