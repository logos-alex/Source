חלק זה (הסבר על עשו וישמעאל) משולב בפרק הקודם בנוסח א'.

כַּאֲשֶׁר אֶשְׁאַל, הֲלְאֹרֶךְ יָמִים יִהְיֶה לָהֶם כְּכֹל אֲשֶׁר רָאִיתִי.

וַיַּרְאֵנִי אֶת הֲמוֹן עַמּוֹ, וַיֹּאמֶר אֵלַי: לָכֵן בַּעֲבוּר אַרְבַּעַת הַמּוֹרָדוֹת אֲשֶׁר רָאִיתָ, יִהְיֶה לִי מֵהֶם קֶצֶף, וּבָהֶם יִהְיֶה גְּמוּל מַעֲשֵׂיהֶם מֵאִתִּי.

וּבַמּוֹרָד הָרְבִיעִי, מֵאָה שָׁנָה וְשָׁעָה אַחַת שֶׁל עוֹלָם — וְהִיא גַם כֵּן מֵאָה שָׁנָה — תִּהְיֶינָה בְרָעָה בָּעַמִּים.

וָאֹמַר: קַדְמוֹן, וְכַמָּה זְמַן שְׁעַת עוֹלָם.

וַיֹּאמֶר: שְׁתֵּים עֶשְׂרֵה שָׁנָה שַׂמְתִּי לָעוֹלָם הַזֶּה שֶׁלֹּא יִשְׂבַּע, לְהַחֲזִיק בָּעַמִּים וּבְזַרְעֲךָ, וְעַד קֵץ הַזְּמַן יִהְיֶה כְּכֹל אֲשֶׁר רָאִיתָ; סְפֹר וְדַע וְהַבֵּט בַּתְּמוּנָה.

וָאֵרֶא אִישׁ יוֹצֵא מִצַּד שְׂמֹאל הַגּוֹיִם.

וַיֵּצְאוּ אֲנָשִׁים וְנָשִׁים וִילָדִים מִצַּד הַגּוֹיִם, עַמִּים רַבִּים, וַיִּשְׁתַּחֲווּ לוֹ.

וְעוֹד אֲנִי מַבִּיט, וַיֵּצְאוּ מִצַּד יָמִין, וּמִקְצָתָם בּוֹשִׁים מֵאוֹתוֹ אִישׁ, וּמִקְצָתָם מַכִּים אוֹתוֹ, וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ.

וָאֵרֶא כִּי הִשְׁתַּחֲווּ לוֹ, וַיָּרָץ עֲזָאזֵל וַיִּשְׁתַּחוּ, וַיִּשַּׁק אֶת פָּנָיו וַיָּשָׁב וַיַּעֲמֹד אַחֲרָיו.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מִי הָאִישׁ הַנִּכְלָם וְהַמֻּכֶּה מִן הַגּוֹיִם, וְעִם עֲזָאזֵל מִשְׁתַּחֲוֶה לוֹ.

וַיַּעַן וַיֹּאמֶר: שְׁמַע אַבְרָהָם, הָאִישׁ אֲשֶׁר רָאִיתָ נִכְלָם וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוֶה לוֹ, זֶה מְנוּחַת הַגּוֹיִם, מִזַּרְעֲךָ יִהְיֶה לְעַם בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל הָעוֹלָם הַזֶּה בְּגוּפוֹ.

וּבַשָּׁנָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל קֵץ עוֹלָמִי אָקִים אֶת הָאִישׁ הַזֶּה מִזַּרְעֲךָ...

אֲשֶׁר רָאִיתָ מֵעַמִּי, לוֹ יִדַּמּוּ כֻלָּם, וּמְשָׁלִים כְּקוֹרְאִים מִמֶּנִּי, עוֹבְרִים בְּאוֹרוֹתָם.

וַאֲשֶׁר רָאִיתָ מִצַּד שְׂמֹאל בַּתְּמוּנָה יוֹצְאִים וּמִשְׁתַּחֲוִים לוֹ, מֵהַגּוֹיִם רַבִּים יִבְטְחוּ בוֹ.

וַאֲשֶׁר רָאִיתָ מִזַּרְעֲךָ מִצַּד יָמִין, מִקְצָתָם בּוֹשִׁים וּמַכִּים וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ, רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ, וְהוּא יְנַסֶּה אֶת הַמִּשְׁתַּחֲוִים לוֹ מִזַּרְעֲךָ, בְּקֵץ הַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה, בְּקֵץ הָעוֹלָם.

בְּטֶרֶם יָחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, יָבוֹא מִשְׁפָּטִי עַל הָעַמִּים הָעַרְמוּמִים, עַל הָעָם הַמֻּבְדָּל לִי מִזַּרְעֲךָ.

וּבַיָּמִים הָהֵם אָבִיא עַל כָּל הַתֵּבֵל עֶשֶׂר מַכּוֹת בְּרָעָה וּמַחֲלָה, וֶאֱנָקַת נֶפֶשׁ מָרָה אָבִיא עַל הַדּוֹרוֹת אֲשֶׁר עַל הָאָרֶץ, בַּעֲבוּר קֶצֶף וְהַשְׁחָתַת בְּרוּאֵיהֶם אֲשֶׁר יַכְעִיסוּנִי בָהֶם.

וְאָז יִוָּתְרוּ מִזַּרְעֲךָ אֲנָשִׁים צַדִּיקִים, בְּמִסְפָּר שְׁמוּרִים לִי, מְמַהֲרִים בִּכְבוֹד שְׁמִי אֶל הַמָּקוֹם הַמּוּכָן לָהֶם מֵאָז, אֲשֶׁר רָאִיתָ שָׁמֵם בַּתְּמוּנָה.

וְיִהְיוּ הַחַיִּים נִתְמָכִים בִּזְבָחִים וּבְמַתָּנוֹת שֶׁל צֶדֶק אֱמֶת, בָּעוֹלָם הַצַּדִּיק, וְאַחֲרַי יִשְׂמְחוּ תָמִיד, וְיַשְׁחִיתוּ אֶת מַשְׁחִיתֵיהֶם, וִיבַזּוּ אֶת מְבַזֵּיהֶם בְּחֶרְפָּתָם; יִהְיוּ רֹקְקִים בִּפְנֵי הַמְבַזִּים אוֹתִי, בִּרְאוֹתָם אוֹתִי שָׂמֵחַ עִם עַמִּי, וּמְקַבֵּל אֶת הַנֶּעֱנִים אֵלַי.

רְאֵה אַבְרָהָם אֶת אֲשֶׁר רָאִיתָ, וְאֶת אֲשֶׁר שָׁמַעְתָּ וְדַע — לֵךְ אֶל גּוֹרָלְךָ, וְהִנְנִי עִמְּךָ עַד עוֹלָם.

וּבְעוֹדֶנּוּ מְדַבֵּר, וָאֶמָּצֵא עַל הָאָרֶץ, וָאֹמַר: קַדְמוֹן, אֵינֶנִּי עוֹד בַּכָּבוֹד אֲשֶׁר הָיִיתִי בּוֹ לְמַעְלָה, וַאֲשֶׁר בְּלִבִּי חָפַצְתִּי לְהָבִין לֹא הֵבַנְתִּי.

וַיֹּאמֶר אֵלַי: אֶת מַאֲוַיֵּי לִבְּךָ אַגִּיד לְךָ.

הִגַּדְתָּ לָדַעַת אֶת הַנֶּגַע הָעֲשִׂירִי אֲשֶׁר עַל הַגּוֹיִם, הֲכִינֹתִי וְהִזְמַנְתִּי, אַחֲרֵי כְלוֹת שְׁנֵים עָשָׂר שָׁעוֹת עַל הָאָרֶץ.

שְׁמַע אֶת אֲשֶׁר אַגִּיד לְךָ, יִהְיֶה כֵן: הָרִאשׁוֹנָה — צָרָה מִדֹּחַק רָב.

הַשְּׁנִיָּה — שְׂרֵפוֹת אֵשׁ בֶּעָרִים.

הַשְּׁלִישִׁית — אֲבַדּוֹן הַיָּם בַּבְּהֵמָה.

הָרְבִיעִית — רָעָב לְתוֹשְׁבֵי מִשְׁפַּחְתָּם.

הַחֲמִישִׁית — בְּרַעַשׁ וּבְחֶרֶב אֲבַדּוֹן.

הַשִּׁשִּׁית — בָּרָד וְרֹב שֶׁלֶג.

הַשְּׁבִיעִית — חַיַּת שָׂדֶה תִּהְיֶה קִבְרָם.

הַשְּׁמִינִית — רָעָב וָדֶבֶר יָבִיאוּ כִלָּיוֹן.

הַתְּשִׁיעִית — עֹנֶשׁ חֶרֶב וּמָנוֹס בְּצָרוֹת.

הָעֲשִׂירִית — רַעַם וְקוֹלוֹת וּרְעָשִׁים לְהַשְׁחָתָה.

אָז אֶתְקַע בַּשּׁוֹפָר מִן הָאֲוִיר וְאֶשְׁלַח אֶת בְּחִירִי, וּמִלֵּאתִי בוֹ מִדַּת כָּל כֹּחִי אַחַת, וְהוּא יִקְרָא לְעַמִּי הַנֶּאֱסָף מִן הַגּוֹיִם.

וַאֲנִי אֶשְׂרֹף בָּאֵשׁ אֶת הַמְבַזִּים לָהֶם, הַשּׁוֹלְטִים בָּהֶם לְעוֹלָם, וְאֶתֵּן אֶת הַמְבַזִּים אוֹתִי לְחֶרְפָּה מֵאֵת עוֹלָם הַזֶּה, כִּי אַתָּה הֲכִינוֹתָ לָהֶם מַאֲכַל אֵשׁ שְׁאוֹל, וּמְעוֹף בְּלִי הֶפְסֵק בָּאֲוִיר, תַּחַת הָאָרֶץ, בַּתַּחְתִּיּוֹת.

*[הַטֶּקְסְט הַסְּלָאבִי שֶׁהוּעֲלָה מִסְתַּיֵּם כָּאן (עמ' 53), בְּתוֹךְ נְאוּם הַמַּלְאָךְ עַל הַמַּכּוֹת וְקֵץ הַיָּמִים]*
