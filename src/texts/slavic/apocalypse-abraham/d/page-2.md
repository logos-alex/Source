## קטע ב' — נאום האש־מים־אדמה, שריפת בית תרח, קריאת ה׳, שליחות יהואל
**Л. 56א–57ד**


> כולל תוספותיו הפולמוסיות של מעבד הפליאה — מסומנות ומובחנות מגוף השחזור

> כפי שהתבקש — קטע זה כולל הפעם גם את תוספותיו של מעבד הפליאה (המסומנות במאמר במפורש כ'תוספת אנטי-יהודית פולמוסית'). תוספות אלה מובחנות טיפוגרפית (רקע אפור, כתב נטוי, מתויגות 'תוספת המעבד') מגוף השחזור עצמו, כיוון שאין הן חלק מהנוסח העברי המשוער אלא תוספת ביזנטית-סלאבית מאוחרת (המאה ה-15 ואילך) שהוכנסה לתוך גוף הסיפור הקדום. הן מתורגמות לעברית (לא 'משוחזרות'), ברישום עברי סביר אך פחות מחויב מבחינה מתודולוגית.


*[Л. 56א, המשך]*

וּבַמַּחְשָׁבוֹת הָאֵלֶּה בָּא אַבְרָם אֶל אָבִיו וַיֹּאמֶר תֶּרַח אָבִי, נִכְבָּד הָאֵשׁ מֵאֱלֹהֶיךָ הַזָּהָב וְהַכֶּסֶף וְהָאֶבֶן וְהָעֵץ, כִּי בְּהִשָּׂרְפָם נִכְנָעִים לָאֵשׁ, וְהָאֵשׁ שֹׂחֶקֶת בָּם בְּאָכְלָהּ אֶת אֱלֹהֶיךָ; וְגַם אוֹתָהּ לֹא אֶקְרָא אֱלֹהִים, כִּי נִכְנַעַת הִיא לַמַּיִם, נִכְבָּדִים הַמַּיִם מִמֶּנָּה, כִּי גֹבְרִים הֵם עַל הָאֵשׁ וּמַמְתִּיקִים פְּרִי הָאֲדָמָה; וְגַם אוֹתָם לֹא אֶקְרָא אֱלֹהִים, כִּי הַמַּיִם הוֹלְכִים אֶל הָאָרֶץ, עַל כֵּן אֶקְרָא נִכְבֶּדֶת מֵהֶם הָאָרֶץ, כִּי גֹבֶרֶת הִיא עַל יְסוֹד הַמָּיִם; וְגַם אוֹתָהּ לֹא אֶקְרָא אֱלֹהִים, כִּי הַשֶּׁמֶשׁ מְיַבְּשָׁהּ וּבְנֵי אָדָם עוֹבְדִים אוֹתָהּ לִמְלָאכָה; נִכְבָּד מִן הָאָרֶץ אֶקְרָא אֶת הַשֶּׁמֶשׁ, כִּי בְקַרְנֶיהָ מְאִירָה הִיא אֶת כָּל הָאָרֶץ; וְגַם אוֹתָהּ לֹא אֶקְרָא אֱלֹהִים, כִּי בְּבוֹא הַלַּיְלָה תֶּחְשַׁךְ בָּאֹפֶל; וְאַף לֹא הַיָּרֵחַ וְלֹא הַכּוֹכָבִים אֶקְרָא אֱלֹהִים, כִּי רַק בַּלַּיְלָה יָאִירוּ וּבַיּוֹם יֶחְשַׁךְ אוֹרָם׃ אַךְ שְׁמַע נָא תֶּרַח אָבִי, אַגִּיד לְךָ אֶת הָאֱלֹהִים אֲשֶׁר עָשָׂה אֶת הַכֹּל: הוּא הָאֱלֹהִים הָאֱמֶת אֲשֶׁר צָבַע אֶת הַשָּׁמַיִם, וְהִזְהִיב אֶת הַשֶּׁמֶשׁ, וְהֵאִיר אֶת הַיָּרֵחַ וְאֶת הַכּוֹכָבִים, וְהוֹבִישׁ אֶת הָאָרֶץ בְּתוֹךְ מַיִם רַבִּים, וְאֹתְךָ בָּרָא נָבוֹן, וְאֹתִי עַתָּה דָּרַשׁ לְעוֹרֵר בִּי מַחְשָׁבָה כִּי יֵרָאֶה לָנוּ הָאֱלֹהִים בְּעַצְמוֹ׃ וְכָזֹאת הָיָה בְּדַבֵּר אַבְרָם אֶת אֵלֶּה אֶל תֶּרַח אָבִיו בַּחֲצַר בֵּיתוֹ׃


### תוספת המעבד
*Л. 56ב-ג*

> הֲתָבִין אֶת אֲשֶׁר נִבָּא אַבְרָם הַנִּכְבָּד בְּאָמְרוֹ כִּי יֵרָאֶה לָנוּ הָאֱלֹהִים בְּעַצְמוֹ? מַה פֵּשֶׁר "יֵרָאֶה בְּעַצְמוֹ"? אִם לֹא תֵדַע, שְׁתֹק! פֵּשֶׁר הַדָּבָר: אַבְרָם, בְּצַפּוֹתוֹ לְלֵדַת הַמָּשִׁיחַ, רָאָה מֵרֹאשׁ בְּעֵינֵי רוּחוֹ אֵת כָּל אֲשֶׁר הָיָה עָתִיד לִהְיוֹת, וְכֵן הָיָה׃


*[Л. 56ג]*

וַיְהִי כִּשְׁמֹעַ תֶּרַח אָבִיו אֶת אֵלֶּה וַיַּבֵּט בְּעֵינַיִם כְּבֵדוֹת אֶל אַבְרָם, כִּי לֹא יָשְׁרוּ בְעֵינָיו דִּבְרֵי אַבְרָם, וַיֶּאֱהַב אֶת הַשֶּׁקֶר אֲשֶׁר מָסַר לוֹ נָחוֹר אָבִיו׃ וַיֹּאמֶר אַבְרָם בְּלִבּוֹ אֶבְחַן אֶת אֱלֹהֵי אָבִי אִם יוּכְלוּ לְעָזְרוֹ׃ וַיִּקַּח אֵשׁ וַיַּצֵּת אַבְרָם אֶת הַבַּיִת אֲשֶׁר עָמְדוּ שָׁם אֱלֹהֵי אָבִיו׃ וַיַּרְא אָרָן אֲחִי אַבְרָם, וַיְבַקֵּשׁ לְהַצִּיל אֶת הָאֱלֹהִים וַיֹּצֵא אֹתָם, וְהוּא נִשְׂרַף; "וַיָּמָת הָרָן עַל פְּנֵי תֶּרַח אָבִיו" (בראשית יא, כח)׃


*[Л. 56ד]*

וַיֶּאֱהַב ה׳ אֱלֹהִים אֶת אַבְרָם, וַיֹּאמֶר אֱלֹהִים אֶל אַבְרָם אַבְרָם אַבְרָם, בִּקַּשְׁתָּ אֶת אֱלֹהֵי הָאֱלֹהִים, צֵא מִבֵּית אָבִיךָ וְלֵךְ אֶל הָאָרֶץ אֲשֶׁר אַרְאֶךָּ, וְאֶעֶשְׂךָ לְגוֹי גָּדוֹל וְנִבְרְכוּ בְךָ כֹּל מִשְׁפְּחוֹת הָאֲדָמָה׃


### תוספת המעבד
*Л. 56ד, המשך*

> הֲתָבִין מַה פֵּשֶׁר "וְנִבְרְכוּ בְךָ כֹּל מִשְׁפְּחוֹת הָאֲדָמָה"? יַעַן כִּי אֲדוֹנֵנוּ יֵשׁוּעַ הַמָּשִׁיחַ הוֹאִיל לְהִוָּלֵד מִשֵּׁבֶט אַבְרָם, מִבַּת יוֹיָקִים, לֹא בְאַבְרָם עַצְמוֹ יִתְבָּרְכוּ, כִּי אִם בֵּאלֹהִים אֲשֶׁר עָשָׂה שָׁמַיִם וָאָרֶץ׃ כִּי אֵין אֱמוּנָה שֶׁיַּאֲמִינוּ בְאַבְרָם, אַךְ גַּם אַבְרָם נִתְבָּרֵךְ בְּשֵׁם יֵשׁוּעַ, כִּי מִשִּׁבְטוֹ יָצְאָה הַבְּתוּלָה הַטְּהוֹרָה אֲשֶׁר מִמֶּנָּה נוֹלַד יֵשׁוּעַ, אֲשֶׁר בִּשְׁמוֹ נִתְבָּרְכוּ הָעַמִּים וְנִטְבְּלוּ׃ וְעַתָּה הָבֵן: אֲשֶׁר נִבָּא אַבְרָם הַנִּכְבָּד בְּאָמְרוֹ יֵרָאֶה לָנוּ הָאֱלֹהִים בְּעַצְמוֹ — זֶה הוּא אֲשֶׁר נִתְקַיֵּם׃


*[Л. 56ד–57א]*

וַיְהִי בְּדַבְּרִי לִפְנֵי תֶרַח אָבִי בְּבֵיתוֹ, וְקוֹל הֶחָזָק נָפַל מִן הַשָּׁמַיִם בְּזֶרֶם אֵשׁ, קוֹרֵא וְאוֹמֵר אַבְרָם אַבְרָם, וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר אֵל אֱלֹהִים יהוה [1] אַתָּה מְבַקֵּשׁ בְּשִׂכְלְךָ וּבְמַחְשֶׁבֶת לִבְּךָ, אֲנִי הוּא אֲשֶׁר אַתָּה מְבַקֵּשׁ; צֵא מִבֵּית תֶּרַח אָבִיךָ פֶּן תּוּמַת בַּחֲטָאָיו בְּבֵית אָבִיךָ׃ וָאֵלֵךְ וּבְטֶרֶם יָצָאתִי מִשַּׁעַר הֶחָצֵר, וְהִנֵּה קוֹל רַעַם נוֹרָא וַתֵּרֶד אֵשׁ מִן הַשָּׁמַיִם וַתִּשְׂרֹף אֶת הַבַּיִת וְאֵת כָּל אֲשֶׁר בּוֹ עַד אַרְבָּעִים אַמָּה בָּאָרֶץ׃


*[Л. 57א]*

וְאַחֲרֵי כֵן קָרָא הַקּוֹל אֵלַי פַּעֲמַיִם לֵאמֹר אַבְרָם אַבְרָם, וָאַעַן הִנְנִי אֲדֹנָי׃ וַיֹּאמֶר אֵלַי אֲנִי הוּא, אַל תִּירָא, כִּי אֲנִי קֶדֶם עוֹלָם וְשַׁדָּי [2]; אֲנִי בָּרָאתִי בָרֵאשִׁית שָׁמַיִם וָאָרֶץ וְאֶת הַמָּאוֹר הָרִאשׁוֹן וְאֶת הָאוֹר וְאֶת הָעוֹלָמִים; אֲנִי לְךָ מָגֵן וְעֵזֶר; לֵךְ קַח לְךָ עֶגְלָה מְשֻׁלֶּשֶׁת וְאַיִל מְשֻׁלָּשׁ וְעֵז מְשֻׁלֶּשֶׁת וְתֹר וְגוֹזָל, וְהַקְרֵב לִי קָרְבָּן טָהוֹר אֲשֶׁר אֲצַוְּךָ לִשְׁמֹר לְעוֹלָם׃ וְתִרְאֶה גְדוֹלוֹת אֲשֶׁר לֹא רָאִיתָ, וְיַעַן אֲשֶׁר אָהַבְתָּ לְבַקְּשֵׁנִי קָרָאתִי לְךָ יְדִידִי׃ אַךְ הִנָּזֵר מִכָּל מַאֲכָל עָשׂוּי בָּאֵשׁ וּמִשְּׁתִיַּת יַיִן וּמִסִּיכָה אַרְבָּעִים יוֹם, וְרַק אַחֲרֵי כֵן תַּקְרִיב לִי אֶת הַקָּרְבָּן אֲשֶׁר צִוִּיתִי, בַּמָּקוֹם אֲשֶׁר אַרְאֶךָּ עַל הָהָר הַגָּבֹהַּ, וְשָׁם אַרְאֲךָ אֶת מַעֲשֵׂי דְבָרִי, הַנְּכוֹנִים וְהַנִּגְאָלִים, וְאַגִּיד לְךָ אֶת אֲשֶׁר מְחַכֶּה לְעֹשֵׂי הָרַע וְהַצֶּדֶק בְּמִין הָאָדָם׃


*[Л. 57ב]*

וַיְהִי בְּעוֹד אֲנִי שֹׁמֵעַ אֶת הַקּוֹל מְדַבֵּר אֵלַי אֶת הַדְּבָרִים הָאֵלֶּה, וָאַבֵּט סָבִיב וְאֵין נִשְׁמַת אָדָם, וַתֶּחֱרַד רוּחִי וַתָּמָת נַפְשִׁי, וָאֱהִי כְּאֶבֶן וָאֶפֹּל עַל פָּנַי אַרְצָה, כִּי לֹא הָיָה בִי כֹּחַ לַעֲמֹד עַל הָאָרֶץ׃ וַיְהִי בְּעוֹדֶנִּי שֹׁכֵב עַל פָּנַי אַרְצָה, וָאֶשְׁמַע אֶת קוֹל הַקָּדוֹשׁ אֹמֵר לֵךְ יָהוֹאֵל [3] שֵׁם־בְּשֵׁם, בְּכֹחַ שְׁמִי הַנִּפְלָא קַדֵּשׁ לִי אֶת הָאִישׁ הַזֶּה וְהַצִּילֵהוּ מִפַּחְדּוֹ׃

בִּדְמוּת אִישׁ, וַיַּחֲזֵק בְּיָדִי הַיְמָנִית וַיַּעֲמִידֵנִי עַל רַגְלַי וַיֹּאמֶר אֵלַי קוּם אַבְרָם יְדִיד הָאֱלֹהִים אֲשֶׁר אֲהֵבְךָ׃ בִּהְיוֹתְךָ אָחוּז בְּפַחַד אֱנוֹשׁ, שֻׁלַּחְתִּי אֵלֶיךָ לְחַזְּקֶךָ וּלְבָרֶכְךָ בְּשֵׁם הָאֱלֹהִים אֲשֶׁר אֲהֵבְךָ, בּוֹרֵא שָׁמַיִם וָאָרֶץ׃ חֲזַק וֶאֱמַץ וּמַהֵר אֵלָיו׃ יָהוֹאֵל קְרָאַנִי מִי אֲשֶׁר יֶחֶרְדוּ מִפָּנָיו הָעֹמְדִים עִמִּי בַּשָּׁמַיִם הַשְּׁבִיעִי מֵעַל הָרָקִיעַ, וְכֹחַ הַצֶּדֶק הֶחָתוּם בַּשֵּׁם שׁוֹכֵן בִּי׃ אֲנִי לְבַדִּי הוּא אֲשֶׁר נִתַּן לוֹ לְעַצֵּר, כִּפְקֻדָּתוֹ, אֶת הַחַיּוֹת הַכְּרוּבִים מֵאֵיבָה אִישׁ בְּרֵעֵהוּ, וּלְלַמֵּד אֶת נֹשְׂאָיו אֶת שִׁירַת הַשָּׁעָה הַשְּׁבִיעִית לְעֶרֶב כְּמִנְיַן בְּנֵי אָדָם׃ עָלַי הֻפְקַד לִבְלֹם אֶת הַלִּוְיָתָן, כִּי עָלַי לְהַכְנִיעַ כָּל מִתְקוֹמֵם וּמְאַיֵּם מִן הָרֶמֶשׂ אֲשֶׁר בַּמַּיִם׃ אֲנִי אֲשֶׁר נִתַּן לוֹ לִפְתֹּחַ אֶת הַשְּׁאוֹל לְהַשְׁבִּית אֶת עֲבֹדַת הַמֵּתִים׃ אֲנִי אֲשֶׁר צֻוֵּיתִי לִשְׂרֹף אֶת בֵּית אָבִיךָ עִמּוֹ, כִּי לַמֵּתִים כִּבֵּד׃ אֲנִי הַשָּׁלוּחַ אֵלֶיךָ לְבָרֵךְ אֹתְךָ וְאֶת הָאָרֶץ אֲשֶׁר הֵכִין לְךָ קֶדֶם עוֹלָם, אֲשֶׁר אֵלָיו קָרָאתָ, וּבַעֲבוּרְךָ בָּאתִי אֶל הָאָרֶץ׃ קוּם אַבְרָם, לֵךְ, חֲזַק, שְׂמַח וָגִיל, וַאֲנִי עִמְּךָ, כִּי הוּכַן לְךָ כָבוֹד מֵאֵת קֶדֶם עוֹלָם׃ לֵךְ הַקְרֵב אֶת הַקָּרְבָּן כַּמִּצְוָה׃ צֻוֵּיתִי לִהְיוֹת עִמְּךָ וְעִם הַזֶּרַע הֶעָתִיד לָצֵאת מִמְּךָ, וְעִמִּי מִיכָאֵל [4] יְבָרֶכְךָ עַד עוֹלָם׃ חֲזַק וָלֵךְ׃


*[Л. 57ג–ד]*

וָאָקוּם וָאֵרֶא אֶת הָאוֹחֵז בִּימִינִי וְהַמַּעֲמִידֵנִי עַל רַגְלַי: מַרְאֵה גוּפוֹ כְּסַפִּיר, וּפָנָיו כְּתַרְשִׁישׁ, וּשְׂעַר רֹאשׁוֹ כַּשֶּׁלֶג, וּמִצְנַפְתּוֹ כְּמַרְאֵה עָנָן, וּלְבוּשׁוֹ אַרְגָּמָן, וּבִימִינוֹ שַׁרְבִיט זָהָב׃ וַיֹּאמֶר אֵלַי אַבְרָם, וָאֹמַר הִנְנִי עַבְדֶּךָ׃ וַיֹּאמֶר אַל יְבַהֶלְךָ מַרְאִי וְאַל תֵּחַת נַפְשְׁךָ מִדְּבָרִי׃ לֵךְ עִמִּי וַאֲנִי הֹלֵךְ עִמָּךְ; עַד הַקָּרְבָּן אֶרָאֶה, וְאַחֲרָיו לֹא אֵרָאֶה עַד עוֹלָם׃ חֲזַק וָלֵךְ׃

## הערות

**[1]** אֵל אֱלֹהִים יהוה — כך במפורש קמצ'טנוב-מילקוב (הע' 149): הנוסח הסלאבי "Бога, богов и Творца" תואם לנוסחת תה' נ, א / יהו' כב, כב (אֵל אֱלֹהִים יְהוָה), כשהתרגום מחליף בכבוד את הטטרגרמטון ב"בורא" (Творца).

**[2]** קֶדֶם עוֹלָם וְשַׁדָּי — כך במפורש קמצ'טנוב-מילקוב (הע' 152): הסלאבית "пре вѣкъ и крѣпо" משקפת עברית שַׁדַּי; ההגדרות הופכות בהמשך הטקסט לכינויי-קבע של האל.

**[3]** יָהוֹאֵל — כך במפורש קמצ'טנוב-מילקוב (הע' 158), בעקבות אוספנסקי: עברית Yaho'el/Yao'el, צירוף שמות הקודש יהוה ו-אל.

**[4]** מִיכָאֵל — שם קבוע במקרא (דנ' י, יג ועוד); כאן שר-צבא מלאכי משני ליהואל.
