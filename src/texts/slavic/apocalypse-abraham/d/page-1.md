## קטע א' — סיפור האלילים — מרמות, בר־אשתא
**Л. 54–56א**


> השחזור הבא מבוסס אך ורק על נוסח היסוד של כ״י בארסוב 620, כפי שנדפס במאמרם של קמצ׳טנוב ומילקוב (2021), ללא שילוב חילופי הנוסח מן העדים המקבילים (קול׳, סול׳, סינ׳, סילב׳) — אלה הושארו לשלב מאוחר יותר, כפי שסוכם. הטקסט מחולק לפי חתכי העמודים בכתב־היד (א/ב/ג/ד בכל דף), ומופיע לפני כל קטע ציון הדף הרלוונטי, לצורך אימות והשוואה. הערות בסוגריים מפנות לרשימת ההערות בסוף הקובץ, ובעיקר עוסקות בזיהוי שמות האלילים.


*[Л. 54א–ב]*

אֵלֶּה הַדְּבָרִים אֲשֶׁר הָגָה אַבְרָם בְּלִבּוֹ בַּיּוֹם אֲשֶׁר שִׁבַּר אֶת אֱלֹהֵי תֶרַח אָבִיו וְנָחוֹר אָחִיו׃ בְּבָחֳנִי מִי בָאֱלֹהִים הֶחָזָק, וַאֲנִי אַבְרָם בַּעֲבֹד אֶת עֲבֹדָתִי, כְּכַלּוֹתִי לְשָׁרֵת אֶת אֱלֹהֵי תֶרַח אָבִי, אֱלֹהֵי הָעֵץ וְהַנְּחֹשֶׁת וְהַבַּרְזֶל, וָאָבוֹא הַהֵיכָלָה לְשָׁרֵת, וָאֶמְצָא אֶת הָאֱלֹהִים הַנִּקְרָא מִרְמוֹת [1], פְּסִיל אֶבֶן, נֹפֵל עַל פָּנָיו אַרְצָה לְרַגְלֵי אֱלֹהֵי הַבַּרְזֶל נָכִי [2]׃

וַיְהִי כִּרְאוֹתִי כֵן וַיִּבָּהֵל לִבִּי, וָאֹמַר בְּלִבִּי כִּי לְבַדִּי לֹא אוּכַל לַהֲשִׁיבוֹ אֶל מְקוֹמוֹ, כִּי כָבֵד הוּא, פָּסוּל מֵאֶבֶן גְּדוֹלָה׃ וָאֵלֵךְ וָאַגֵּד לְאָבִי. וַיֵּלֶךְ עִמִּי, וַיְהִי כַּאֲשֶׁר הֵחֵלּוֹנוּ לַהֲסִיעוֹ לַהֲשִׁיבוֹ אֶל מְקוֹמוֹ וַתִּפֹּל רֹאשׁוֹ מֵעָלָיו׃


*[Л. 54ג]*

וְכַאֲשֶׁר נָשָׂאתִי אֶת הָרֹאשׁ, וַיַּרְא אָבִי כִּי נָפְלָה רֹאשׁ מִרְמוֹת מֵעָלָיו, וַיֹּאמֶר אֵלַי אַבְרָם, וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר הָבִיאָה לִּי מִן הַבַּיִת מַפְסֶלֶת קְטַנָּה. וָאָבִיא, וַיִּכְרֹת רֹאשׁ מֵאֱלֹהִים אַחֵר שֶׁל אֶבֶן וַיְדַבְּקֶהָ אֶל מִרְמוֹת הַנֹּפֵל, וַיִּפְסֹל עוֹד חֲמִשָּׁה אֱלֹהִים אֲחֵרִים וַיִּתְּנֵם לִי וַיְצַוֵּנִי לְמָכְרָם בְּדֶרֶךְ הָעִיר׃


*[Л. 54ד]*

וָאֶחְבֹּשׁ אֶת חֲמוֹר אָבִי וָאָשִׂים אוֹתָם עָלָיו, וָאֵצֵא אֶל הַמָּלוֹן לְמָכְרָם׃ וְהִנֵּה סֹחֲרִים בָּאִים מִפַּנְדָּלָא אֲשֶׁר בַּאֲרָם עַל גְּמַלִּים בְּדֶרֶךְ מִצְרַיִם לִסְחֹר... וַיְהִי בְּדַבְּרִי עִמָּם וַיִּשְׁאַג אֶחָד מִגְּמַלֵּיהֶם, וַיִּבָּעֵת הַחֲמוֹר וַיָּנָס וַיַּפֵּל אֶת הָאֱלֹהִים מֵעָלָיו, וַיִּשָּׁבְרוּ שְׁלֹשָׁה מֵהֶם וּשְׁנַיִם נִשְׁאֲרוּ׃ וַיְהִי כִּשְׁמֹעַ הָאֲרַמִּים כִּי אֱלֹהִים הֵם, וַיֹּאמְרוּ אֵלַי לָמָּה לֹא הִגַּדְתָּ לָּנוּ כִּי אֱלֹהִים הֵם עִמְּךָ, וְקָנִינוּ אֹתָם בְּטֶרֶם שָׁמַע הַחֲמוֹר אֶת שַׁאֲגַת הַגָּמָל, וְלֹא הָיָה הֶפְסֵד; תְּנָה לָּנוּ אֶת הַנִּשְׁאָרִים וְנִתֵּן לְךָ מְחִיר הָגוּן׃ וַיִּתְּנוּ לִי מְחִיר הַנִּשְׁאָרִים, וָאֶתְעַצֵּב בְּלִבִּי עַל הַשְּׁבוּרִים אֵיךְ אָבִיא לְאָבִי אֶת הַמֶּחִיר הַזֶּה׃ וְאֶת שִׁבְרֵי הַשְּׁבוּרִים הִשְׁלַכְתִּי בְּמֵי נְהַר גּוּר [3] הַהֹלֵךְ בַּמָּקוֹם הַהוּא, וַיִּטְבְּעוּ וְלֹא נִרְאוּ עוֹד׃

וַיְהִי בְּשׁוּבִי וַיֶּהֱמֶה לִבִּי וּמַחְשְׁבֹתַי נָבֹכוּ, וָאֹמַר בְּלִבִּי מַה־הָרָעָה אֲשֶׁר עֹשֶׂה אָבִי׃ הֲלֹא גָדוֹל מֵהֶם מִי אֲשֶׁר בְּחָכְמָתוֹ נִפְסְלוּ וְהוּקְמוּ? הֲלֹא טוֹב לְהִשְׁתַּחֲוֹת לְאָבִי, כִּי הֵם מַעֲשֵׂה יָדָיו? הִנֵּה מִרְמוֹת נָפַל וְלֹא יוּכַל קוּם בְּהֵיכָלוֹ, וַאֲנִי לְבַדִּי לֹא יָכֹלְתִּי הֲקִימוֹ עַד בֹּא אָבִי וַנְּקִימֵהוּ יַחְדָּו, וּבְעָמְלֵנוּ נָפְלָה רֹאשׁוֹ מֵעָלָיו, וַיָּשֶׂם לוֹ רֹאשׁ אֶבֶן אַחֶרֶת, וְאוֹתוֹ עָזַב בְּלֹא רֹאשׁ, וְעוֹד עָשָׂה חֲמִשָּׁה אֱלֹהִים הַנֹּפְלִים מֵעַל הַחֲמוֹר; וְלֹא יָכְלוּ לְהוֹשִׁיעַ אֶת נַפְשָׁם, וְלֹא לְהַעֲנִישׁ אֶת הַחֲמוֹר עַל אֲשֶׁר הִפִּילָם, וְלֹא יָצְאוּ שִׁבְרֵיהֶם מִן הַנָּהָר׃ וָאֹמַר בְּלִבִּי אִם כֵּן הוּא, לֹא יוּכַל מִרְמוֹת אֱלֹהֵי אָבִי, אֲשֶׁר רֹאשׁוֹ פְּסוּל מֵאֶבֶן אַחֶרֶת, לְהוֹשִׁיעַ אָדָם אוֹ לִשְׁמֹעַ תְּפִלַּת אָדָם׃


*[Л. 55א]*

וּבַמַּחְשָׁבוֹת הָאֵלֶּה שַׁבְתִּי אֶל בֵּית אָבִי, וָאַשְׁקֶה אֶת הַחֲמוֹר וָאֶתֵּן לוֹ תֶּבֶן, וָאוֹצִיא אֶת הַכֶּסֶף וָאֶתְּנֵהוּ בְּיַד תֶּרַח אָבִי׃ וַיְהִי כִּרְאוֹתוֹ וַיִּשְׂמַח וַיֹּאמֶר רָצוּי אַתָּה אַבְרָם לֵאלֹהַי, כִּי הֵבֵאתָ מְחִיר הָאֱלֹהִים וְלֹא הָיָה עֲמָלִי לָרִיק׃ וָאַעַן וָאֹמַר לוֹ שְׁמַע נָא תֶּרַח אָבִי, הָאֱלֹהִים הֵמָּה חַיָּבִים לְהוֹדוֹת לְךָ, כִּי אַתָּה אֱלֹהֵיהֶם, כִּי אַתָּה עֲשִׂיתָם, וְחַסְדָּם מָוֶת וְכֹחָם הֶבֶל; אֲשֶׁר לֹא הוֹשִׁיעוּ אֶת נַפְשָׁם, אֵיךְ יְבָרְכוּ אֹתְךָ אוֹ אֹתִי? בִּגְלָלִי קִבַּלְתָּ אֶת זֶה, כִּי בְּשִׂכְלִי הֵבֵאתִי לְךָ כֶּסֶף בְּעַד הָאֱלֹהִים הַשְּׁבוּרִים׃ וַיְהִי כְּשָׁמְעוֹ אֶת דְּבָרַי וַיִּחַר אַפּוֹ בִּי מְאֹד, כִּי דִבַּרְתִּי דְּבָרִים קָשִׁים עַל אֱלֹהָיו, וַאֲנִי בִּרְאוֹתִי אֶת חֲמָתוֹ וָאֵצֵא הַחוּצָה׃


*[Л. 55ב]*

וְאַחֲרֵי כֵן קָרָא לִי וַיֹּאמֶר אַבְרָם, וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר קְחוּ קֵיסָם מֵעֲצֵי הַבְּרוֹשׁ אֲשֶׁר עָשִׂיתִי מֵהֶם אֱלֹהִים בְּטֶרֶם בֹּאֲךָ, וְנַעֲשֶׂה אֹכֶל לָאֲרוּחָה׃ וַיְהִי בְּלָקְטִי אֶת הַקֵּיסָמִים וָאֶמְצָא בֵּינֵיהֶם אֵל קָטֹן שֻׂם בֵּין הָעֵצִים מִשְּׂמֹאלִי, וְעַל מִצְחוֹ כָּתוּב אֱלֹהִים בַּר־אִשָּׁתָא [4]׃ וּבְמָצְאִי אֹתוֹ לֹא הִגַּדְתִּי לְאָבִי כִּי מָצָאתִי בֵּין הַקֵּיסָמִים אֵת בַּר־אִשָּׁתָא אֱלֹהֵי הָעֵץ, וּבְהַבְעִירִי אֵשׁ לַעֲשׂוֹת אֹכֶל לְאָבִי, וָאָשִׂים אֶת בַּר־אִשָּׁתָא אֵצֶל הַכִּירָה, וָאֹמַר אֵלָיו בְּתֹקֶף בַּר־אִשָּׁתָא רְאֵה פֶּן תִּכְבֶּה הָאֵשׁ עַד בֹּאִי; וְאִם תָּחֵל לִכְבּוֹת, נְפַח בָּהּ וְתִבְעַר, וָאֵצֵא וָאֶתֵּן לוֹ אֶת מִצְוָתִי׃


*[Л. 55ג]*

וּבְשׁוּבִי וָאֵרֶא אֶת בַּר־אִשָּׁתָא נֹפֵל עַל פָּנָיו וְרַגְלָיו אֲחוּזוֹת בָּאֵשׁ, וָאֶרְאֶה כֵן וָאֶשְׂחַק וָאֹמַר בְּלִבִּי אָכֵן בַּר־אִשָּׁתָא יָדַעְתָּ לְהַחֲזִיק בָּאֵשׁ וּלְבַשֵּׁל אֹכֶל׃ וְעוֹד אֲנִי מְדַבֵּר זֹאת בְּלִבִּי וְהוּא הוֹלֵךְ וְנִשְׂרָף לְאַט עַד הֱיוֹתוֹ אֵפֶר׃ וָאָבִיא אֶת הָאֹכֶל לְאָבִי, וַיֹּאכַל, וָאֶתֵּן לוֹ חָלָב וְיַיִן וַיֵּשְׁתְּ, וַיִּשְׂבַּע, וַיְבָרֶךְ אֶת מִרְמוֹת אֱלֹהָיו׃ וָאֹמַר אֵלָיו תֶּרַח אַל תְּבָרֵךְ וְאַל תְּהַלֵּל אֶת אֱלֹהֶיךָ מִרְמוֹת, כִּי אִם הַלֵּל אֶת אֱלֹהֶיךָ בַּר־אִשָּׁתָא, כִּי מֵאַהֲבָתוֹ אֹתְךָ נָפַל אֶל תּוֹךְ הָאֵשׁ לְבַשֵּׁל לְךָ אֹכֶל׃ וַיֹּאמֶר אָבִי אַיֵּה הוּא עַתָּה, וָאֹמַר נִשְׂרַף בְּאֵשׁ גְּדוֹלָה וַיְהִי לְאֵפֶר׃ וַיֹּאמֶר גָּדוֹל כֹּחַ בַּר־אִשָּׁתָא, אֶעֱשֶׂה הַיּוֹם אַחֵר וִיבַשֵּׁל לִי מָחָר אֹכֶל׃


*[Л. 55ד]*

וַאֲנִי כְּשָׁמְעִי אֶת דִּבְרֵי אָבִי וָאֶשְׂחַק בְּלִבִּי וָאֵאָנַח בְּחָרוֹן בְּמָרַת נַפְשִׁי, וָאֹמַר אֵיךְ יִהְיֶה עֹזֵר לְאָבִי גּוּף עָשׂוּי וּפָסוּל בְּיָדוֹ? הֲכִי הִכְנִיעַ אֶת הַגּוּף לְנַפְשׁוֹ, וְהַנֶּפֶשׁ לָרוּחַ, וְהָרוּחַ לְאִוֶּלֶת וּבַעֲרוּת? וָאֹמַר בְּלִבִּי אֲדַבְּרָה עִמּוֹ פַּעַם אֶחָת פָּנִים אֶל פָּנִים וְאַגִּיד לוֹ אֶת מַחְשְׁבוֹתָי׃ וָאַעַן וָאֹמַר תֶּרַח אָבִי, כָּל אֱלֹהִים אֲשֶׁר תְּהַלֵּל תִּתְהוֹלֵל בְּמַחְשְׁבוֹתֶיךָ׃ הִנֵּה אֱלֹהֵי נָחוֹר אָבִיךָ הָעֹמְדִים בַּהֵיכָל הַקָּדוֹשׁ נִכְבָּדִים מֵאֱלֹהֶיךָ; הִנֵּה זַךְ [5] אֱלֹהֵי נָחוֹר אָבִיךָ נִכְבָּד מִמִּרְמוֹת אֱלֹהֶיךָ, כִּי עָשׂוּי זָהָב, עָשׂוּי וּפָסוּל בִּידֵי אָדָם; וְאִם יִבְלֶה בַּיָּמִים יְחֻדַּשׁ, וּמִרְמוֹת אִם יִשָּׁבֵר לֹא יְחֻדַּשׁ עוֹד, כִּי פְּסוּל מֵאֶבֶן הוּא׃ וְכֵן אֱלֹהֵי יוֹאָב [6] הָעֹמֵד עִם זַךְ נִכְבָּד מֵאֱלֹהֵי בַר־אִשָּׁתָא:


*[Л. 56א, פתיחה]*

הוּא עָשׂוּי עֵץ וּמְצֻפֶּה כֶסֶף, נָעִים לְעֵין אָדָם; וֶאֱלֹהֶיךָ בַר־אִשָּׁתָא, בְּטֶרֶם נַעֲשָׂה, הָיָה נָטוּעַ בָּאֲדָמָה, גָּדוֹל וְנִפְלָא בְּעָנְפָיו וּבְצִיצָיו; וְכִי נִכְרַת בְּגַרְזֶן וְנַעֲשָׂה בְאָמָּנוּתְךָ וַיְהִי לֵאלֹהִים; וְהִנֵּה יָבֵשׁ כְּבָר וַיֹּאבַד לַחְלֻחוֹ, וּמִגֹּבַהּ נָפַל אַרְצָה, וּמִגְּדֻלָּה בָּא לְמִזְעָר, וְתָאֳרוֹ יָבֵשׁ, וְהוּא נִשְׂרַף בָּאֵשׁ וַיְהִי לְאֵפֶר, וְאֵינֶנּוּ עוֹד, וְאַתָּה אוֹמֵר תַּעֲשֶׂה הַיּוֹם אַחֵר וִיבַשֵּׁל לְךָ מָחָר אֹכֶל׃ הַנַּח לַמְאַבֵּד אֶת נַפְשׁוֹ שֶׁיֹּאבַד לְנַפְשׁוֹ׃

## הערות

**[1]** מִרְמוֹת — כך גם קמצ׳טנוב-מילקוב עצמם (הע׳ 138, בעקבות Philonenko-Sayar/Philonenko), המפנים לעברית miremoth = "מרמה/הונאה"; הצורה מִרְמוֹת (רבים/מוגדל מן מִרְמָה) משתלבת יפה בתעתיק הסלאבי מָרוּמַאפ.

**[2]** נָכִי — השם הסלאבי (нахи / нахина) אינו מפוענח במאמר. הועלה כאן ללא זיהוי ודאי; אפשרות הראויה לבדיקה נוספת היא שיבוש-תעתיק של נְחֻשְׁתָּן (עי׳ מל״ב יח, ד), בהתחשב בכך שהאליל הזה מתואר מפורשות כאליל "ברזל" (יתכן חילוף חומר-גלם בתרגום).

**[3]** נְהַר גּוּר — תעתיק ישיר של הסלאבית рѣка Гуръ; זהות הנהר אינה ודאית.

**[4]** בַּר־אִשָּׁתָא — כך קמצ׳טנוב-מילקוב (הע׳ 140): מארמית, "אליל עץ", bar ishatha "בן/בני האש". התעתיק הסלאבי варисатъ תואם.

**[5]** זַךְ — כך קמצ׳טנוב-מילקוב (הע׳ 143): משורש עברי zkh, "בריאה טהורה".

**[6]** יוֹאָב — כך במאמר (הע׳ 142: "ишав"); בכת״י אחרים "יאבון", הקרוב יותר לשמות הסלאביים.


## הערות מתודולוגיה

הבחירה בלשון המקרא (ולא בעברית משנאית טהורה) נעשתה במודע: הטקסט הסלאבי עצמו רצוף בניבים מקראיים מובהקים (הנני, ויען ויאמר, וכדומה), התואמים לזמן החיבור המשוער (סמוך ל-70 לסה״נ או המאה השנייה, כמצוין במאמר, עמ׳ 171 בטקסט הרוסי). לא נכללו כאן תוספותיו הפולמוסיות האנטי-יהודיות של מעבד הפליאה עצמו, המסומנות במפורש במנגנון ההערות של המאמר (למשל בהמשך הטקסט, החל מקטע פרשנות הצליבה); רק שכבת הסיפור המקורי משוחזרת. יש לאמת קטע זה מול הצילום/הפקסימיליה במידת האפשר, ולהמשיך משם אל קטע ב׳ (נאום האש-מים-אדמה לתרח).
