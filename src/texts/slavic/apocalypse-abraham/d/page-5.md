## הערות

**[1]** לִוְיָתָן — שם עברי מקראי (איוב מ-מא; תה' קד, כו); נשמר כלשונו כפי שהוא גם בתעתיק הסלאבי леоуиафанъ.


## הערות מתודולוגיה

וַיֹּאמֶר אֵלַי קֶדֶם עוֹלָם הֶחָזָק אַבְרָם, וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר הַבֵּט מִלְמַעְלָה וְהִנֵּה תַחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם לִי וְהַגֵּד לִי אֶת מִסְפָּרָם׃ וְאֶת כֹּחָם אֶתֵּן לְזַרְעֲךָ — עַמִּים וְאָדָם אֲשֶׁר הֻבְדְּלוּ לִי לְנַחֲלָתִי עִם עֲזָאזֵל׃ וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק, כֵּן יְהִי זַרְעִי לְפָנֶיךָ, וְאַל יֶחֱרֶה אַפְּךָ עַל בְּחִירְךָ׃ הִנֵּה מֵאָז לֹא הֱבִיאַנִי, מְחָרֵף אוֹתִי עֲזָאזֵל, וְאֵיךְ אֵין הוּא עַתָּה לְפָנֶיךָ? הִתְחַזֵּק עִמּוֹ! וַיֹּאמֶר אֵלַי הַבֵּט עַתָּה תַּחַת רַגְלֶיךָ אֶל הַמֶּרְחָב, וְהָבֵן אֶת הַבְּרִיאָה הַמְצֻיֶּרֶת מֵרֹאשׁ עַל הַמֶּרְחָב הַזֶּה, אֶת הַבְּרוּאִים אֲשֶׁר בּוֹ, וְאֶת הַמּוּכָנִים אַחֲרָיו לְעוֹלָמִים׃

וָאַבֵּט אֶל הַמֶּרְחָב וָאֵרֶא אֶת דְּמוּת הַשָּׁמַיִם וַאֲשֶׁר בָּהֶם, וְאֶת הָאָרֶץ וּפִרְיָהּ, וְאֶת תְּנוּעָתָהּ, וְאֶת נַפְשׁוֹת אֲשֶׁר בָּהּ, וְאֶת כֹּחַ הָאָדָם וְאֶת רִשְׁעַת נַפְשׁוֹתָם וְאֶת צִדְקָתָם, וְאֶת רֵאשִׁית מַעֲשֵׂיהֶם, וְאֶת הַשְּׁאוֹל וּמְעֻנֶּיהָ, וְאֶת אֲבַדּוֹן אֲשֶׁר בָּהּ׃ וָאֵרֶא שָׁם אֶת הַיָּם וְאִיָּיו, וְאֶת בְּהֶמְתּוֹ וְדָגָיו וְאֶת הַלִּוְיָתָן (1), וְאֶת אֲחִיזָתוֹ וּמִשְׁכָּבוֹ וּמְעוֹנוֹתָיו, וְאֶת הַתֵּבֵל הַנִּשְׁעֶנֶת עָלָיו, וְאֶת נִעְנוּעָיו, וְאֶת מְעוֹנוֹתָיו הָעֲתִידִים לְהֵהָרֵס בִּגְלָלוֹ׃ וָאֵרֶא שָׁם אֶת הַנְּהָרוֹת וּמְקוֹרוֹתָם וְסִבּוּבֵיהֶם, וָאֵרֶא שָׁם אֶת גַּן עֵדֶן וּפִרְיוֹ, וְאֶת הַמַּעְיָנוֹת הַיֹּצְאִים מִמֶּנּוּ אֶת הַנָּהָר, וְאֶת טוּבוֹ וְאֶת פְּרִיחָתוֹ עֹשֵׂי צֶדֶק׃ וָאֵרֶא בּוֹ מַאֲכָל וּמְנוּחָה, וָאֵרֶא שָׁם עַם רַב, אֲנָשִׁים וְנָשִׁים וִילָדִים, מֶחֱצָה מֵהֶם מִצַּד דְּמוּת הַיָּמִין, וְחֶצְיָם מִצַּד דְּמוּת הַשְּׂמֹאל׃

וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק, מַה זֹּאת הַתְּמוּנָה הַבְּרוּאָה הַזֹּאת? וַיֹּאמֶר אֵלַי זֹאת רְצוֹנִי לַנִּמְצָא בָאוֹר, וְנוֹחַ הוּא לְפָנַי; וְאַחַר כֵּן צִוָּה וּבִדְבָרִי הָיוּ, כֹּל אֲשֶׁר גָּזַרְתִּי לִהְיוֹת, אֲשֶׁר כֻּלָּם נִרְשְׁמוּ מֵרֹאשׁ וְעָמְדוּ לְפָנַי בְּטֶרֶם הִבָּרְאָם, כַּאֲשֶׁר רָאִיתָ׃ וָאֹמַר אָדוֹן חָזָק וְקֶדֶם עוֹלָם, מִי הֵם הַנֶּחְשָׁבִים בַּתְּמוּנָה הַזֹּאת מִזֶּה וּמִזֶּה? וַיֹּאמֶר אֵלַי אֵלֶּה מִצַּד הַשְּׂמֹאל — הֲמוֹן הָעַמִּים אֲשֶׁר הָיוּ מִלְּפָנִים וַאֲשֶׁר אַחֲרֶיךָ מְיֹעָדִים, מֵהֶם לְמִשְׁפָּט וּלְתִקּוּן, וּמֵהֶם לְנָקָם וּלְאַבְדָּן בְּקֵץ הָעוֹלָם; וְאֵלֶּה מִצַּד הַיָּמִין הֵם הָעָם הַמֻּבְדָּל לִי מֵעַם עֲזָאזֵל — אֵלֶּה אֲשֶׁר הֲכִינוֹתִי לְהִוָּלֵד מִמְּךָ וְלִקָּרֵא עַמִּי׃

רְאֵה עוֹד בַּתְּמוּנָה מִי הַמֵּסִית אֶת חַוָּה, וּמִי אֹכֵל אֶת פְּרִי הָעֵץ, וְדַע מַה יִּהְיֶה לְזַרְעֲךָ בַּיָּמִים אֲשֶׁר לֹא תוּכַל לְהָבִין, אֲנִי אַגִּיד לְךָ, כִּי נָחוּ פָנַי עָלֶיךָ וְאַגִּיד לְךָ אֶת הַשָּׁמוּר בְּלִבִּי׃ וָאַבֵּט בַּתְּמוּנָה, וַתֵּלַכְנָה עֵינַי אֶל צַד גַּן עֵדֶן, וָאֵרֶא שָׁם אִישׁ גָּדוֹל מְאֹד בְּקוֹמָתוֹ וְנוֹרָא בְרָחְבּוֹ, אֵין דֻּגְמָתוֹ בְּמַרְאֶה, שָׁלוּב עִם אִשָּׁה, וְהִיא דּוֹמָה לוֹ בְּמַרְאֶה וּבְקוֹמָה, וּשְׁנֵיהֶם עֹמְדִים תַּחַת עֵץ אֶחָד מֵעֲצֵי עֵדֶן, וּפְרִי הָעֵץ הַהוּא כְּאֶשְׁכּוֹל גֶּפֶן, וּמֵאֲחוֹרֵי הָעֵץ עוֹמֵד כְּמוֹ נָחָשׁ בַּעַל דְּמוּת, וְיָדַיִם וְרַגְלַיִם לוֹ כְּאָדָם, וּכְנָפַיִם עַל כְּתֵפָיו — שֵׁשׁ מִיָּמִין וְשֵׁשׁ מִשְּׂמֹאל, וְהוּא אוֹחֵז בְּאֶשְׁכּוֹל הָעֵץ וּמַאֲכִיל אֶת שְׁנֵיהֶם׃ וָאֵרֶא אֶת הַשְּׁלוּבִים, וָאֹמַר מִי הֵם הַשְּׁלוּבִים הָאֵלֶּה, וּמִי הוּא הָעוֹמֵד בֵּינֵיהֶם? וַיֹּאמֶר זֶה שֵׂכֶל הָאָדָם, זֶה אָדָם, וְזוֹ מַחְשַׁבְתּוֹ בָּאָרֶץ, הִיא חַוָּה; וַאֲשֶׁר בֵּינֵיהֶם הוּא קְלוֹן הָרֵאשִׁית וְהָאֲבַדּוֹן, הוּא עֲזָאזֵל׃

וָאֹמַר קֶדֶם עוֹלָם, לָמָּה נָתַתָּ לוֹ שִׁלְטוֹן כָּזֶה לְהַשְׁחִית אֶת דּוֹרוֹת בְּנֵי הָאָדָם בְּמַעֲשָׂיו עַל הָאָרֶץ? וַיֹּאמֶר אֵלַי שְׁמַע אַבְרָם: הַחֲפֵצִים בָּרָע וַאֲשֶׁר שָׂנֵאתִי בְּעֹשֵׂיהֶם — עֲלֵיהֶם נָתַתִּי לוֹ שִׁלְטוֹן, וְלִהְיוֹת אֲהוּבִים לוֹ׃ וָאַעַן וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק, לָמָּה חָפַצְתָּ שֶׁיִּהְיֶה הָרַע נֶחְמָד בְּלֵב הָאָדָם, וְהוּא שָׂנוּא לְפָנֶיךָ, וְעַל כֵּן זֶה עֹשֶׂה מַעֲשֶׂה שֶׁאֵין בּוֹ תוֹעֶלֶת? וַיֹּאמֶר אֵלַי כָּל זֶה בִּגְלָלְךָ וּבִגְלַל הַנִּבְדָּלִים מִזַּרְעֲךָ, לְמַעַן תִּרְאֶה בַתְּמוּנָה אֶת הַכָּבֵד עֲלֵיהֶם, וַאֲנִי אַגִּיד לְךָ מַה יִּהְיֶה, וְכָל אֲשֶׁר יִהְיֶה בְּאַחֲרִית הַיָּמִים; רְאֵה עַתָּה בַתְּמוּנָה כָּל אֲשֶׁר רָאִיתָ׃

וָאֵרֶא שָׁם לְפָנַי בַּבְּרִיאָה אֲשֶׁר הָיְתָה: וָאֵרֶא כְּמוֹ אָדָם וְחַוָּה עִמּוֹ, וְעִמָּם הָאוֹיֵב הָעַרְמוּמִי, וְקַיִן הָעוֹשֶׂה עָוֶל הָאוֹיֵב, וְהֶבֶל הַנֶּהֱרָג בִּידֵי עֹשֵׂה הֶעָוֶל, וְאֶת הָאֲבֵדָה הַמּוּבָאָה וְהַנְּתוּנָה עָלָיו׃ וָאֵרֶא שָׁם אֶת הַזְּנוּת וְאֶת הַחוֹמְדִים אוֹתָהּ, וְאֶת טֻמְאָתָהּ וְאֶת קִנְאָתָהּ, וְאֶת אֵשׁ כִּלְיוֹנָם בְּתַחְתִּיּוֹת הָאָרֶץ׃
