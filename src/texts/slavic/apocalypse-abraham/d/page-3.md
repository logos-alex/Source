## קטע ג' — המסע אל חורב, הקרבן, עימות עם עזאזל, שירת השבח
**Л. 57ד–59ב**


*[Л. 57ד–58א]*

וַנֵּלֶךְ שְׁנֵינוּ יַחְדָּו אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה, וְלֹא אָכַלְתִּי לֶחֶם וְלֹא שָׁתִיתִי מַיִם, כִּי מַאֲכָלִי הָיָה מַרְאֵה הַמַּלְאָךְ הַהֹלֵךְ עִמִּי, וּדְבָרוֹ הָיָה מַשְׁקִי׃ וַנָּבֹא אֶל הַר הָאֱלֹהִים הַנִּכְבָּד חוֹרֵב, וָאֹמַר אֶל הַמַּלְאָךְ מְשׁוֹרֵר קֶדֶם עוֹלָם, אֵין בְּיָדִי מְאוּמָה לְקָרְבָּן, וּמִזְבֵּחַ אֵינֶנִּי רֹאֶה בָּהָר; וְאֵיךְ אַקְרִיב קָרְבָּן? וַיֹּאמֶר הַבֵּט אַחֲרֶיךָ׃ וְהִנֵּה אַחֲרֵינוּ הֹלְכִים כָּל הַמְיֹעָדִים לְקָרְבָּן: עֵגֶל וְאַיִל וְעֵז וְתֹר וְגוֹזָל׃ וַיֹּאמֶר אֵלַי הַמַּלְאָךְ אַבְרָם, וָאֹמַר הִנֵּנִי׃ וַיֹּאמֶר אֵלַי קַח אֶת כֻּלָּם, שְׁחָטֵם וּבַתֵּר אֶת הַבְּהֵמוֹת לִשְׁנַיִם עַד תֹּם, וְאֶת הָעוֹף לֹא תְבַתֵּר, וּתְנֵם לָאֲנָשִׁים אֲשֶׁר אַרְאֶה לְךָ עֹמְדִים אֶצְלְךָ, כִּי הֵם הַמִּזְבֵּחַ בָּהָר לְהַקְרִיב קָרְבָּן לְקֶדֶם עוֹלָם׃ וְאֶת הַתֹּר וְהַגּוֹזָל תֵּן לִי, כִּי אֶעֱלֶה בְּכַנְפֵי הָעוֹף לְהַרְאוֹתְךָ בַּשָּׁמַיִם וּבָאָרֶץ וּבַיַּמִּים וּבַתְּהֹמוֹת וּבַמְּעָרוֹת וּבְגַן עֵדֶן וּנְהָרוֹתָיו, וּבְכָל הַתֵּבֵל, וְאֶת יָפְיָהּ תִּרְאֶה בַּכֹּל׃

וָאֶעֱשֶׂה כְּכֹל אֲשֶׁר צִוַּנִי הַמַּלְאָךְ, וָאֶתֵּן לַמַּלְאָכִים הַבָּאִים אֵלֵינוּ אֶת הַבְּהֵמוֹת הַמְבֻתָּרוֹת, וְאֶת הָעוֹפוֹת לָקַח יָהוֹאֵל הַמַּלְאָךְ׃ וָאֲחַכֶּה לְקָרְבַּן הָעֶרֶב, וְהִנֵּה עוֹף טָמֵא יָרַד עַל הַפְּגָרִים וָאֲגָרְשֵׁהוּ׃ וַיֹּאמֶר אֵלַי הָעוֹף הַטָּמֵא מַה תְּבַקֵּשׁ אַבְרָם בַּמְּרוֹמִים הַקְּדוֹשִׁים, אֲשֶׁר שָׁם לֹא יֹאכְלוּ וְלֹא יִשְׁתּוּ וְאֵין מַאֲכַל אָדָם — הָאֵשׁ תֹּאכַל הַכֹּל וְאוֹתְךָ תִּצְרֹב׃


*[Л. 58ב]*

וַיְהִי בְּעוֹד הָעוֹף מְדַבֵּר, וָאֶשְׁאַל אֶת הַמַּלְאָךְ מָה זֶּה אֲדֹנִי׃ וַיַּעַן זֶה קָלוֹן, זֶה עֲזָאזֵל [1]׃ וַיֹּאמֶר אֵלָיו בּוֹשׁ עֲזָאזֵל, כִּי כְּבוֹד אַבְרָם בַּשָּׁמַיִם וּכְבוֹדְךָ בָּאָרֶץ, כִּי בָּחַרְתָּ בָהּ לְמוֹשַׁב טֻמְאָתְךָ וָאֱהַבְתָּהּ; עַל כֵּן שָׂמְךָ קֶדֶם עוֹלָם הֶחָזָק הָאָדוֹן יוֹשֵׁב הָאָרֶץ, וּבְךָ רוּחַ הַשֶּׁקֶר הָרָע בְּכֹל, וּבְךָ חֵמָה וָפֶגַע בְּדוֹר הָרְשָׁעִים; אַךְ לֹא נָתַן קֶדֶם עוֹלָם הֶחָזָק אֶת גּוּפוֹת הַצַּדִּיקִים בְּיָדְךָ, כִּי בָהֶם תִּכּוֹן חַיֵּי צֶדֶק וַאֲבֵדַת רֶשַׁע׃ שְׁמַע נָא, "רֵעַ", בּוֹשׁ מִמֶּנִּי, כִּי לֹא כָל הַצַּדִּיקִים נִתְּנוּ בְיָדְךָ לְנַסּוֹת׃ סוּר מֵעַל הָאִישׁ הַזֶּה, לֹא תוּכַל לְהַסִּיתוֹ, כִּי אוֹיֵב הוּא לְךָ וּלְכָל הֹלְכֵי אַחֲרֶיךָ וְאֹהֲבֵי חֶפְצְךָ׃ הַלְּבוּשׁ אֲשֶׁר הָיָה שֶׁלְּךָ בַּשָּׁמַיִם נִתַּן לוֹ עַתָּה, וְהַכִּלָּיוֹן אֲשֶׁר הָיָה בּוֹ עָבַר אֵלֶיךָ׃


*[Л. 58ג]*

וַיֹּאמֶר אֵלַי הַמַּלְאָךְ אַבְרָם, וָאַעַן הִנְנִי עַבְדֶּךָ׃ וַיֹּאמֶר דַּע מֵעַתָּה כִּי בְךָ בָּחַר קֶדֶם עוֹלָם אֲשֶׁר אָהַבְתָּ, חֲזַק וַעֲשֵׂה בַשִּׁלְטוֹן אֲשֶׁר צִוְּךָ עַל דֹּבְרֵי שֶׁקֶר עַל הַצֶּדֶק׃ וּבְלֹא יָכָלְתִּי לְהַכְלִים אֶת הַמְפַזֵּר בָּאָרֶץ סוֹדוֹת שָׁמַיִם וְהַמּוֹרֵד בְּכָל יָכוֹל, אֱמֹר אֵלָיו הֱיֵה גַחֶלֶת בְּתַנּוּר הַשְּׁאוֹל! לֵךְ עֲזָאזֵל אֶל אֶרֶץ מִדְבָּר; יִעוּדְךָ לִהְיוֹת עַל אֵלֶּה אֲשֶׁר נִבְרְאוּ עִם הַכּוֹכָבִים וְהֶעָבִים, עִם בְּנֵי אָדָם אֲשֶׁר גּוֹרָלָם אַתָּה וְהַחַיִּים בִּזְכוּתְךָ; אֵיבָה הִיא עִקַּר "חֲסִידוּתְךָ"; בְּגִין אֲבַדּוֹנְךָ אָבַד מִלְּפָנַי׃ וָאֲדַבֵּר אֶת הַדְּבָרִים אֲשֶׁר לִמְּדַנִי הַמַּלְאָךְ, וַיֹּאמֶר אַבְרָם, וָאַעַן הִנְנִי עַבְדֶּךָ׃ וַיֹּאמֶר אֵלַי הַמַּלְאָךְ אַל תַּעַן אוֹתוֹ׃ וַיִּקְרָא אֵלַי הַמַּלְאָךְ שֵׁנִית וַיֹּאמֶר מֵעַתָּה, כָּל אֲשֶׁר יְדַבֵּר אֵלֶיךָ, אַל תַּעֲנֵהוּ, פֶּן יִהְיֶה לִרְצוֹנוֹ שְׁלִיטָה בְךָ, כִּי כָל יָכוֹל נָתַן לוֹ עַד עוֹלָם כֹּחַ וּרְצוֹן; אַל תַּעֲנֵהוּ! וְכֵן עָשִׂיתִי כַּאֲשֶׁר צִוַּנִי הַמַּלְאָךְ׃ וְכָל אֲשֶׁר קָרָא לִי לָרֶדֶת אֵלָיו, לֹא עֲנִיתִיו׃


*[Л. 58ד]*

וַיְהִי כְּבוֹא הַשֶּׁמֶשׁ, וְהִנֵּה עָשָׁן כְּעֶשֶׁן הַכִּבְשָׁן, וּמַלְאָכִים מֵעַל עֲשַׁן הַכִּבְשָׁן עָלוּ נֹשְׂאִים אֶת הַקָּרְבָּנוֹת הַמְבֻתָּרִים׃ וַיִּקָּחֵנִי הַמַּלְאָךְ בִּימִינוֹ וַיּוֹשִׁיבֵנִי עַל כְּנַף הַיּוֹנָה הַיְמָנִית, וְהוּא יָשַׁב עַל כְּנַף הַתּוֹר הַשְּׂמָאלִית, אֲשֶׁר לֹא נִשְׁחֲטוּ וְלֹא בֻתְּרוּ׃ וַיַּעֲלֵנִי עַד רֹאשׁ לַהַב הָאֵשׁ, וַנָּעֹף כָּרוּחַ אֶל הַשָּׁמַיִם הַנְּטוּיִם עַל הָרָקִיעַ, וָאֵרֶא מִן הַגֹּבַהּ אֲשֶׁר עָלִינוּ אֵלָיו אוֹר עָצוּם אֲשֶׁר לֹא יְסֻפָּר׃ וְהִנֵּה בָאוֹר הַהוּא רָאִיתִי אֵשׁ בְּעֵרָה לָעָם, וְהָמוֹן עַם רַב הָיוּ מְשַׁנִּים תָּמִיד מַרְאֵיהֶם וְצוּרָתָם, מִשְׁתַּחֲוִים וְקוֹרְאִים בְּלָשׁוֹן אֲשֶׁר לֹא הִכַּרְתִּי׃


*[Л. 59א]*

וָאֹמַר אֶל הַמַּלְאָךְ לָמָּה הֶעֱלִיתַנִי הֵנָּה, כִּי לֹא אוּכַל עוֹד לְהַבִּיט, תַּשְׁתִּי כֹחִי וְרוּחִי סָרָה מִמֶּנִּי׃ וַיֹּאמֶר אֵלַי הִשָּׁאֵר עִמָּדִי, אַל תִּירָא! תַּכִּיר אוֹתוֹ הַהֹלֵךְ יָשָׁר אֵלֵינוּ בְּתוֹךְ קוֹלוֹת הַתְּהִלָּה; הוּא קֶדֶם עוֹלָם אֲשֶׁר אֲהֵבְךָ, אוֹתוֹ עַצְמוֹ לֹא תִרְאֶה, אַךְ אַל יֵחָלַשׁ רוּחֲךָ מִן הַקּוֹלוֹת הָאֵלֶּה, כִּי אֲנִי עִמְּךָ וְאֶעֶזְרֶךָּ׃

וַיְהִי עוֹדֶנּוּ מְדַבֵּר וְהִנֵּה אֵשׁ קָמָה לְנֶגְדֵּנוּ מִסָּבִיב, וְקוֹל מִתּוֹךְ הָאֵשׁ כְּקוֹל מַיִם רַבִּים וְכִשְׁאוֹן יָם נִגְרָשׁ׃ וְהַמַּלְאָךְ אֲשֶׁר עִמִּי נָפַל עַל פָּנָיו וַיִּשְׁתַּחוּ, וַאֲנִי בִּקַּשְׁתִּי לִנְפֹּל עַל פָּנַי וְלֹא הָיָה מָקוֹם לִנְפֹּל עָלָיו, כִּי הָרָמָה אֲשֶׁר עָמַדְנוּ עָלֶיהָ שְׁנֵינוּ עָלְתָה וְיָרְדָה חֲלִיפוֹת׃ וַיֹּאמֶר אַתָּה אַבְרָם, הִשְׁתַּחֲוֵה וְשִׁיר אֶת הַשִּׁירָה אֲשֶׁר לִמַּדְתִּיךָ׃ וָאֶשְׁתַּחוּ וָאָשִׁיר אֶת הַשִּׁירָה אֲשֶׁר לִמְּדַנִי׃ וַיֹּאמֶר שִׁיר בְּלִי חֶדֶל, וָאָשִׁיר וְהוּא שָׁר עִמִּי:


*[Л. 59ב–ג: שירת השבח]*

קֶדֶם עוֹלָם, חָזָק, קָדוֹשׁ אֱלֹהִים,

יְחִיד הַמֶּמְשָׁלָה, יְלִיד עַצְמוֹ, בִּלְתִּי נִשְׁחָת,

טְהוֹר מִכָּל טֻמְאָה, בִּלְתִּי נוֹלָד, תָּמִים, בִּלְתִּי מֵת,

שָׁלֵם בַּכֹּל, מֵאִיר מֵעַצְמוֹ, בִּלְתִּי מֵת,

אֵין לוֹ אָב וְאֵין לוֹ תוֹלָדוֹת [2], רָם, אֵשִׁי,

צַדִּיק, אוֹהֵב אָדָם, נְדִיב חֶסֶד, נוֹתֵן מַתָּנוֹת, מָגִנִּי,

אֶרֶךְ אַפַּיִם, רַב חֶסֶד [3], אֱלֹהַי הַגָּדוֹל,

נֶצַח, חָזָק, קָדוֹשׁ צְבָאוֹת, נִכְבָּד מְאֹד, אֵל אֵל אֵל אֵל [4], יָהוֹאֵל!

אַתָּה אֲשֶׁר אָהֲבָה נַפְשִׁי, שׁוֹמֵר עוֹלָם,

זוֹרֵחַ כָּאֵשׁ, מְאוֹר, קוֹל רָעָם, כְּמַרְאֵה בָרָק,

רוֹאֶה הַכֹּל, מְקַבֵּל תְּפִלַּת מוֹקִירֶיךָ,

וּפוֹנֶה מֵעַל בַּקָּשַׁת הַמַּמְרִים עַל מֶמְשַׁלְתְּךָ בִּתְלוּנוֹתָם,

מַבְדִּיל בֵּין הַצַּדִּיקִים לָרְשָׁעִים הַמְעֹרָבִים בָּעוֹלָם הַכָּלֶה הַזֶּה,

וּמְחַדֵּשׁ אֶת דּוֹר הַצַּדִּיקִים׃

אַתָּה, הָאוֹר, בְּאוֹר הַשַּׁחַר תָּאִיר עַל בְּרִיאָתְךָ —

מִפָּנֶיךָ גַּם בָּאָרֶץ יוֹם,

וּבִמְעוֹנוֹת שְׁמֵי מְרוֹמֶיךָ אֵין צֹרֶךְ בְּאוֹר אַחֵר

מִלְּבַד זֹהַר הָאוֹר אֲשֶׁר לֹא יְסֻפָּר מִפָּנֶיךָ׃

קַבֵּל תְּפִלָּתִי לְרָצוֹן,

וְגַם אֶת הַקָּרְבָּן אֲשֶׁר הֵכִין לְעַצְמוֹ

בְּעַדִי לְמַעַן מְבַקְשֶׁיךָ׃

## הערות

**[1]** עֲזָאזֵל — שם עברי מקראי (ויק' טז), נשמר כלשונו; זהותו כאן כמלאך נופל/שד-מדבר תואמת למסורת המשתקפת בס' חנוך.

**[2]** אֵין לוֹ אָב וְאֵין לוֹ תוֹלָדוֹת — בסלאבית везроне (הע' 187 במאמר, ללא פענוח ודאי); תרגום לפי ההקשר.

**[3]** אֶרֶךְ אַפַּיִם רַב חֶסֶד — כך במפורש קמצ'טנוב-מילקוב (הע' 188), בהשוואה לשמ' לד, ו.

**[4]** אֵל אֵל אֵל אֵל — כך בסילוו' (הע' 189 במאמר): שמעני, ריבוע-שם המכוון לטטרגרמטון, בעל אופי לחשי/השבעתי.
