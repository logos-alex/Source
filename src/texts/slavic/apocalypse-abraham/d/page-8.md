## קטע ח' — התוספת הפולמוסית הגדולה; חתימת החיבור — עשר המכות בפירוט
**Л. 63א–64ג**


> אזהרת תוכן: הקטע הבא הוא תוספת פולמוסית אנטי-יהודית חריפה וארוכה (פנייה ישירה 'יהודי ארור') — פרשנות כריסטולוגית מפורשת של החיבור, שאין לה כל זיקה לנוסח היהודי הקדום. הובאה כאן בתרגום מלא לבקשת המשתמש, למטרות תיעוד מדעי-פילולוגי מלא של שכבות הנוסח, ומסומנת בבירור כתוספת (רקע אפור, כתב נטוי) הנפרדת לחלוטין משכבת השחזור העברי.


### תוספת המעבד
*Л. 63א–ד*

> הֲתָבִין, יְהוּדִי אָרוּר, מַה בִּשֵּׂר הָאֱלֹהִים לְזָקֵנְךָ אַבְרָם בְּאָמְרוֹ "יֵצֵא אִישׁ מִשִּׁבְטְךָ וְרַבִּים יִכָּשְׁלוּ בּוֹ" — כִּי הָאִישׁ הַזֶּה הוּא בֶּן הָאֱלֹהִים מִן הַגְּבִירָה הַטְּהוֹרָה הַבְּתוּלָה מִרְיָם מִשֵּׁבֶט אַבְרָם, כַּאֲשֶׁר נֹאמַר בְּהֶמְשֵׁךְ שַׁלְשֶׁלֶת הַיּוֹחֲסִין? וּמִלִּים "רַבִּים יִכָּשְׁלוּ בּוֹ" — עֲלֵיכֶם הֵן, כִּי נִכְשַׁלְתֶּם, לֹא הִכַּרְתֶּם אֶת בֶּן הָאֱלֹהִים׃ וּמִי הֵם "אֲשֶׁר יַבְזֶה"? אַתֶּם, הַנְּתוּנִים בְּעַבְדוּת הַגּוֹיִם׃ וּמִי הֵם הָאֲחֵרִים, אֲשֶׁר עֲלֵיהֶם נֶאֱמַר "לַמִּשְׁתַּחֲוִים לוֹ יַגְדִּיל"? אֲנַחְנוּ הַנּוֹצְרִים, כִּי אָנוּ מִשְׁתַּחֲוִים לוֹ — לָאָב, וְלַבֵּן שֶׁבָּא לָעוֹלָם, וּלְרוּחַ הַקֹּדֶשׁ אֲשֶׁר שָׁרְתָה עַל הַשְּׁלִיחִים הַקְּדוֹשִׁים, וְכֻלָּם יַחְדָּו נְכַבֵּד בֵּאלֹהוּת אַחַת בְּלֹא הִפָּרְדוּת וּבְלֹא הִבָּדְלוּת׃ "וּמִן הַגּוֹיִם רַבִּים יִבְטְחוּ בּוֹ" — רַבִּים מִן הַגּוֹיִם נִטְבְּלוּ בְּשֵׁם הָאָב וְהַבֵּן וְרוּחַ הַקֹּדֶשׁ, חֲמִשִּׁים וְאֶחָד עַמִּים, וְעֶשְׂרִים וְאֶחָד נִשְׁאֲרוּ, כִּי "שְׁלַחְתִּיו וְחַיֵּי הַצֶּדֶק יִרְבּוּ"׃ כַּאֲשֶׁר נִשְׁלַח הַבֵּן מֵאֵת הָאָב, אָז הֵחֵלָּה לִגְדֹּל הָאֱמוּנָה הַיְשָׁרָה אֲשֶׁר מֵאָז בִּשְּׂרוּ הַנְּבִיאִים׃ וְאַחַר כֵּן אָמַר "יָבֹא מִשְׁפָּטִי עַל צֶאֱצָאֵי עַמְּךָ הָעַקְשִׁים" — וְזֶה נִתְקַיֵּם בְּמִשְׁפַּט הָאֱלֹהִים: בָּא אַסְפַּסְיָנוּס וְשָׁבָה אֶת אֶרֶץ יְרוּשָׁלַיִם וְהַגָּלִיל וּשְׁאָר הָאֲרָצוֹת׃ וּבְאָמְרוֹ "בַּיָּמִים הָהֵם אָבִיא עַל כָּל הַבְּרִיאָה עֶשֶׂר מַכּוֹת רָעוֹת, וּמַחֲלָה וַאֲנָחָה וְצָרַת נַפְשָׁם" — גַּם זֶה נִתְקַיֵּם כִּדְבָרָיו׃ כַּאֲשֶׁר שָׁב אַסְפַּסְיָנוּס לִמְלֹךְ בְּרוֹמָא, הֲלֹא שָׁלַח אֶת בְּנוֹ טִיטוּס עַל יְרוּשָׁלַיִם לְלָכְדָהּ וְלִשְׁבּוֹת אֶת שְׁאֵרִית תּוֹשָׁבֶיהָ? הֲלֹא נִתְקַיֵּם בָּכֶם דְּבַר ה': "בְּצָרַת נַפְשְׁכֶם נְפוֹצוֹתֶם"? וְעוֹד אָמַר "כָּל זֹאת אֶתֵּן אֶתְכֶם, כִּי הִכְעַסְתֶּם אֶת שֵׁם הָאֱלֹהִים" — אֶת בַּר־אַבָּא הַשּׁוֹדֵד בִּקַּשְׁתֶּם מִפִּילָטוֹס וְשִׁחְרַרְתֶּם, וְתַחַת הַשּׁוֹדֵד תְּלִיתֶם עַל הַצְּלָב אֶת יֵשׁוּעַ הַנִּקְרָא מָשִׁיחַ, מוֹשִׁיעַ עוֹלַמְכֶם׃ וּדְבָרָיו "אָז יִשָּׁמְרוּ בְיָדִי צַדִּיקֵי זַרְעֲךָ הַשּׁוֹאֲפִים לִכְבוֹד שְׁמִי" גַּם הֵם נִתְקַיְּמוּ: פֶּטְרוֹס וּפָאבְלוֹס, שְׁנֵי הַשְּׁלִיחִים הָרָאשִׁים הַגְּדוֹלִים, וְגַם יוֹחָנָן וּמַתְיָה וְלוּקָס וּמַרְקוֹס (אֵלֶּה אַרְבַּעַת מְבַשְּׂרֵי הַבְּשׂוֹרָה), אַנְדְּרֵאס וְשִׁמְעוֹן, יַעֲקֹב וּבַר־תַּלְמַי, תּוֹמָא וּפִילִיפּוֹס, וּשְׁאָר שִׁבְעִים הַתַּלְמִידִים — כֻּלָּם צֶאֱצָאֵי אַבְרָם, כַּאֲשֶׁר אָמַר עָלָיו פָּאבְלוֹס הַגָּדוֹל "מִשֵּׁבֶט בִּנְיָמִין"׃ אֲנָשִׁים צַדִּיקִים הָיוּ, שְׁמוּרִים בְּיַד הָאֱלֹהִים, שׁוֹאֲפִים לִכְבוֹדוֹ אֶל יְרוּשָׁלַיִם שֶׁל מַעְלָה׃ הַשְּׁלִיחִים הָאֵלֶּה עָבְרוּ בְּכָל תֵּבֵל, בְּשֵׁם הָאֱלֹהִים, כְּדָגִים אִלְּמִים לָכְדוּ אֶת הָעַמִּים וְטִהֲרוּם מִקָּרְבְּנוֹת טְמֵאִים, בְּגוּף הָאָדוֹן הֶאֱכִילוּם וּבְדָמוֹ הִשְׁקוּ אֶת כָּל הָאָרֶץ׃ וּמִלִּים "לַמָּקוֹם הַמּוּכָן מֵאָז" — מְכַוְּנוֹת לִירוּשָׁלַיִם שֶׁל מַעְלָה, אֲשֶׁר שָׁם כְּבוֹד הָאָב וְהַבֵּן וְרוּחַ הַקֹּדֶשׁ זוֹרֵחַ מֵאוֹר הַשֶּׁמֶשׁ׃ "וְתָמִיד יִשְׂמְחוּ בִּי, וְיַשְׁמִידוּ אֶת מַשְׁמִידֵיהֶם, וְיוֹכִיחוּ אֶת מְחָרְפֵיהֶם" — כִּי בָּאוּ הַשְּׁלִיחִים כְּצֹאן בְּתוֹךְ זְאֵבִים אֶל תּוֹךְ עַמִּים בּוּרִים, וְקִבְּלוּ מֵהֶם גִּדּוּפִים וַעֲלָבוֹנוֹת רַבִּים, אַךְ לְבַסּוֹף לֹא יָכְלוּ הַגּוֹיִם לַעֲמֹד בִּפְנֵי רוּחַ הַקֹּדֶשׁ הַמְדַבֵּר מִפִּי הַשְּׁלִיחִים, וְהֵם הָפְכוּ אֶת הַזְּאֵבִים לִכְבָשִׂים וְעוֹלָלִים׃ "וּכְשֶׁיִּרְאוּנִי יִשְׂמְחוּ, וְעִם עַמִּי יְקַבְּלוּ אֶת הַשָּׁבִים אֵלַי" — כֵּן עִם הַשְּׁלִיחִים יִתְעַלּוּ בְּחֶסֶד כָּל הַשָּׁבִים אֵלַי מִתַּעְתּוּעֵי הַגּוֹיִם׃


#### שיבה לגוף הכתוב


*[Л. 64א]*

רְאֵה אַבְרָם אֶת אֲשֶׁר רָאִיתָ, שְׁמַע אֶת אֲשֶׁר שָׁמַעְתָּ, וְהָבֵן אֶת אֲשֶׁר יָדַעְתָּ, לֵךְ אֶל נַחֲלָתִי, וַאֲנִי עִמְּךָ עַד עוֹלָם׃

וּבְעוֹד הוּא מְדַבֵּר וְהִנְנִי עַל הָאָרֶץ, וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! אֵינֶנִּי עוֹד בַּכָּבוֹד אֲשֶׁר הָיִיתִי בּוֹ בָּהָר׃ וְכָל אֲשֶׁר חָפְצָה נַפְשִׁי בְּלִבִּי לְהָבִין, לֹא הֵבַנְתִּי׃


*[Л. 64ב: עשר המכות בפירוט]*

וַיֹּאמֶר אֵלַי אֶת מִשְׁאֲלַת לִבְּךָ אַגִּיד, כִּי חָפַצְתָּ לָדַעַת אֶת עֶשֶׂר הַמַּכּוֹת אֲשֶׁר הֵכַנְתִּי לַגּוֹיִם וְהֵכִינוֹתִי לְעֵת עֲבֹר הַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה עַל הָאָרֶץ׃ שְׁמַע אֶת אֲשֶׁר אֲגַלֶּה לְךָ: הָרִאשׁוֹנָה — עֹנִי גָדוֹל; הַשְּׁנִיָּה — רֹב שְׂרֵפוֹת אֵשׁ בֶּעָרִים; הַשְּׁלִישִׁית — דֶּבֶר מַשְׁחִית לַבְּהֵמָה; הָרְבִיעִית — רָעָב לָאָרֶץ וּלְעַמֶּיהָ; הַחֲמִישִׁית — מוֹת שָׂרִים, אֲבֵדָה בִּרְעִידוֹת אֶרֶץ וּבַחֶרֶב; הַשִּׁשִּׁית — חַיּוֹת רָעוֹת תִּהְיֶינָה קִבְרֵיהֶם; הַשְּׁבִיעִית — רָעָב וָדֶבֶר יִתְחַלְּפוּ לְאָבְדָּנָם; (הַשְּׁמִינִית — חֲסֵרָה בַּמָּקוֹר) [1]; הַתְּשִׁיעִית — [עֹנֶשׁ] בַּחֶרֶב וּבְמָנוֹס מַר; הָעֲשִׂירִית — רְעָמִים וּרְעִידוֹת אֶרֶץ מַחֲרִיבוֹת׃

אָז אֶתְקַע בָּאֲוִיר וְאֶשְׁלַח אֶת בְּחִירִי, אֲשֶׁר בּוֹ מִדָּה אַחַת מִכֹּחִי, וְהוּא יִקְרָא אֶת עַמִּי הַנִּלְחָצִים בִּידֵי הַגּוֹיִם, וַאֲנִי עִמּוֹ אֶשְׂרֹף אֶת הַמְחָרְפִים אוֹתָם וְהַשּׁוֹלְטִים בָּהֶם בָּעוֹלָם הַזֶּה׃ אֶת הַמְכַסִּים אוֹתִי בְּלַעַג אֶתֵּן לְבִזָּיוֹן בָּעוֹלָם הַבָּא, כִּי הֲכִינוֹתִים לְמַאֲכַל אֵשׁ שְׁאוֹל, לְעוּף בְּלִי הֶרֶף בַּאֲוִיר בְּתַחְתִּיּוֹת הַשְּׁאוֹל, לְמַלֵּא בִטְנָם בְּרִמָּה׃ אָז יִרְאוּ אֶת צַדִּיקֵי הַבּוֹרֵא — אֶת אֲשֶׁר בָּחֲרוּ בִרְצוֹנִי וְקִיְּמוּ מִצְוֹתַי׃ הַשְּׂמֵחִים עִמִּי יִשְׂמְחוּ עַל אֲבֵדַת הַנּוֹפְלִים אֲשֶׁר הָלְכוּ אַחֲרֵי הָאֱלִילִים וְרִצְחוֹתֵיהֶם; יֵרָקְבוּ בְּמֵעֵי הַתּוֹלַעַת הָעַרְמוּמִית שֶׁל עֲזָאזֵל, יִצָּרְבוּ בְּאֵשׁ לְשׁוֹן עֲזָאזֵל׃ כִּי קִוִּיתִי כִּי יָבֹאוּ אֵלַי, וְהֵם לֹא אָבוּ, קָרְאוּ לְזָר וְנִלְווּ אֶל מִי שֶׁלֹּא הֻכְשְׁרוּ לוֹ, וְאֶת הָאָדוֹן כָּל יָכוֹל עָזָבוּ׃


*[Л. 64ג: חתימה]*

עַל כֵּן אַבְרָם, רְאֵה וּשְׁמַע: דּוֹרְךָ הַשְּׁבִיעִי יֵלֵךְ עִמְּךָ וְיֵצֵא אֶל אֶרֶץ נָכְרִיָּה; יְשַׁעְבְּדוּם וְיַרְבּוּ לְעַנּוֹתָם בְּמֶשֶׁךְ שָׁעָה אַחַת שֶׁל עוֹלָם הָרֶשַׁע; וְלָעָם אֲשֶׁר יִהְיוּ מְשֻׁעְבָּדִים לוֹ, אֲנִי אֶהְיֶה הַשּׁוֹפֵט׃

וַיֹּאמֶר ה׳ אֶל אַבְרָם כָּזֹאת: הֲשֹׁמֵעַ אַתָּה אַבְרָם, אֶת אֲשֶׁר הִגַּדְתִּי לְךָ עַל אֲשֶׁר יִהְיֶה לְעַמְּךָ בְּאַחֲרִית הַיָּמִים? וַיִּשְׁמַע אַבְרָם אֶת דִּבְרֵי הָאֱלֹהִים וַיָּשֶׂם אֹתָם בְּלִבּוֹ׃
