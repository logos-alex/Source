## קטע ו' — אליל הקנאה, חורבן ההיכל, ארבעת הדורות, האיש הנרדף
**Л. 61ב–62ג**


*[Л. 61ב]*

וָאֵרֶא שָׁם אֶת הַגְּנֵבָה וְאֶת הַשּׁוֹאֲפִים אֵלֶיהָ וְאֶת עָנְשָׁם הֶעָתִיד בַּמִּשְׁפָּט הַגָּדוֹל׃ וָאֵרֶא אֲנָשִׁים עֲרֻמִּים אִישׁ מוּל אָחִיו, וְאֶת קְלוֹנָם וְתַאֲוָתָם אִישׁ לְאָחִיו, וְאֶת גְּמוּלָם׃ וָאֵרֶא שָׁם אֶת הַתַּאֲוָה וּבְיָדָהּ רֹאשׁ כָּל עֹשֵׂה עָוֶל, וְאֶת בִּזָּיוֹנָהּ וְאֶת פִּזּוּרָהּ הַנָּתוּן לָאֲבַדּוֹן׃ וָאֵרֶא שָׁם דְּמוּת אֱלִיל הַקִּנְאָה [1], דּוֹמֶה לָאֱלִילֵי הָעֵץ אֲשֶׁר עָשָׂה אָבִי, וְגוּפוֹ נְחֹשֶׁת מַבְרִיקָה, וּלְפָנָיו אֲנָשִׁים מִשְׁתַּחֲוִים לוֹ, וּמִזְבֵּחַ מוּלוֹ, וּנְעָרִים נִשְׁחָטִים עָלָיו לִפְנֵי הָאֱלִיל׃


*[Л. 61ג]*

וָאֶשְׁאָלֵהוּ מָה הָאֱלִיל הַזֶּה, וּמָה הַמִּזְבֵּחַ הַזֶּה, וּמִי הַמֻּקְרָב, וּמִי הֵם הַכֹּהֲנִים הָאֵלֶּה? וּמָה הַהֵיכָל הַזֶּה אֲשֶׁר אֲנִי רֹאֶה, הַנֶּהְדָּר בְּאָמָּנוּת וּבְיֹפִי כְּבוֹדְךָ הַיֹּצֵא מִן הַכִּסֵּא? וַיֹּאמֶר שְׁמַע אַבְרָם! אֲשֶׁר רָאִיתָ — הַהֵיכָל וְהַמִּזְבֵּחַ וְהַיֹּפִי — זוֹ מַחְשֶׁבֶת כְּהֻנַּת שֵׁם כְּבוֹדִי, אֲשֶׁר בָּהּ תִּשְׁכֹּן כָּל תְּפִלַּת אָדָם, וְקָרְבְּנוֹת מְלָכִים וּנְבִיאִים, וְקָרְבָּן מְסֻיָּם אֲשֶׁר גָּזַרְתִּי לְהַקְרִיב לִי בְּדוֹרוֹת הַבָּאִים שֶׁל עַמְּךָ׃ וְהַגּוּף אֲשֶׁר רָאִיתָ — זֶה חֲרוֹנִי אֲשֶׁר יַכְעִיסוּ בּוֹ צֶאֱצָאֶיךָ אוֹתִי, וְהָאִישׁ אֲשֶׁר רָאִיתָ הוֹרֵג — הוּא הַמֵּסִית לְקָרְבְּנוֹת דָּמִים אֲשֶׁר יִהְיוּ לִי לְעֵדִים בַּמִּשְׁפָּט הָאַחֲרוֹן וְרֵאשִׁית הַבְּרִיאָה׃

וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! לָמָּה בָּרָאתָ אֶת הַהֲוָיָה כָּזֹאת, וְאַחַר נָתַתָּ אֶת יְדִיעָתָהּ? וַיֹּאמֶר אֵלַי שְׁמַע אַבְרָם, וְהָבֵן אֶת אֲשֶׁר אֹמַר לְךָ, וַעֲנֵה עַל אֲשֶׁר אֶשְׁאָלֶךָּ׃ מַדּוּעַ לֹא שָׁמַע תֶּרַח אָבִיךָ לִדְבָרֶיךָ וְלֹא סָר מֵעֲבוֹדַת שֵׁדֵי הָאֱלִילִים, עַד אֲשֶׁר אָבַד עִם כָּל בֵּיתוֹ?


*[Л. 61ד]*

וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! רַק מִפְּנֵי שֶׁבָּחַר שֶׁלֹּא לִשְׁמֹעַ לִי, וְגַם אֲנִי לֹא הָלַכְתִּי בְּדַרְכֵי מַעֲשָׂיו׃ וַיֹּאמֶר אֵלַי שְׁמַע אַבְרָם: כְּמוֹ שֶׁרְצוֹן אָבִיךָ בּוֹ וּרְצוֹנְךָ בְּךָ, כֵּן רְצוֹן חֶפְצִי בִּי נָכוֹן הוּא לְכָל הַיָּמִים הַבָּאִים אֲשֶׁר לֹא יָדַעְתָּ מֵאָז, וְגַם לִימֵי זַרְעֲךָ הָעֲתִידִים אֲשֶׁר תִּרְאֶה בְעֵינֶיךָ׃ הַבֵּט בַּתְּמוּנָה׃

וָאַבֵּט וָאֵרֶא אֶת הַתְּמוּנָה רוֹעֶשֶׁת, וּמִצִּדָּהּ הַשְּׂמָאלִי יָצְאוּ עַמִּים גּוֹיִם וַיִּגְזְלוּ אֶת הָאֲנָשִׁים וְהַנָּשִׁים וְהַיְלָדִים אֲשֶׁר מִצַּד הַיָּמִין, מֵהֶם שָׁחֲטוּ וּמֵהֶם הֶחֱזִיקוּ לָהֶם׃ וָאֵרֶא כִּי רָצוּ אֶל אַרְבַּעַת פִּתְחֵי הַהֵיכָל וַיַּצִּיתוּהוּ בָּאֵשׁ, וְאֶת הַקֳּדָשִׁים אֲשֶׁר בּוֹ בָּזָזוּ׃


*[Л. 62א]*

וָאֹמַר קֶדֶם עוֹלָם! אֶת בְּנֵי עַמִּי אֲשֶׁר קִבַּלְתָּ בּוֹזְזִים הֲמוֹן גּוֹיִם, מֵהֶם הוֹרְגִים וּמֵהֶם מַחֲזִיקִים כְּגֵרִים; אֶת הַהֵיכָל הִצִּיתוּ בָאֵשׁ, וְאֶת הַכֵּלִים הַיָּפִים אֲשֶׁר בּוֹ בּוֹזְזִים וּמַחֲרִיבִים׃ קֶדֶם עוֹלָם הֶחָזָק! אִם כֵּן הוּא, לָמָּה הֶעֱצַבְתָּ עַתָּה אֶת לִבִּי? וְלָמָּה יִהְיֶה כָּזֹאת? וַיֹּאמֶר אֵלַי שְׁמַע אַבְרָם! רָאִיתָ בַתְּמוּנָה כִּי צֶאֱצָאֵי זַרְעֲךָ יַכְעִיסוּנִי בִּגְלַל הָאֱלִיל אֲשֶׁר רָאִיתָ, וּבִרְצִיחַת אָדָם מִתּוֹךְ קִנְאַת דָּת; אֲשֶׁר רָאִיתָ כֵּן יִהְיֶה׃

וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! יַעַבְרוּ נָא מֵעַתָּה מַעֲשֵׂי הָרֶשַׁע וְכֹפֶר הָאֱלֹהִים, אַךְ הַרְאֵנִי אֶת עֹשֵׂי הַמִּצְווֹת הַשָּׁוִים בְּצִדְקָם לְמַעֲשָׂיו; הֲתוּכַל לַעֲשׂוֹת זֹאת? וַיֹּאמֶר אֵלַי עֵת הַצֶּדֶק תִּפְגְּשֵׁם, רִאשׁוֹנָה בִּזְכוּת קְדֻשַּׁת מְלָכִים וְצֶדֶק שֹׁפְטִים אֲשֶׁר בָּרָאתִי מֵרֹאשׁ לִמְשֹׁל בָּהֶם וְעַל יָדָם; וּמֵהֶם יֵצְאוּ אֲנָשִׁים אֲשֶׁר יִשְׁקְדוּ עַל אֲשֶׁר סִפַּרְתִּי לְךָ וַאֲשֶׁר רָאִיתָ׃


*[Л. 62ב]*

וָאַעַן וָאֹמַר חָזָק וְצַדִּיק, מְקֻדָּשׁ בְּכֹחוֹ! חֲמֹל נָא עַל בַּקָּשָׁתִי: הוֹאִיל וַהֲעֵלִיתַנִי אֶל רוּמְךָ, הוֹדַע נָא לִידִידְךָ אֶת אֲשֶׁר אֲנִי שׁוֹאֵל: עַד מָתַי יִמָּשֵׁךְ אֲשֶׁר רָאִיתִי? וַיַּרְאֵנִי הֲמוֹן עַמּוֹ וַיֹּאמֶר אֵלַי יִהְיוּ אַרְבָּעָה דוֹרוֹת [2] אֲשֶׁר יַכְעִיסוּנִי, וּבִגְלַל זֹאת יִהְיֶה כְּמַעֲשֵׂיהֶם גְּמוּלָם מֵאִתִּי׃ וְהַדּוֹר הָרְבִיעִי, מֵאָה שָׁנָה וְשָׁעָה אַחַת שֶׁל עוֹלָם, יִהְיֶה בְמָרוֹרִים וּבְחֶרְפָּה בְּתוֹךְ הַגּוֹיִם׃ וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! כַּמָּה תֶּאֱרַךְ שְׁעַת הָעוֹלָם? וַיֹּאמֶר שְׁתֵּים עֶשְׂרֵה שָׁנָה שֶׁל הָעוֹלָם הַזֶּה הֵכַנְתִּי לְעוֹלָם הָרֶשַׁע לִמְשֹׁל בַּגּוֹיִם וּבְעַמְּךָ; וְעַד קֵץ הַיָּמִים יִהְיֶה כַּאֲשֶׁר רָאִיתָ׃ סְפֹר וְהָבֵן וְהַבֵּט בַּתְּמוּנָה׃


*[Л. 62ב–ג: האיש הנרדף]*

וָאַבֵּט וָאֵרֶא אִישׁ יוֹצֵא מִצַּד הַגּוֹיִם הַשְּׂמָאלִי; וְאַחֲרָיו יָצְאוּ מִצַּד הַגּוֹיִם אֲנָשִׁים וְנָשִׁים וִילָדִים, הֲמוֹנִים רַבִּים, וַיִּשְׁתַּחֲווּ לוֹ׃ וּבְעוֹדֶנִּי מַבִּיט וְהִנֵּה מִצַּד הַיָּמִין יָצְאוּ אֲנָשִׁים, מֵהֶם מְבַזִּים אֶת הָאִישׁ הַהוּא וּמֵהֶם מַכִּים אוֹתוֹ, וּמֵהֶם מִשְׁתַּחֲוִים לוֹ; וַעֲזָאזֵל רָץ וַיִּשְׁתַּחוּ וַיִּשַּׁק לְפָנָיו, וַיִּפֶן וַיַּעֲמֹד אַחֲרָיו׃

וָאֹמַר קֶדֶם עוֹלָם הֶחָזָק! מִי הָאִישׁ הַזֶּה, הַנִּבְזֶה וְהַמֻּכֶּה, אֲשֶׁר הַגּוֹיִם מִשְׁתַּחֲוִים לוֹ עִם עֲזָאזֵל? וַיֹּאמֶר שְׁמַע אַבְרָם! הָאִישׁ אֲשֶׁר אַתָּה רֹאֶה נִבְזֶה וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוִים לוֹ — זוֹ נֶחָמַת הַגּוֹיִם [3] אֲשֶׁר תֵּצֵא מִמְּךָ בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל עוֹלָם הַכְּפִירָה׃ וּבַשָּׁנָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל עוֹלָמִי הָאַחֲרוֹן אָקִים אֶת הָאִישׁ הַזֶּה מִזַּרְעֲךָ, אֲשֶׁר רָאִיתָ יוֹצֵא מִתּוֹךְ עַמִּי; וְאַחֲרָיו יֵלְכוּ כָּל הַהוֹלְכִים בְּלֵב וָלֵב׃ אֵלֶּה אֲשֶׁר רָאִיתָ יוֹצְאִים מִצַּד שְׂמֹאל הַתְּמוּנָה וּמִשְׁתַּחֲוִים לוֹ — הֵם רַבִּים מִן הַגּוֹיִם הַבּוֹטְחִים בּוֹ; וּבְנֵי עַמְּךָ אֲשֶׁר רָאִיתָ מִצַּד יָמִין, מֵהֶם מְחָרְפִים וּמַכִּים וּמֵהֶם מִשְׁתַּחֲוִים לוֹ — רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ, וְהוּא יִבְחַן אֶת מוֹקִירָיו מִזַּרְעֲךָ עַד קֵץ הַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה לְהַשְׁבִּית אֶת עוֹלָם הָרֶשַׁע׃

## הערות

**[1]** אֱלִיל הַקִּנְאָה — כך במפורש קמצ'טנוב-מילקוב (הע' 219), בזיקה מפורשת ליח' ח, ג-ה ('סמל הקנאה המקנה'). המונח העברי המקורי היה קרוב ודאי ל'סֵמֶל הַקִּנְאָה' שביחזקאל, אך נשמר כאן 'אֱלִיל' בהתאמה להקשר האנטי-אלילי של הפרק.

**[2]** אַרְבָּעָה דוֹרוֹת — כך במפורש קמצ'טנוב-מילקוב (הע' 223), בזיקה לבר' טו, יג ('וְעִנּוּ אֹתָם אַרְבַּע מֵאוֹת שָׁנָה').

**[3]** נֶחָמַת הַגּוֹיִם — קטע קשה ושנוי במחלוקת; המהדירים עצמם (הע' 230) רואים כאן ברמז את בשורת ביאת המשיח, ולדעתם 'זו ללא ספק תוספת מאוחרת' על הסיפור היהודי הקדום. עם זאת, קטע זה אינו מסומן במפורש כתוספת מעבד הפליאה (בניגוד לקטעים בקטע ב׳), ולכן הובא כאן כחלק מגוף הכתוב הרציף, בצירוף הסתייגות זו.
