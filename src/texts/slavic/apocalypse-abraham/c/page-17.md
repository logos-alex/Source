וַיֹּאמֶר: שְׁתֵּים־עֶשְׂרֵה שָׁנָה שַׂמְתִּי לָעוֹלָם הַזֶּה [1] שֶׁלֹּא יִשְׂבַּע, לְהַחֲזִיק בָּעַמִּים וּבְזַרְעֲךָ, וְעַד קֵץ הַזְּמַן יִהְיֶה כְּכֹל אֲשֶׁר רָאִיתָ; סְפֹר וְדַע וְהַבֵּט בַּתְּמוּנָה. 

וָאַבִּיט וָאֵרֶא אִישׁ יוֹצֵא מִצַּד שְׂמֹאל הַגּוֹיִם, [2] וַיֵּצְאוּ אֲנָשִׁים וְנָשִׁים וִילָדִים מִצַּד הַגּוֹיִם הָרַבִּים, וַיִּשְׁתַּחֲווּ לוֹ. 

וְעוֹד אֲנִי מַבִּיט, וַיֵּצְאוּ מִצַּד יָמִין, וּמִקְצָתָם בּוֹשִׁים מֵאוֹתוֹ אִישׁ, וּמִקְצָתָם מַכִּים אוֹתוֹ, וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ. 

וָאֵרֶא כִּי הִשְׁתַּחֲווּ לוֹ, וַיָּרָץ עֲזָאזֵל וַיִּשְׁתַּחוּ, וַיִּשַּׁק אֶת פָּנָיו [3] וַיָּשָׁב וַיַּעֲמֹד אַחֲרָיו. 

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מִי הָאִישׁ הַנִּכְלָם וְהַמֻּכֶּה מִן הַגּוֹיִם, וְעִם עֲזָאזֵל מִשְׁתַּחֲוֶה לוֹ. 

וַיַּעַן וַיֹּאמֶר: שְׁמַע אַבְרָהָם, הָאִישׁ אֲשֶׁר רָאִיתָ נִכְלָם וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוֶה לוֹ, זֶה מְנוּחַת הַגּוֹיִם, מִזַּרְעֲךָ יִהְיֶה לְעַם בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים־עֶשְׂרֵה שֶׁל קֵץ עוֹלָם הַכְּלִמָּה; וּבַשָּׁנָה הַשְּׁתֵּים-עֶשְׂרֵה שֶׁל קֵץ עוֹלָמִי אָקִים אֶת הָאִישׁ הַזֶּה מִזֶּרַע עַמִּי, אֲשֶׁר רָאִיתָ — לוֹ יִדַּמּוּ כֻלָּם, וּבְמָשָׁל כְּמִשְׁתַּנֶּה בְּעֵצוֹתָיו, כְּקוֹרְאִים מִמֶּנִּי.

וַאֲשֶׁר רָאִיתָ מִצַּד שְׂמֹאל בַּתְּמוּנָה יוֹצְאִים וּמִשְׁתַּחֲוִים לוֹ, מֵהַגּוֹיִם רַבִּים יִבְטְחוּ בּוֹ. 

וַאֲשֶׁר רָאִיתָ מִזַּרְעֲךָ מִצַּד יָמִין, מִקְצָתָם בּוֹשִׁים וּמַכִּים, מִקְצָתָם מִשְׁתַּחֲוִים לוֹ — רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ. 

וְהוּא יְנַסֶּה אֶת הַמִּשְׁתַּחֲוִים לוֹ מִזַּרְעֲךָ, בְּקֵץ הַשָּׁעָה הַשְּׁתֵּים־עֶשְׂרֵה, בְּעָרֹב עוֹלָם הָרֶשַׁע. 

בְּטֶרֶם יָחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, יָבוֹא מִשְׁפָּטִי עַל הָעַמִּים הַעַרְמוּמִיִּים מִזֶּרַע עַמְּךָ הַמֻּבְדָּל לִי. 

בַּיָּמִים הָהֵם אָבִיא עַל כָּל הַתֵּבֵל עֶשֶׂר מַכּוֹת בְּרָעָה וּמַחֲלָה [4] וַאֲנָקַת רוּחָם, וְאֶת מָרַת נַפְשָׁם אָבִיא עַל הַדּוֹרוֹת אֲשֶׁר עַל הָאָרֶץ, בַּעֲבוּר קֶצֶף וְהַשְׁחָתַת בְּרוּאֵיהֶם אֲשֶׁר יַכְעִיסוּ אוֹתִי.

וְאָז יִוָּתְרוּ מִזַּרְעֲךָ אֲנָשִׁים צַדִּיקִים בְּמִסְפָּר שְׁמוּרִים לִי, [5] מְמַהֲרִים לִכְבוֹד שְׁמִי אֶל הַמָּקוֹם הַמּוּכָן לָהֶם מֵאָז, אֲשֶׁר רָאִיתָ שָׁמֵם בַּתְּמוּנָה.
