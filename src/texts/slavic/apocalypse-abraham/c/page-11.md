קַדְמוֹן וְגִבּוֹר, קָדוֹשׁ אֵל, יָחִיד מוֹשֵׁל, עַצְמִי, בְּלִי כִלָּיוֹן, בְּלִי טֻמְאָה, בְּלִי מוֹלֶדֶת, בְּלִי דֹפִי, בְּלִי מָוֶת, שָׁלֵם מֵעַצְמְךָ, מֵאִיר מֵעַצְמְךָ, בְּלִי אֵם, בְּלִי אָב, בְּלִי תוֹלְדוֹת, רָם, אִישׁ אֵשׁ, יָשָׁר, אֹהֵב אָדָם, אוֹהֵב, נַדְבָן, נוֹתֵן מַתָּנוֹת, קַנַּאי, אֶרֶךְ אַפַּיִם, רַב חֶסֶד — הוּא אֱלֹהַי, גִּבּוֹר וְקָדוֹשׁ, צְבָאוֹת הַמְּפֹאָר — יָאוֹאֵל. אַתָּה אֲשֶׁר אָהֲבָה נַפְשִׁי, שׁוֹמְרִי לְעוֹלָם, אֵשׁ יוֹשֶׁבֶת, מַזְהִיר, מֵאִיר, בַּעַל קוֹל רַעַם, מַרְאֶה בָרָק, רַב עֵינַיִם, מְקַבֵּל תְּפִלַּת מְכַבְּדֶיךָ, וּמַסֵּב פָּנָיו מִתְּפִלַּת הָאוֹחֲזִים בְּךָ בְּעֹשֶׁק הַמַּכְעִיסִים אוֹתְךָ; מַתִּיר אֶת הָרְשָׁעִים וְהַצַּדִּיקִים הַמְעֹרְבָּבִים שֶׁל הַתֵּבֵל בָּעוֹלָם הַנִּפְסָד, וּמְחַדֵּשׁ אֶת עוֹלָם הַצַּדִּיקִים. אַתָּה מֵאִיר אוֹר לִפְנֵי הָאוֹר הַפְּנִימִי עַל בְּרוּאֶיךָ, וּמִפָּנֶיךָ יוֹם בָּאָרֶץ; וּבִשְׁמֵי מְעוֹנוֹתֶיךָ בְּלִי חֶסְרוֹן אוֹר אַחֵר, זֹהַר אֲשֶׁר לֹא יְסֻפַּר, מֵאוֹרוֹת פָּנֶיךָ. קַבֵּל תְּפִלָּתִי וְהֵעָרֵב בָּהּ, כָּכָה גַּם אֶת הַקָּרְבָּן אֲשֶׁר עָשָׂה לְעַצְמוֹ עַל יָדִי, הַמְבַקֵּשׁ אוֹתְךָ הֵיטֵב; קַבְּלֵנִי, וְלַמְּדֵנִי וְהַרְאֵנִי, הוֹדַע לְעַבְדְּךָ כְּכֹל אֲשֶׁר הִבְטַחְתַּנִי.

וּבְעוֹדֶנִּי אוֹמֵר אֶת הַשִּׁיר, וּפִי הָאֵשׁ אֲשֶׁר עַל הַמִּפְרָשׂ עָלָה לְמַעְלָה, וָאֶשְׁמַע קוֹל כִּמְבוּכַת הַיָּם, וְלֹא חָדַל מִבְּעֵרַת הָאֵשׁ הָרַבָּה. וְכַאֲשֶׁר עָלְתָה הָאֵשׁ לְמַעְלָה מִתְנוֹשְׂאָה, וָאֵרֶא תַחַת הָאֵשׁ כִּסֵּא, וּמִן הָאֵשׁ רַבֵּי הָעֵינַיִם סָבִיב, אֹמְרִים שִׁיר יָפֶה וְנָעִים, וְתַחַת הַכִּסֵּא אַרְבַּע חַיּוֹת אֵשׁ — וּמַרְאֵיהֶן וְקוֹלָן אֶחָד, כָּל אַחַת אַרְבַּעַת פָּנִים, וְעַל גּוּפָן אַרְבָּעָה רָאשִׁים. מַרְאֵה פְנֵיהֶן: אַרְיֵה, שׁוֹר, אָדָם, נֶשֶׁר, כֵּן פְּנֵי הָאַרְבַּע חַיּוֹת, וּלְכָל אַחַת אַרְבַּע כְּנָפַיִם מִכְּתֵפֵיהֶן וּמֵחֶצְיָן וּמִמָּתְנֵיהֶן. וּבִכְנָפַיִם אֲשֶׁר מִכְּתֵפֵיהֶן יְכַסּוּ פְּנֵיהֶן, וּבִשְׁתֵּי כְנָפַיִם מִמָּתְנֵיהֶן יְכַסּוּ רַגְלֵיהֶן, וּבִכְנָפַיִם הָאֶמְצָעִיּוֹת יִפְרְשׂוּ וְעָפוּ יָשָׁר. וַיְהִי כְּכַלּוֹתָם אֶת הַשִּׁיר וַיַּבִּיטוּ אִישׁ אֶל רֵעֵהוּ וַיְאַיְּמוּ אִישׁ בְּרֵעֵהוּ. וַיְהִי כִּרְאוֹת הַמַּלְאָךְ אֲשֶׁר עִמִּי כִּי יְאַיְּמוּ, וַיַּעַזְבֵנִי בַּמָּקוֹם וַיָּרָץ אֲלֵיהֶם מַהֵר, וַיַּסֵּב אֶת פְּנֵי כָּל חַיָּה מִנֶּגֶד פְּנֵי רֵעוּתָהּ, לְבִלְתִּי רְאוֹת אִישׁ אֶת פְּנֵי רֵעֵהוּ מְאַיְּמוֹת, וַיְלַמְּדֵם שִׁיר הַשָּׁלוֹם אֲשֶׁר בּוֹ עִם הַקַּדְמוֹן. וַאֲנִי עוֹד עוֹמֵד וּמַבִּיט, וָאֵרֶא בִּיסוֹד הַחַיּוֹת מֶרְכָּבָה, גַּלְגַּלֵּי אֵשׁ, וְכָל גַּלְגַּל מָלֵא עֵינַיִם סָבִיב, וּמֵעַל הַגַּלְגַּלִּים הַכִּסֵּא אֲשֶׁר רָאִיתִי, וְהוּא נִסְתָּר בָּאֵשׁ, וְאֵשׁ סוֹבֵב אוֹתוֹ סָבִיב, וְאוֹר אֲשֶׁר לֹא יְסֻפָּר סוֹבֵב אֶת מַחְנֵה הָאֵשׁ.

וָאֶשְׁמַע אֶת קוֹל קְדֻשָּׁתָם כְּקוֹל אִישׁ אֶחָד, וַיָּבֹא אֵלַי קוֹל מִתּוֹךְ הָאֵשׁ, אֹמֵר: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: הַבֵּט אֶל הַמִּפְרָשִׂים אֲשֶׁר תַּחְתֶּיךָ, אֲשֶׁר עֲלֵיהֶם נִצְעָדְתָּ, וּרְאֵה כִּי בְּשׁוּם מִפְרָשׂ אֵין אַחֵר זוּלָתִי אֲשֶׁר בִּקַּשְׁתָּ אוֹ אֲשֶׁר אֲהֵבְךָ. וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִבְקְעוּ הַמִּפְרָשִׂים, וְמִתַּחְתַּי הַשָּׁמַיִם. וָאֵרֶא עַל הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו, אֵשׁ פְּרוּסָה, וְאוֹר רַב, וָטַל, וַהֲמוֹן מַלְאָכִים, וּגְבוּרַת כָּבוֹד בִּלְתִּי-נִרְאֶה מִלְּמַעְלָה — מִלְּבַד הַחַיּוֹת אֲשֶׁר רָאִיתִי, לֹא רָאִיתִי שָׁם אַחֵר. וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הַמִּפְרָשׂ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים אֶת מִצְוֹת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כַּאֲשֶׁר עָמַדְתִּי עַל הָעוֹמְדִים עָלָיו. וְלֹא הָיָה עַל הַמִּפְרָשׂ הַהוּא בְּשׁוּם צוּרָה אַחֶרֶת כֹּחַ, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הַמִּפְרָשׂ הַשְּׁבִיעִי, הַכֹּחַ אֲשֶׁר רָאִיתִי, פָּקַד עַל הַמִּפְרָשׂ הַשִּׁשִּׁי. וְנִשָּׂא, וָאֵרֶא בַּמִּפְרָשׂ הַחֲמִישִׁי כֹּחוֹת הַכּוֹכָבִים, וַיְצַוּוּ לְקַיֵּם אֶת הַמִּצְוֹת, וַיִּשְׁמְעוּ לָהֶם יְסוֹדוֹת הָאָרֶץ. וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם אַבְרָהָם. וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: הַבֵּט מִלְּמַעְלָה, וְהִנֵּה תַּחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם. וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם — עָפָר וָאֵפֶר. וַיֹּאמֶר לִי: כְּמִסְפַּר הַכּוֹכָבִים וְכֹחָם אֶתֵּן לְזַרְעֲךָ, גּוֹיִם וְעַמִּים מֻבְדָּלִים לִי בְגוֹרָלִי, עִם עֲזָאזֵל.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יְדַבֵּר נָא עַבְדְּךָ לְפָנֶיךָ, וְאַל יִחַר אַפְּךָ עַל בְּחִירְךָ: הִנֵּה בְּטֶרֶם הֲרִימוֹתַנִי, חֵרְפַנִי עֲזָאזֵל; וְעַתָּה, כְּשֶׁאֵינֶנּוּ לְפָנֶיךָ, תְּכוֹנֵן עִמּוֹ. וַיֹּאמֶר אֵלַי: הַבֵּט עַתָּה תַּחַת רַגְלֶיךָ אֶל הַמִּפְרָשׂ, וְהָבֵן אֶת הַבְּרִיאָה הַמְצֻיֶּרֶת מֵאָז עַל הַמִּפְרָשׂ הַזֶּה, וְאֶת הַבְּרוּאִים אֲשֶׁר בּוֹ, וְהָעוֹלָמוֹת הַמּוּכָנִים לְאַחֲרֶיהָ. וָאַבִּיט תַּחַת רְקִיעַ רַגְלַי, וָאֵרֶא כְּדִמְיוֹן הַשָּׁמַיִם וַאֲשֶׁר בּוֹ; וְשָׁם הָאָרֶץ וּפְרִיָּהּ, וּתְנוּעָתָהּ, וְנַפְשָׁהּ, וְכֹחַ בְּנֵי אָדָם שֶׁבָּהּ, וְרֶשַׁע נַפְשׁוֹת הָרְשָׁעִים שֶׁבָּהּ, וְאֶת הֶצְדֵּקָתָם וְאֶת מַעֲשֵׂיהֶם וְאֶת תְּהוֹם וְעִנּוּיָהּ וְתַחְתִּיּוֹתֶיהָ, וְאֶת הָאֲבַדּוֹן אֲשֶׁר בָּהֶן. וָאֵרֶא שָׁם אֶת הַיָּם וְאִיָּיו וּבְהֶמְתּוֹ וְדָגָיו, וְאֶת הַלִּוְיָתָן וּמֶמְשַׁלְתּוֹ וּמִשְׁכָּבוֹ וּמְאוּרוֹתָיו, וְאֶת הַתֵּבֵל הַשּׁוֹכֶבֶת עָלָיו, וּתְנוּעוֹתָיו, וְהֶהָרֵס בַּעֲבוּרוֹ. וָאֵרֶא שָׁם נְהָרוֹת וּמְקוֹרוֹתָם וְסִבּוּבֵיהֶם. וָאֵרֶא שָׁם אֶת גַּן עֵדֶן וְאֶת פִּרְיוֹ, וְאֶת הַמַּעְיָן וְהַנָּהָר הַיּוֹצֵא מִמֶּנּוּ, וְרֹב עֵצָיו וּפְרִיחָתָם וּפֵרוֹתָם, וְאֶת עֹשֵׂי הַצֶּדֶק וּמַאֲכָלָם וּמְנוּחָתָם. וָאֵרֶא שָׁם עַם רַב, אֲנָשִׁים וְנָשִׁים וִילָדִים, מַחֲצִיתָם מִצַּד יָמִין וּמַחֲצִיתָם מִצַּד הַתְּמוּנָה הַשְּׂמָאלִי.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מָה הַתְּמוּנָה הַזֹּאת שֶׁל הַבְּרִיאָה. וַיֹּאמֶר אֵלַי: זֹאת רְצוֹנִי בַּנִּמְצָא בְּמַחְשָׁבָה, וְהָיָה טוֹב לְפָנַי; וְאַחַר כֵּן צִוִּיתִי בִּדְבָרִי וְהָיוּ, כְּכֹל אֲשֶׁר קָבַעְתִּי לִהְיוֹת — אֲשֶׁר לְכֻלָּם נִרְשַׁם מֵרֹאשׁ, וְעָמְדוּ לְפָנַי הַבְּרוּאִים מֵאָז, כְּכֹל אֲשֶׁר רָאִיתָ. וָאֹמַר: אָדוֹן גִּבּוֹר וְקַדְמוֹן, מִי הָעָם בַּתְּמוּנָה הַזֹּאת מִזֶּה וּמִזֶּה. וַיֹּאמֶר אֵלַי: אֵלֶּה אֲשֶׁר מִצַּד שְׂמֹאל, הֲמוֹן עַמִּים אֲשֶׁר הָיוּ מֵאָז, וּמוּכָנִים אַחֲרֶיךָ מֵהֶם לְמִשְׁפָּט וְתִקּוּן, וּמֵהֶם לְנָקָם וּלְאָבְדָן בְּקֵץ הָעוֹלָם; וַאֲשֶׁר מִצַּד יָמִין בַּתְּמוּנָה, אֵלֶּה הָעָם הַמֻּבְדָּל לִי, עַם מִתּוֹךְ הָעַמִּים עִם עֲזָאזֵל — אֵלֶּה אֲשֶׁר הֵכַנְתִּי לְהִוָּלֵד מִמְּךָ וּלְהִקָּרֵא עַמִּי. הַבֵּט עוֹד בַּתְּמוּנָה —
