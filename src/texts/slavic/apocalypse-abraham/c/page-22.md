וַיֹּאמֶר אֵלַי: אַדְרַבָּה עֵת הַצֶּדֶק תִּפְגַּע, לִפְנֵי דֻגְמַת מְלָכִים וְשׁוֹפְטִים בְּצֶדֶק, אֲשֶׁר בָּרָאתִי מֵאָז לִמְשֹׁל מִתּוֹכָם בָּהֶם; וּמֵהֶם יֵצְאוּ אֲנָשִׁים אֲשֶׁר יְמַהֲרוּ אֶת אֲשֶׁר הִגַּדְתִּי לְךָ וְרָאִיתָ. וָאַעַן וָאֹמַר: גִּבּוֹר, הַמְקֻדָּשׁ בְּכֹחוֹ, רַחוּם בְּבַקָּשָׁתִי, כִּי עַל כֵּן הוֹדַעְתָּ לִי וְהִגְלֵיתָ אֵלַי, כַּאֲשֶׁר הֶעֱלֵיתָ אוֹתִי אֶל רוּמְךָ; לָכֵן הוֹדַעְתָּ לְיְדִידְךָ זֶה כְּכֹל אֲשֶׁר שָׁאַלְתִּי — הַהְאָרַכְתָּ אוֹ הָיָה כְּכֹל אֲשֶׁר רָאִיתִי. וַיַּרְאֵנִי אֶת הֲמוֹן עַמּוֹ, וַיֹּאמֶר אֵלַי: לָכֵן רָאִיתָ אֶת אַרְבַּעַת הַמּוֹרָדוֹת [יב] — יִהְיֶה לִי מֵהֶם קֶצֶף, וּבָהֶם גְּמוּל מַעֲשֵׂיהֶם מֵאִתִּי. וּבָרְבִיעִי — מוֹרָד שֶׁל שְׁנֵים עָשָׂר שָׁנָה, וּשְׁעַת עוֹלָם אַחַת, וְהִיא גַּם כֵּן שָׁנָה, תִּהְיֶה בְרָעָה בָּעַמִּים, וְשָׁעָה בְּחֶסֶד וּבְחֶרְפָּה בַּגּוֹיִם.
