וַיְהִי כִּרְאוֹתִי אֶת הָעוֹף מְדַבֵּר, וָאֹמַר אֶל הַמַּלְאָךְ: מַה זֹּאת אֲדֹנִי. וַיֹּאמֶר: זֶה עָוֹן — זֶה עֲזָאזֵל. וַיֹּאמֶר אֵלָיו: חֶרְפָּה לְךָ עֲזָאזֵל, כִּי חֵלֶק אַבְרָהָם בַּשָּׁמַיִם וְחֶלְקְךָ בָאָרֶץ, כִּי בָּזֶה בָּחַרְתָּ וְאָהַבְתָּ לְשֶׁבֶת בִּטֻמְאָתְךָ. לָכֵן נְתָנְךָ הַקַּדְמוֹן, הַגִּבּוֹר הַשַּׁלִּיט, יוֹשֵׁב הָאָרֶץ, וּבְךָ כָּל רוּחַ שֶׁקֶר וְרֶשַׁע, וּבְךָ חֵמָה וְנִגְעֵי דּוֹרוֹת בְּנֵי הַכְּפִירָה. כִּי לֹא הִנִּיחַ הַקַּדְמוֹן הַגִּבּוֹר אֶת גּוּפוֹת הַצַּדִּיקִים בְּיָדְךָ, שֶׁתְּהִי בָּם חַיֵּי הַצֶּדֶק וּכְלִיּוֹן הָרֶשַׁע. שְׁמַע יוֹעֵץ, בּוֹשׁ מִמֶּנִּי, כִּי לֹא עַל כָּל הַצַּדִּיקִים נִתַּן לְךָ לְנַסּוֹתָם. סוּר מֵעַל הָאִישׁ הַזֶּה, לֹא תוּכַל לְהַתְעוֹתוֹ, כִּי אוֹיִבְךָ הוּא, וְשֶׁל הַהֹלְכִים אַחֲרֶיךָ וְהָאֹהֲבִים אֶת אֲשֶׁר תַּחְפֹּץ. דַּע מִזֶּה כִּי הַקַּדְמוֹן אֲשֶׁר אֲהֵבוֹ בָּחַר בְּךָ; חֲזַק וַעֲשֵׂה בַשִּׁלְטוֹן הַזֶּה כְּכֹל אֲשֶׁר אֲצַוְּךָ עַל הַמְחָרֵף אֶת הַצֶּדֶק, שֶׁלֹּא אוּכַל לְחָרֵף אֶת הַנִּפְזָר בְּסוֹד שָׁמַיִם. אֱמֹר לוֹ: תְּהִי גַחֶלֶת כִּבְשַׁן הָאָרֶץ. לֵךְ עֲזָאזֵל אֶל אֶרֶץ אֲשֶׁר לֹא יַעַבְרוּ בָהּ, כִּי שָׁם נַחֲלָתְךָ, עִם הַכּוֹכָבִים וְעִם הֶעָנָן הַנּוֹלָדִים עִם בְּנֵי אָדָם, אֲשֶׁר חֶלְקָם אַתָּה, וּבִהְיוֹתָם עֵדִים כְּנֶגְדְּךָ, וְאֵיבָתְךָ צֶדֶק. לָכֵן בַּאֲבַדּוֹנְךָ תְּהִי כָלֶה מֵאִתִּי, אָבוּד. וָאֲדַבֵּר אֶת הַדְּבָרִים אֲשֶׁר לִמְּדַנִי הַמַּלְאָךְ.
