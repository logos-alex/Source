רְאֵה אַבְרָהָם אֶת אֲשֶׁר רָאִיתָ, וּשְׁמַע אֵת אֲשֶׁר שָׁמַעְתָּ, וְדַע אֶת אֲשֶׁר יָדַעְתָּ — לֵךְ אֶל גּוֹרָלְךָ, וְהִנְנִי עִמְּךָ עַד עוֹלָם. וְעוֹד הוּא מְדַבֵּר, וָאֶמָּצֵא עַל הָאָרֶץ, וָאֹמַר: קַדְמוֹן, אֵינֶנִּי עוֹד בַּכָּבוֹד אֲשֶׁר הָיִיתִי בּוֹ לְמַעְלָה, וְאֵת אֲשֶׁר בְּלִבִּי חָפְצָה נַפְשִׁי לְהָבִין לֹא הֵבַנְתִּי. וַיֹּאמֶר אֵלַי הָאֵל, קוֹל מִן הֶעָנָן: אַבְרָהָם, אֲנִי אֲדֹנֵי הַשָּׁמַיִם וְהָאָרֶץ. רְאֵה, אַבְרָהָם, כָּל אֲשֶׁר רָאִיתָ, וְשָׁמַע, וְדַע, וְעַתָּה הַגֶּד לִי, אֲדֹנָי, מָה יִקְרֶה אֶת זַרְעִי בְּאַחֲרִית הַיָּמִים. וַיְהִי קוֹל אֶל אַבְרָהָם, אֹמֵר: יָקוּם אִישׁ מִזַּרְעֲךָ, וְרַבִּים יִכָּשְׁלוּ בּוֹ. מִקְצָתָם יַכְלִים אוֹתוֹ, וּמִקְצָתָם יְרוֹמֵם אֶת הַמִּשְׁתַּחֲוִים לוֹ; וּמֵהַגּוֹיִם רַבִּים יִבְטְחוּ בּוֹ, כִּי אֲנִי שְׁלַחְתִּיו. וְיַחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, וְאַחַר כֵּן יָבוֹא מִשְׁפָּטִי עַל עַרְמוּמֵי עַמֵּי זַרְעֲךָ. וְאָבִיא בַּיָּמִים הָהֵם עַל כָּל הַתֵּבֵל עֶשֶׂר מַכּוֹת בְּרָעָה וּמַחֲלָה וַאֲנָחָה וּמְרִירוּת נַפְשָׁם, בַּעֲבוּר אֲשֶׁר הִכְעִיסוּנִי. וְאָז יִשָּׁאֲרוּ מִזַּרְעֲךָ אֲנָשִׁים צַדִּיקִים וּשְׁמוּרִים לִי, נָחוּצִים אֶל כְּבוֹדִי, אֶל הַמָּקוֹם הַמּוּכָן לָהֶם, וְיִשְׂמְחוּ בִּי תָמִיד, וְאֶת מַשְׁחִיתֵיהֶם יַשְׁחִיתוּ וְאֶת מְבַזֵּיהֶם יְבַזּוּ, בִּרְאוֹתָם אוֹתִי שָׂמֵחַ עִם עַמִּי, מְקַבֵּל אֶת הַשָּׁבִים אֵלַי בִּתְשׁוּבָה. שָׁמַע אַבְרָהָם אֶת דִּבְרֵי הָאֱלֹהִים, וַיִּקְבְּלֵם בְּלִבּוֹ.

---

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מָה הַתְּמוּנָה הַזֹּאת שֶׁל הַבְּרִיאָה. וַיֹּאמֶר אֵלַי: זֹאת רְצוֹנִי בַּנִּמְצָא בְּמַחְשָׁבָה, וְהָיָה טוֹב לְפָנַי; וְאַחַר כֵּן צִוִּיתִי בִּדְבָרִי וְהָיוּ, כְּכֹל אֲשֶׁר קָבַעְתִּי לִהְיוֹת — אֲשֶׁר לְכֻלָּם נִרְשַׁם מֵרֹאשׁ, וְעָמְדוּ לְפָנַי הַבְּרוּאִים מֵאָז, כְּכֹל אֲשֶׁר רָאִיתָ. וָאֹמַר: אָדוֹן גִּבּוֹר וְקַדְמוֹן, מִי הָעָם בַּתְּמוּנָה הַזֹּאת מִזֶּה וּמִזֶּה. וַיֹּאמֶר אֵלַי: אֵלֶּה אֲשֶׁר מִצַּד שְׂמֹאל, הֲמוֹן עַמִּים אֲשֶׁר הָיוּ מֵאָז, וּמוּכָנִים אַחֲרֶיךָ מֵהֶם לְמִשְׁפָּט וְתִקּוּן, וּמֵהֶם לְנָקָם וּלְאָבְדָן בְּקֵץ הָעוֹלָם; וַאֲשֶׁר מִצַּד יָמִין בַּתְּמוּנָה, אֵלֶּה הָעָם הַמֻּבְדָּל לִי, עַם מִתּוֹךְ הָעַמִּים עִם עֲזָאזֵל — אֵלֶּה אֲשֶׁר הֵכַנְתִּי לְהִוָּלֵד מִמְּךָ וּלְהִקָּרֵא עַמִּי. הַבֵּט עוֹד בַּתְּמוּנָה — מִי הוּא הַמַּתְעֶה אֶת חַוָּה, וְאֵי זֶה פְּרִי הָעֵץ, וְתֵדַע מָה יִהְיֶה וְכַמָּה יִהְיֶה לְזַרְעֲךָ נַחֲלַת הַיָּמִים, וְאֶת אֲשֶׁר לֹא תוּכַל לְהָבִין אֲנִי אַגִּיד לְךָ, כִּי מָצָאתָ חֵן לְפָנַי; לְךָ אֲגַלֶּה אֶת אֲשֶׁר צָפַנְתִּי בְלִבִּי.

וַיֹּאמֶר אֵלַי: אַדְרַבָּה עֵת הַצֶּדֶק תִּפְגַּע, לִפְנֵי דֻגְמַת מְלָכִים וְשׁוֹפְטִים בְּצֶדֶק, אֲשֶׁר בָּרָאתִי מֵאָז לִמְשֹׁל מִתּוֹכָם בָּהֶם; וּמֵהֶם יֵצְאוּ אֲנָשִׁים אֲשֶׁר יְמַהֲרוּ אֶת אֲשֶׁר הִגַּדְתִּי לְךָ וְרָאִיתָ. וָאַעַן וָאֹמַר: גִּבּוֹר, הַמְקֻדָּשׁ בְּכֹחוֹ, רַחוּם בְּבַקָּשָׁתִי, כִּי עַל כֵּן הוֹדַעְתָּ לִי וְהִגְלֵיתָ אֵלַי, כַּאֲשֶׁר הֶעֱלֵיתָ אוֹתִי אֶל רוּמְךָ; לָכֵן הוֹדַעְתָּ לְיְדִידְךָ זֶה כְּכֹל אֲשֶׁר שָׁאַלְתִּי — הַהְאָרַכְתָּ אוֹ הָיָה כְּכֹל אֲשֶׁר רָאִיתִי. וַיַּרְאֵנִי אֶת הֲמוֹן עַמּוֹ, וַיֹּאמֶר אֵלַי: לָכֵן רָאִיתָ אֶת אַרְבַּעַת הַמּוֹרָדוֹת [יב] — יִהְיֶה לִי מֵהֶם קֶצֶף, וּבָהֶם גְּמוּל מַעֲשֵׂיהֶם מֵאִתִּי. וּבָרְבִיעִי — מוֹרָד שֶׁל שְׁנֵים עָשָׂר שָׁנָה, וּשְׁעַת עוֹלָם אַחַת, וְהִיא גַּם כֵּן שָׁנָה, תִּהְיֶה בְרָעָה בָּעַמִּים, וְשָׁעָה בְּחֶסֶד וּבְחֶרְפָּה בַּגּוֹיִם.
