הָאֵשׁ נִכְבֶּדֶת מֵאֱלֹהֶיךָ הַזָּהָב וְהַכֶּסֶף וְהָאֶבֶן וְהַבַּרְזֶל וְהָעֵץ, כִּי הָאֵשׁ שׂוֹרֶפֶת אֶת אֱלֹהֶיךָ, וֵאלֹהֶיךָ הַנִּשְׂרָפִים נִכְנָעִים לָאֵשׁ; הָאֵשׁ מִתְלוֹצֶצֶת מֵהֶם וְאוֹכַלְתָּם. אַךְ גַּם אוֹתָהּ לֹא אֶקְרָא אֵל, כִּי הִיא נִכְנַעַת לַמַּיִם, וְהַמַּיִם נִכְבָּדִים מִמֶּנָּה, כִּי גוֹבְרִים עַל הָאֵשׁ וּמַמְתִּיקִים פֵּרוֹת הָאָרֶץ. אַךְ גַּם אוֹתָם לֹא אֶקְרָא אֵל, כִּי הַמַּיִם נִכְנָעִים תַּחַת הָאָרֶץ; וְהָאָרֶץ נִכְבֶּדֶת מִן הַמַּיִם, כִּי גוֹבֶרֶת עַל טֶבַע הַמַּיִם — אַךְ גַּם אוֹתָהּ לֹא אֶקְרָא אֵל, כִּי הַשֶּׁמֶשׁ מְיַבֶּשֶׁת אוֹתָהּ, וְהִיא נְתוּנָה לִמְלֶאכֶת אָדָם. וְהַשֶּׁמֶשׁ נִכְבֶּדֶת מִן הָאָרֶץ, כִּי בְּקַרְנֶיהָ הִיא מְאִירָה לְכָל הַיְקוּם — אַךְ גַּם אוֹתוֹ לֹא אֶקְרָא אֵל, כִּי בְּבוֹא הַלַּיְלָה הוּא נֶעְדָּם בַּחֹשֶׁךְ; וְגַם אֶת הַיָּרֵחַ וְהַכּוֹכָבִים לֹא אֶקְרָא אֵל, כִּי בִּזְמַנָּם בַּלַּיְלָה הֵם מַעֲמִימִים אֶת אוֹרָם. אַךְ שְׁמַע, תֶּרַח אָבִי, וְאַגִּיד לְךָ אֶת הָאֵל אֲשֶׁר עָשָׂה אֶת הַכֹּל: הוּא הָאֵל הָאֲמִתִּי, אֲשֶׁר צָבַע אֶת הַשָּׁמַיִם, וְהִזְהִיב אֶת הַשֶּׁמֶשׁ, וְהֵאִיר אֶת הַיָּרֵחַ וְאִתָּהּ הַכּוֹכָבִים, וְיִבֵּשׁ אֶת הָאָרֶץ בְּתוֹךְ מַיִם רַבִּים. וְאוֹתְךָ עַצְמְךָ שָׁת בַּחֹשֶׁךְ, וְאוֹתִי עוֹרֵר מִמְּבוּכַת מַחְשְׁבוֹתַי. אִם יֵרָאֶה לָנוּ בְּעַצְמוֹ — הוּא הָאֵל.

וַיְּהִי כְּכַלּוֹת אַבְרָהָם לְדַבֵּר אֶת הַדְּבָרִים הָאֵלֶּה אֶל תֶּרַח אָבִיו בְּחַצְרוֹת בֵּיתוֹ, וְשָׁמַע תֶּרַח אָבִיו וְהִבִּיט בְּעֵינָיו הַכְּבֵדוֹת בְּאַבְרָהָם — כִּי נֶחְשָׁב בְּעֵינָיו לְהַשְׁמִיד אֶת יְמֵי אַבְרָהָם — וַיַּחֱזִיק בְּמִרְמִית נָחוֹר אָבִיו אֲשֶׁר הוֹרִישׁ לוֹ.

> **הערת המהדורה:** כאן מוסיף עורך הפאלאה פנייה פרשנית קצרה ("הבינות נא, נביא טוב, מה אמר אברהם: אם ייראה לנו האל בעצמו — מה הוא 'להיראות בעצמו'? ואם לא תדע — דום; כי אברהם ראה בעיניו הרוחניות את כל העתידות, וכך היה"). פנייה זו — כפניית ה"יהודי" שבפרק ז' — היא תוספת העורך שאינה מחזון אברהם המקורי; ראו נספח, הערה 11.
