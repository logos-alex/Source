וַיְּהִי כְּכַלּוֹת אַבְרָהָם לְדַבֵּר אֶת הַדְּבָרִים הָאֵלֶּה אֶל תֶּרַח אָבִיו בְּחַצְרוֹת בֵּיתוֹ, וְשָׁמַע תֶּרַח אָבִיו וְהִבִּיט בְּעֵינָיו הַכְּבֵדוֹת בְּאַבְרָהָם — כִּי נֶחְשָׁב בְּעֵינָיו לְהַשְׁמִיד אֶת יְמֵי אַבְרָהָם — וַיַּחֱזִיק בְּמִרְמִית נָחוֹר אָבִיו אֲשֶׁר הוֹרִישׁ לוֹ.

> **הערת המהדורה:** כאן מוסיף עורך הפאלאה פנייה פרשנית קצרה ("הבינות נא, נביא טוב, מה אמר אברהם: אם ייראה לנו האל בעצמו — מה הוא 'להיראות בעצמו'? ואם לא תדע — דום; כי אברהם ראה בעיניו הרוחניות את כל העתידות, וכך היה"). פנייה זו — כפניית ה"יהודי" שלהלן — היא תוספת העורך שאינה מחזון אברהם המקורי; ראו נספח, הערה 11.

וַיְהִי אַבְרָהָם בְּתוֹךְ עַצְמוֹ וַיֹּאמַר: אֲנַסֶּה אֶת אֱלֹהֵי אָבִי, אִם יוּכְלוּ לְהוֹשִׁיעַ לְעַצְמָם. 

וַיִּקַּח אַבְרָהָם אֵשׁ וַיִּצְתֵ אֶת הַהֵיכָל אֲשֶׁר שָׁם עָמְדוּ אֱלֹהֵי אָבִיו. 

וַיַּרְא הָרָן אֲחִי אַבְרָהָם, וַיְקַנֵּא לָאֱלִילִים, וַיְחַפֵּץ לְהוֹצִיאָם — וְהוּא עַצְמוֹ נִשְׂרַף שָׁם וַיָּמָת הָרָן לְפָנָיו. 

כִּי קֹדֶם לָכֵן לֹא מֵת בֵּן לְפָנִים אָבִיו, כִּי אִם אָב לְפָנִים בְּנוֹ; וּמִזֶּה הַחֵלּוּ בָּנִים לָמוּת לְפָנִים אֲבוֹתֵיהֶם. 

וַיֶּאֱהַב הָאֵל אֶת אַבְרָהָם, וַיֹּאמֶר אֵלָיו: אַבְרָהָם אַבְרָהָם, אֶת הָאֵל בַּאֱלֹהִים בִּקַּשְׁתָּ; צֵא מִבֵּית אָבִיךָ וְלֵךְ אֶל הָאָרֶץ אֲשֶׁר אַרְאֶךָּ, וְאֶעֶשְׂךָ לְגוֹי גָּדוֹל, כִּי בְךָ יִתְבָּרְכוּ כֹּל מִשְׁפְּחוֹת הָאָרֶץ.

> **הערת המהדורה:** לאחר הקריאה מוסיף עורך הפאלאה פנייה פולמוסית אל "היהודי" המבארת את הברכה שבאברהם ברוח נוצרית (תוספת העורך שאינה מחזון אברהם המקורי; ראו נספח, הערה 11), ואחריה שב המספר לדברי החזון: "ויהי כאשר חשב אברהם כזאת אל תרח אביו בחצר ביתו".

וַיְהִי בְּעֵת אֲשֶׁר חָשַׁב אַבְרָהָם כָּזֹאת אֶל תֶּרַח אָבִיו בְּחַצַר בֵּיתוֹ, וְקוֹל גִּבּוֹר נָפַל מִן הַשָּׁמַיִם בְּזֶרֶם אֵשׁ, קוֹרֵא: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. 

וַיֹּאמֶר: אֶת הָאֵל הַבּוֹרֵא אַתָּה מְבַקֵּשׁ בְּמַחְשֶׁבֶת לִבְּךָ — אֲנִי הוּא. 

צֵא מֵאֵת תֶּרַח אָבִיךָ וּמִן הַבַּיִת, פֶּן תּוּמַת גַּם אַתָּה בְּחַטֹּאות בֵּית אָבִיךָ. 

וָאֵצֵא, וַיְהִי כְּצֵאתִי וּבְטֶרֶם הִגַּעְתִּי אֶל פֶּתַח הֶחָצֵר, וְקוֹל רַעַם גָּדוֹל בָּא וְאֵשׁ יָרְדָה מִן הַשָּׁמַיִם וַתִּשְׂרֹף אֶת בֵּית אָבִיו וְאֶת כָּל אֲשֶׁר בּוֹ — אֲנָשִׁים וּבְהֵמוֹת — עַד הָאָרֶץ, אַרְבָּעִים אַמָּה. 

אָז בָּא הַקּוֹל אֵלַי פַּעֲמַיִם: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. 

וַיֹּאמֶר: אֲנִי הוּא, אַל תִּירָא, כִּי אֲנִי לִפְנֵי הָעוֹלָמִים וְאֵל גִּבּוֹר, אֲשֶׁר בָּרָאתִי הָעוֹלָמִים וְאֵת הָאָרֶץ, וְאַחַר כֵּן אֶת מְאוֹר הָעוֹלָם וְהָאוֹר הָרִאשׁוֹן.
