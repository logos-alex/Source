קוֹל גִּבּוֹר נָפַל מִן הַשָּׁמַיִם בְּזֶרֶם אֵשׁ, קוֹרֵא: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: אֶת הָאֵל הַבּוֹרֵא אַתָּה מְבַקֵּשׁ בְּמַחְשֶׁבֶת לִבְּךָ — אֲנִי הוּא. צֵא מֵאֵת תֶּרַח אָבִיךָ וּמִן הַבַּיִת, פֶּן תּוּמַת גַּם אַתָּה בְּחַטֹּאות בֵּית אָבִיךָ. וָאֵצֵא, וַיְהִי כְּצֵאתִי וּבְטֶרֶם הִגַּעְתִּי אֶל פֶּתַח הֶחָצֵר, וְקוֹל רַעַם בָּא וַיִּשְׂרֹף אוֹתוֹ וְאֶת בֵּיתוֹ וְאֵת כָּל אֲשֶׁר בְּבֵיתוֹ עַד הָאָרֶץ. אָז בָּא הַקּוֹל אֵלַי פַּעֲמַיִם: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: אֲנִי הוּא, אַל תִּירָא, כִּי אֲנִי לִפְנֵי הָעוֹלָמִים וְאֵל גִּבּוֹר, אֲשֶׁר בָּרָאתִי הָעוֹלָמִים וְאֵת הָאָרֶץ, וְאַחַר כֵּן אֶת מְאוֹר הָעוֹלָם וְהָאוֹר הָרִאשׁוֹן. וַאֲנִי מָגֵן לְךָ וְשׁוֹמְרֶךָ וְעֶזְרְךָ.

לֵךְ קַח לִי עֶגְלָה מְשֻׁלֶּשֶׁת וְאַיִל מְשֻׁלָּשׁ וְעֵז מְשֻׁלֶּשֶׁת, וְתֹר וְגוֹזָל, וְהַקְרֵב לִי קָרְבָּן טָהוֹר, וּבַקָּרְבָּן הַהוּא אָקִים לְךָ עוֹלָמִים, וְאוֹדִיעֲךָ אֶת הַשְּׁמוּרוֹת, וּתְרָאֶה גְּדֹלוֹת אֲשֶׁר לֹא רָאִיתָ, כִּי אָהַבְתָּ לְבַקְּשֵׁנִי וָאֶקְרָא לְךָ אוֹהֲבִי. אַךְ הִשָּׁמֵר לְךָ מִכָּל מַאֲכָל הַבָּא מֵאֵשׁ, וּמִשְּׁתוֹת יַיִן, וּמִסּוּךְ שֶׁמֶן, אַרְבָּעִים יוֹם. אָז תַּקְרִיב לִי אֶת הַקָּרְבָּן אֲשֶׁר צִוִּיתִיךָ, בַּמָּקוֹם אֲשֶׁר אַרְאֶה אוֹתְךָ עַל הָהָר הָרָם, וְשָׁם אַרְאֶה לְךָ אֶת הָעוֹלָמִים, וּבִדְבָרִי אֶת הַבְּרוּאִים וְהַמְכוֹנָנִים וְהָעֲשׂוּיִם וְהַמְחֻדָּשִׁים, וְאוֹדִיעֲךָ בָהֶם אֶת הַבָּאוֹת עַל עֹשֵׂי הָרַע וְעַל הַצַּדִּיקִים בְּדוֹר בְּנֵי אָדָם. וַיְהִי כְּשָׁמְעִי אֶת הַקּוֹל מְדַבֵּר אֵלַי דְּבָרִים כָּאֵלֶּה, וָאַבִּיט הֵנָּה וָהֵנָּה, וְהִנֵּה אֵין נִשְׁמַת אָדָם, וָאֶחְרַד בְּרוּחִי, וַתֹּאבַד נַפְשִׁי מִמֶּנִּי, וָאֱהִי כְּמֵת וָאֶפֹּל אַפַּיִם אַרְצָה כְּאֶבֶן, כִּי לֹא הָיָה עוֹד בִּי כֹּחַ לַעֲמֹד עַל הָאָרֶץ. וָאֶשְׁמַע קוֹל קָדוֹשׁ אֹמֵר: יָהאֵל, בְּתִוּוּךְ שְׁמִי הַמְפֹרָשׁ, קוּם אֶת הָאִישׁ הַזֶּה וְחַזְּקֵהוּ מִן הָרְעָדָה. וַיָּבֹא הַמַּלְאָךְ אֲשֶׁר שֻׁלַּח אֵלַי בִּדְמוּת אִישׁ, וַיַּחֲזֵק בְּיָדִי הַיְמָנִית וַיַּעֲמִידֵנִי עַל רַגְלָי.

וַיֹּאמֶר אֵלַי: קוּם אַבְרָהָם, יְדִיד אֱלֹהִים אֲשֶׁר אֲהֵבְךָ, אַל תַּחֲזִיקְךָ חֲרָדַת אֱנוֹשׁ; הִנֵּה שָׁלוּחַ אֲנִי אֵלֶיךָ לְחַזְּקֶךָ וּלְבָרְכֶךָ בְּשֵׁם הָאֱלֹהִים אֲשֶׁר אֲהֵבְךָ, בּוֹרֵא הַשָּׁמַיִם וְהָאָרֶץ. חֲזַק וּמַהֵר אֵלָיו. אֲנִי הוּא הַנִּקְרָא כֵן [יָהאֵל], עַל יְדֵי הַמֵּנִיעַ אֶת הַנִּמְצָא עִמִּי בָּרָקִיעַ הַשְּׁבִיעִי. גְּבוּרָה בְּתוֹךְ הַשֵּׁם הַמְפֹרָשׁ הַשֹּׁכֵן בִּי. אֲנִי הוּא הַנָּתוּן עַל פִּי מִצְוָתוֹ לְהַשְׁקִיט אֶת חַיּוֹת הַכְּרוּבִים אִישׁ בְּאָחִיו. וּמְלַמֵּד אֲנִי אֶת נֹשְׂאֵי שִׁירָתוֹ בְּתוֹךְ לֵיל אָדָם, בַּשָּׁעָה הַשְּׁלִישִׁית. אֲנִי הוּא הַמְצֻוֶּה לְהַחְזִיק בְּלִוְיָתָן, כִּי בְּיָדִי נִכְנָע מִשְׁלַח כָּל שֶׁרֶץ וְאֵימָתוֹ. אֲנִי הוּא הַמְצֻוֶּה לְהַתִּיר אֶת הַשְּׁאוֹל, לְהַשְׁמִיד אֶת הַמִּתְפַּלְּאִים עַל הַמֵּתִים. אֲנִי הוּא הַמְצֻוֶּה לְהַצִּית אֶת בֵּית אָבִיךָ עִמּוֹ, כִּי כָבוֹד עָשָׂה לַמֵּתִים. שָׁלוּחַ אֲנִי אֵלֶיךָ עַתָּה לְבָרְכֶךָ וְאֶת הָאָרֶץ אֲשֶׁר הֵכִין לְךָ הַקַּדְמוֹן וְנִקְרֵאתָ בָּהּ, וּבַעֲבוּרְךָ יִשַּׁרְתִּי אֶת הַדֶּרֶךְ בָּאָרֶץ. קוּם אַבְרָהָם, לֵךְ הַלֵּךְ בְּעֹז, שְׂמַח וְגִיל, וַאֲנִי עִמְּךָ, כִּי הוּכַן לְךָ כָּבוֹד עוֹלָם מֵאֵת הַקַּדְמוֹן. לֵךְ הַשְׁלֵם אֶת קָרְבַּן הַמִּצְוָה, כִּי הִנְנִי מְמֻנֶּה עִמְּךָ וְעִם הַזֶּרַע הַמּוּכָן מִמְּךָ, וְעִמִּי מִיכָאֵל יְבָרֶכְךָ עַד עוֹלָם. חֲזַק וָלֵךְ. וָאָקוּם וָאֵרֶא אֶת הָאוֹחֵז בִּימִינִי וְהַמַּעֲמִידֵנִי עַל רַגְלָי; וְהָיָה מַרְאֵה גּוּפוֹ [רַגְלָיו] כְּסַפִּיר, וּמַרְאֵה פָנָיו כְּתַרְשִׁישׁ, וּשְׂעַר רֹאשׁוֹ כַּשֶּׁלֶג, וּמִצְנֶפֶת עַל רֹאשׁוֹ כְּמַרְאֵה קֶשֶׁת בֶּעָנָן, וּלְבוּשׁ בְּגָדָיו אַרְגָּמָן, וּמַטֶּה זָהָב בְּיָדוֹ.

וַיֹּאמֶר אֵלַי: אַבְרָהָם. וָאֹמַר הִנְנִי עַבְדֶּךָ. וַיֹּאמֶר:
