וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם וְהָבֵן אֶת אֲשֶׁר אַגִּיד לְךָ, וַעֲנֵנִי כְּכֹל אֲשֶׁר אֶשְׁאָלְךָ: תֶּרַח אָבִיךָ, מַדּוּעַ לֹא שָׁמַע בְּקוֹלְךָ וְלֹא סָר מִשֵּׁדֵי הַפְּסִילִים, עַד אֲשֶׁר אָבַד כָּל בֵּיתוֹ עִמּוֹ. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, אָכֵן לֹא הָיָה בִרְצוֹנוֹ לִשְׁמֹעַ לִי, וְגַם אֲנִי לֹא הָלַכְתִּי אַחֲרֵי מַעֲשָׂיו. וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, כַּאֲשֶׁר עֲצַת אָבִיךָ בּוֹ, כֵּן עֲצָתְךָ בְךָ, כֵּן גַּם עֲצַת רְצוֹנִי מוּכָנָה בִּי, לַיָּמִים הַבָּאִים לִפְנֵיהֶם; לֹא תֵדַע, וְלֹא תִרְאֶה בְעֵינֶיךָ אֶת הָעֲתִידוֹת, כִּי שֶׁל זַרְעֲךָ הֵם. הַבֵּט בַּתְּמוּנָה. וָאַבִּיט וָאֵרֶא, וְהִנֵּה רָעֲדָה הַתְּמוּנָה, וַיָּרָץ מִצִּדָּהּ הַשְּׂמָאלִי עַם נָכְרִי, וַיִּבְזְזוּ אֶת אֲשֶׁר מִצַּד יָמִין, אֲנָשִׁים וְנָשִׁים וִילָדִים; מִקְצָתָם הָרְגוּ, וּמִקְצָתָם הֶחֱזִיקוּ לָהֶם. הִנֵּה רָאִיתִי בָּאִים אֲלֵיהֶם מוֹרָדוֹת, וַיַּצִּיתוּ אֶת הַהֵיכָל בָּאֵשׁ וַיָּבֹזּוּ אֶת כָּל הַקֳּדָשִׁים אֲשֶׁר בּוֹ.
