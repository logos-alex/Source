וָאַבִּיט וָאֵרֶא שָׁם אֵת אֲשֶׁר הָיָה לְפָנַי בַּבְּרִיאָה — וָאֵרֶא אֶת אָדָם וְאֶת חַוָּה אֲשֶׁר עִמּוֹ, וְעִמָּהֶם הָאוֹיֵב הֶעָרוּם, וְאֶת קַיִן פֹּעֵל עָוֶל בְּיַד הָאוֹיֵב, וְאֶת הֶבֶל הַנֶּהֱרָג, וְאֶת הַכִּלָּיוֹן הַמּוּבָא עַל הָרָשָׁע וְהַנָּתוּן לוֹ. 

וָאֵרֶא שָׁם אֶת הַזְּנוּת וְאֶת הַחוֹמְדִים אוֹתָהּ וְאֶת טֻמְאָתָהּ, וְאֶת קִנְאָתָם וְאֵשׁ כִּלְיוֹנָם בְּתַחְתִּיּוֹת הָאָרֶץ. 

וָאֵרֶא שָׁם אֶת הַגְּנֵבָה וְאֶת הַנֶּחְפָּזִים אֵלֶיהָ וְאֶת עֲשִׂיָּתָהּ וְאֶת גְּמוּלָהּ — מִשְׁפָּט גָּדוֹל. 

וָאֵרֶא גַם אֲנָשִׁים עֲרֻמִּים, מִצְחָם זֶה מוּל זֶה, וְעֶרְוָתָם, וְנֵזֶק אִישׁ בְּרֵעֵהוּ וּגְמוּלָם. 

וָאֵרֶא שָׁם אֶת הַתַּאֲוָה, וּבְיָדָהּ רֹאשׁ כָּל עַוְלָה, וְאֶת שְׁתִיקָתָהּ וּפִזּוּרָהּ, נְתוּנָה לְאָבְדָּן.

וָאֵרֶא שָׁם אֶת דְּמוּת פֶּסֶל הַקִּנְאָה, כְּמַעֲשֵׂה חָרָשׁ עֵצִים כַּאֲשֶׁר עָשָׂה אָבִי, גּוּפוֹ כְּמַרְאֵה נְחֹשֶׁת, וּלְפָנָיו אִישׁ מִשְׁתַּחֲוֶה לוֹ, וּמִזְבֵּחַ נֶגְדּוֹ, וּנְעָרִים נִשְׁחָטִים עָלָיו לִפְנֵי פְנֵי הַפֶּסֶל.

וָאֹמַר: מַה זֶּה הַפֶּסֶל, אוֹ מַה הַמִּזְבֵּחַ, אוֹ מִי הַנִּזְבָּחִים, אוֹ מִי הַזֹּבֵחַ, אוֹ מַה זֶּה הַהֵיכָל אֲשֶׁר אֲנִי רוֹאֶה, נָאֶה בְּחָכְמַת מַעֲשֵׂהוּ וּבְיָפְיוֹ, מִתַּחַת לְכִסֵּא כְּבוֹדְךָ. 

וַיֹּאמֶר: שְׁמַע אַבְרָהָם, זֶה אֲשֶׁר רָאִיתָ — הַהֵיכָל וְהַמִּזְבֵּחַ — נָאֶה הוּא לִי, מַחְשֶׁבֶת הַכְּהֻנָּה לְשֵׁם כְּבוֹדִי, אֲשֶׁר בָּהּ יִשְׁכֹּן כָּל קְהַל אֲנָשִׁים, וּמְבוֹא מְלָכִים וּנְבִיאִים, וְכָל קָרְבָּן אֲשֶׁר אָשִׂים לְהֵעָשׂוֹת לִי בְּעַמִּי הַבָּא מִזֶּרְעֲךָ. 

וְהַגּוּף אֲשֶׁר רָאִיתָ הוּא כַּעֲסִי, אֲשֶׁר בּוֹ יַכְעִיסוּנִי הַבָּא מִמְּךָ אֵלָי; וְהָאִישׁ אֲשֶׁר רָאִיתָ נִשְׁחָט, הוּא קָרְבַּן הָרֶצַח הַמַּכְעִיס, אֲשֶׁר יִהְיוּ לִי לְעֵדוּת מִשְׁפָּט בְּקֵץ הַתַּכְלִית, בְּרֹאשִׁית הַבְּרִיאָה. 

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, לָמָּה יְסַדְתָּ אֵפוֹא לִהְיוֹת כָּכָה; הָשֵׁב נָא אֶת הַדַּעַת הַזֹּאת.

וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם וְהָבֵן אֶת אֲשֶׁר אַגִּיד לְךָ, וַעֲנֵנִי כְּכֹל אֲשֶׁר אֶשְׁאָלְךָ: תֶּרַח אָבִיךָ, מַדּוּעַ לֹא שָׁמַע בְּקוֹלְךָ וְלֹא סָר מִשֵּׁדֵי הַפְּסִילִים, עַד אֲשֶׁר אָבַד כָּל בֵּיתוֹ עִמּוֹ. 

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, אָכֵן לֹא הָיָה בִרְצוֹנוֹ לִשְׁמֹעַ לִי, וְגַם אֲנִי לֹא הָלַכְתִּי אַחֲרֵי מַעֲשָׂיו.
