וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, כַּאֲשֶׁר עֲצַת אָבִיךָ בּוֹ, כֵּן עֲצָתְךָ בְךָ, כֵּן גַּם עֲצַת רְצוֹנִי מוּכָנָה בִּי, לַיָּמִים הַבָּאִים לִפְנֵיהֶם; לֹא תֵדַע, וְלֹא תִרְאֶה בְעֵינֶיךָ אֶת הָעֲתִידוֹת, כִּי שֶׁל זַרְעֲךָ הֵם. 

הַבֵּט בַּתְּמוּנָה. 

וָאַבִּיט וָאֵרֶא, וְהִנֵּה רָעֲדָה הַתְּמוּנָה, וַיָּרָץ מִצִּדָּהּ הַשְּׂמָאלִי עַם נָכְרִי, וַיִּבְזְזוּ אֶת אֲשֶׁר מִצַּד יָמִין, אֲנָשִׁים וְנָשִׁים וִילָדִים; מִקְצָתָם הָרְגוּ, וּמִקְצָתָם הֶחֱזִיקוּ לָהֶם. 

הִנֵּה רָאִיתִי בָּאִים אֲלֵיהֶם מוֹרָדוֹת, וַיַּצִּיתוּ אֶת הַהֵיכָל בָּאֵשׁ וַיָּבֹזּוּ אֶת כָּל הַקֳּדָשִׁים אֲשֶׁר בּוֹ.

וָאֹמַר: קַדְמוֹן, הִנֵּה אֵלֶּה אֲשֶׁר מִמֶּנִּי לָקַחְתָּ עָם, בּוֹזְזִים אוֹתָם הֲמוֹן עַמִּים — מִקְצָתָם הוֹרְגִים וּמִקְצָתָם מַחֲזִיקִים כְּגֵרִים, וְאֶת הַהֵיכָל שׂוֹרְפִים בָּאֵשׁ וְאֶת הַמַּעֲשִׂים הַיָּפִים בּוֹ בּוֹזְזִים וְהוֹרְסִים. 

קַדְמוֹן וְגִבּוֹר, אִם כֵּן הוּא, לָמָּה עַתָּה הֶעֱצַבְתָּ אֶת לִבִּי, וְלָמָּה כֵן יִהְיֶה. 

וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, כְּכֹל אֲשֶׁר רָאִיתָ בַּעֲבוּר זַרְעֲךָ הַמַּכְעִיסִים אוֹתִי — הַגּוּפוֹת וְהָרְצִיחוֹת בַּתְּמוּנָה, בַּהֵיכָל, בְּקִנְאָה — כֵּן יִהְיֶה. 

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יַעַבְרוּ נָא הָרָעוֹת הָאֵלֶּה בְּמַעֲשֵׂי הָרֶשַׁע, אַךְ הָקֵם אֶת מִצְוֹתֶיךָ בָהֶם, יוֹתֵר מִמַּעֲשֵׂי צֶדֶק זֶה — כִּי תוּכַל לַעֲשׂוֹת זֹאת.

וַיֹּאמֶר אֵלַי: אַדְרַבָּה עֵת הַצֶּדֶק תִּפְגַּע, לִפְנֵי דֻגְמַת מְלָכִים וְשׁוֹפְטִים בְּצֶדֶק, אֲשֶׁר בָּרָאתִי מֵאָז לִמְשֹׁל מִתּוֹכָם בָּהֶם; וּמֵהֶם יֵצְאוּ אֲנָשִׁים אֲשֶׁר יְמַהֲרוּ אֶת אֲשֶׁר הִגַּדְתִּי לְךָ וְרָאִיתָ. 

וָאַעַן וָאֹמַר: גִּבּוֹר, הַמְקֻדָּשׁ בְּכֹחוֹ, רַחוּם בְּבַקָּשָׁתִי, כִּי עַל כֵּן הוֹדַעְתָּ לִי וְהִגְלֵיתָ אֵלַי, כַּאֲשֶׁר הֶעֱלֵיתָ אוֹתִי אֶל רוּמְךָ; לָכֵן הוֹדַעְתָּ לְיְדִידְךָ זֶה כְּכֹל אֲשֶׁר שָׁאַלְתִּי — הַהְאָרַכְתָּ אוֹ הָיָה כְּכֹל אֲשֶׁר רָאִיתִי.

וַיַּרְאֵנִי אֶת הֲמוֹן עַמּוֹ, וַיֹּאמֶר אֵלַי: לָכֵן רָאִיתָ אֶת אַרְבַּעַת הַמּוֹרָדוֹת [יב] — יִהְיֶה לִי מֵהֶם קֶצֶף, וּבָהֶם גְּמוּל מַעֲשֵׂיהֶם מֵאִתִּי. 

וּבָרְבִיעִי — מוֹרָד שֶׁל מֵאָה שָׁנָה, וּשְׁעַת עוֹלָם אַחַת, וְהִיא גַם כֵּן מֵאָה שָׁנָה, תִּהְיֶה בְרָעָה בָּעַמִּים, וְשָׁעָה בְּחֶסֶד וּבְחֶרְפָּה בַּגּוֹיִם.

וָאֹמַר: קַדְמוֹן וְגִבּוֹר, כַּמָּה זְמַן שְׁעַת עוֹלָם.
