וַיֹּאמֶר אֵלַי: אַבְרָהָם. וָאֹמַר הִנְנִי עַבְדֶּךָ. וַיֹּאמֶר: אַל יַבְעִתְךָ מַרְאִי, וְלֹא שִׂיחִי יַבְהִיל אֶת נַפְשְׁךָ; לֵךְ עִמִּי, וְאֵלֵךְ עִמְּךָ, נִרְאֶה עַד הַקָּרְבָּן, וְאַחַר הַקָּרְבָּן לֹא אֵרָאֶה עוֹד עַד עוֹלָם. חֲזַק וָלֵךְ. וַנֵּלֵךְ שְׁנֵינוּ יַחַד אַרְבָּעִים יוֹם וָלַיְלָה, וְלֶחֶם לֹא אָכַלְתִּי וּמַיִם לֹא שָׁתִיתִי, כִּי מַאֲכָלִי הָיָה לְהִסְתּוֹפֵף בְּעֵינַי אֶל הַמַּלְאָךְ אֲשֶׁר עִמִּי, וְשִׂיחוֹ אֲשֶׁר עִמִּי — הוּא הָיָה מַאֲכָלִי וּמַשְׁקִי. וַנָּבֹא אֶל הַר הָאֵל הַנִּכְבָּד, חוֹרֵב. וָאֹמַר אֶל הַמַּלְאָךְ, מְשׁוֹרֵר הַקַּדְמוֹן: הִנֵּה אֵין עִמִּי קָרְבָּן, וְלֹא מָקוֹם מִזְבֵּחַ אֵדַע בָּהָר, וְאֵיךְ אֶעֱשֶׂה קָרְבָּן. וַיֹּאמֶר: הַבֵּט אַחֲרֶיךָ. וְהִנֵּה כָּל הַקָּרְבָּן הַמְצֻוֶּה בָּא אַחֲרֵינוּ — הָעֵגֶל וְהָעֵז וְהָאַיִל וְהַתּוֹר וְהַגּוֹזָל. וַיֹּאמֶר אֵלַי הַמַּלְאָךְ: אַבְרָהָם. וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: אֶת כֻּלָּם שְׁחַט וְנַתֵּחַ, בַּתֵּר לִשְׁנַיִם בְּיֹשֶׁר; אַךְ הָעוֹף לֹא תְבַתֵּר. וּתְנֵם לָאֲנָשִׁים אֲשֶׁר אַרְאֶה לְךָ עֹמְדִים אֶצְלְךָ, כִּי הֵם הַמִּזְבֵּחַ בָּהָר לְהַקְרִיב קָרְבָּן לַקַּדְמוֹן. וְאֶת הַתּוֹר וְאֶת הַגּוֹזָל תֵּן לִי, כִּי אֲנִי אֶעֱלֶה עַל כַּנְפֵי הָעוֹף לְהַרְאוֹתְךָ בַּשָּׁמַיִם וּבָאָרֶץ וּבַיָּם וּבַתְּהוֹמוֹת וּבַשְּׁאוֹל וּבְגַן עֵדֶן וּבִנְהָרוֹתָיו, וּבִמְלֹא הַתֵּבֵל תִּרְאֶה, וְאֶת סְבִיבָתָהּ תִּרְאֶה בְּכֻלָּן. וָאַעַשׂ כִּדְבַר הַמַּלְאָךְ, וָאֶתֵּן לַמַּלְאָכִים הַבָּאִים אֵלֵינוּ אֶת הַבְּתָרִים, וְאֶת הָעוֹף לָקַח הַמַּלְאָךְ יָפִיל, וָאֲחַכֶּה לְמִנְחַת עֶרֶב. וַתֵּרֶד עוֹף טָמֵא עַל הַגְּוִיּוֹת וָאֲגָרְשֶׁנּוּ. וַיִּקְרָא אֵלַי הָעוֹף הַטָּמֵא הַהוּא, וַיֹּאמֶר: מָה לְּךָ אַבְרָהָם, בַּמְּרוֹמִים הַקְּדוֹשִׁים אֲשֶׁר בָּהֶם לֹא יֹאכְלוּ וְלֹא יִשְׁתּוּ, וְאֵין בָּהֶם מַאֲכַל אֱנוֹשׁ — הַכֹּל שָׁם בָּאֵשׁ יִכָּלוּ, וְיִשְׂרְפוּ אוֹתְךָ.
