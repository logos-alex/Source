וְאֶת הַתּוֹר וְאֶת הַיוֹנָה תֵּן לִי, וַאֲנִי אֶעֱלֶה עַל כַּנְפֵי הָעוֹף לְהַרְאוֹתְךָ בַּשָּׁמַיִם וּבָאָרֶץ וּבַיָּם וּבַתְּהוֹמוֹת וּבַשְּׁאוֹל וּבְגַן עֵדֶן וּבִנְהָרוֹתָיו וּבִמְלֹא הַתֵּבֵל, וְתִרְאֶה אֶת כֻּלָּם סָבִיב. 

וָאַעַשׂ כִּדְבַר הַמַּלְאָךְ, וָאֶתֵּן לַמַּלְאָכִים הַבָּאִים אֵלֵינוּ אֶת הַבְּתָרִים, וְאֶת הָעוֹף לָקַח הַמַּלְאָךְ, וָאֲחַכֶּה לְמִנְחַת עֶרֶב. 

וַתֵּרֶד עוֹף טָמֵא עַל הַפְּגָרִים, וַיִּקְרָא אֵלַי: מָה לְּךָ אַבְרָהָם בַּמְּרוֹמִים הַקְּדוֹשִׁים, אֲשֶׁר שָׁם לֹא יֹאכְלוּ וְלֹא יִשְׁתּוּ, וְאֵין בָּהֶם מַאֲכַל אֱנוֹשׁ, כִּי כֻלָּם בָּאֵשׁ יִכָּלוּ וְיִשְׂרְפוּ אוֹתְךָ.

עֲזֹב אֶת הָאִישׁ הַהֹלֵךְ עִמְּךָ וּבְרַח, כִּי אִם תַּעֲלֶה אֶל הַמָּרוֹם יְאַבְּדוּךָ. 

וַיְהִי כִּרְאוֹתִי אֶת הָעוֹף מְדַבֵּר, וָאֹמַר אֶל הַמַּלְאָךְ: מַה זֹּאת אֲדֹנִי. 

וַיֹּאמֶר: זֶה עָוֹן — עֲזָאזֵל. 

וַיֹּאמֶר אֵלָיו: חֶרְפָּה לְךָ עֲזָאזֵל, כִּי חֵלֶק אַבְרָהָם בַּשָּׁמַיִם וְחֶלְקְךָ בָאָרֶץ, כִּי בָחַרְתָּ וְאָהַבְתָּ אֶת מְקוֹם טֻמְאָתְךָ. 

לָכֵן נְתָנְךָ הַקַּדְמוֹן הַשַּׁלִּיט הַגִּבּוֹר יוֹשֵׁב אָרֶץ, וּבְךָ הָרוּחַ הָרַע כֻּלּוֹ הַשֶּׁקֶר, וּבְךָ חֵמָה וְנִגְעֵי דּוֹרוֹת בְּנֵי הָרֶשַׁע. 

כִּי לֹא נָתַן הַקַּדְמוֹן הַגִּבּוֹר אֶת גּוּפוֹת הַצַּדִּיקִים בְּיָדְךָ, לְמַעַן תִּכּוֹן בָּם חַיֵּי צֶדֶק וַאֲבַדּוֹן הָרְשָׁעִים. 

שְׁמַע יוֹעֵץ, בּוֹשׁ מִמֶּנִּי, כִּי לֹא עַל כָּל הַצַּדִּיקִים נִתַּן לְךָ לְנַסּוֹתָם. 

סוּר מֵעַל הָאִישׁ הַזֶּה, לֹא תוּכַל לְהַתְעוֹתוֹ, כִּי אוֹיִבְךָ הוּא, וְשֶׁל הַהֹלְכִים אַחֲרֶיךָ וְהָאֹהֲבִים אֶת אֲשֶׁר תַּחְפֹּץ.
