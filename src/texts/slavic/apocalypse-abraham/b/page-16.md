וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִפְתְּחוּ הָרְקִיעִים מִתַּחְתַּי, וָאֵרֶא עַל הָרָקִיעַ הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו אֵשׁ פְּרוּסָה, וְאוֹר וָטַל, וַהֲמוֹן מַלְאָכִים, וּגְבוּרַת כָּבוֹד בִּלְתִּי נִרְאֶה מִלְּמַעְלָה — מִלְּבַד הַחַיּוֹת אֲשֶׁר רָאִיתִי, לֹא רָאִיתִי שָׁם אַחֵר. וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הָרָקִיעַ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים אֶת מִצְוַת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כְּפִי אֲשֶׁר הָיִיתִי עוֹמֵד עַל הַתְּלוּיִים בּוֹ. וְלֹא הָיָה בָּרָקִיעַ הַהוּא כֹּחַ בְּשׁוּם צוּרָה אַחֶרֶת, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הָרָקִיעַ הַשְּׁבִיעִי — הַכֹּחַ אֲשֶׁר רָאִיתִי — צֻוָּה לָנוּעַ. וָאֵרֶא שָׁם, בָּרָקִיעַ הַחֲמִישִׁי, אֶת כֹּחוֹת הַכּוֹכָבִים מְקַיְּמִים אֶת הַמִּצְוֹת הַמֻּטָּלוֹת עֲלֵיהֶם, וִיסוֹדוֹת הָאָרֶץ שׁוֹמְעִים לָהֶם. וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם, אַבְרָהָם. וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: הַבֵּט מִלְּמַעְלָה, וְהִנֵּה תַּחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם. וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם — עָפָר וָאֵפֶר. וַיֹּאמֶר לִי: כְּמִסְפַּר הַכּוֹכָבִים וְכֹחָם אֶתֵּן לְזַרְעֲךָ, גּוֹיִם וְעַמִּים מֻבְדָּלִים לִי בְגוֹרָלִי, עִם עֲזָאזֵל. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יְדַבֵּר נָא עַבְדְּךָ לְפָנֶיךָ, וְאַל יִחַר אַפְּךָ עַל בְּחִירְךָ: הִנֵּה בְּטֶרֶם הֲרִימוֹתַנִי, חֵרְפַנִי עֲזָאזֵל; וְעַתָּה, כְּשֶׁאֵינֶנּוּ לְפָנֶיךָ, תְּכוֹנֵן עִמּוֹ.
