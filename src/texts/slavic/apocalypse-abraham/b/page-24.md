בַּיָּמִים הָהֵם אָבִיא עַל כָּל הַתֵּבֵל עֶשֶׂר מַכּוֹת בְּרָעָה וּמַחֲלָה, וְאֶת אֲנָקַת רוּחָם הַמָּרָה אָבִיא עַל הַדּוֹרוֹת אֲשֶׁר עַל הָאָרֶץ, בַּעֲבוּר קֶצֶף וְהַשְׁחָתַת בְּרוּאֵיהֶם אֲשֶׁר יַכְעִיסוּנִי בָהֶם. שְׁמַע אֶת אֲשֶׁר אַגִּיד לְךָ, יִהְיֶה כֵן: הָרִאשׁוֹנָה — צָרָה מִדֹּחַק רָב. הַשְּׁנִיָּה — שְׂרֵפוֹת אֵשׁ רַבּוֹת. הַשְּׁלִישִׁית — אֲבַדּוֹן רָב. הָרְבִיעִית — דֶּבֶר בַּבְּהֵמָה. הַחֲמִישִׁית — רָעָב בְּתוֹשְׁבֵי הָעוֹלָם. הַשִּׁשִּׁית — הֶרֶג בְּחֶרֶב וְאָבְדָן. הַשְּׁבִיעִית — בָּרָד וְרֹב שֶׁלֶג. הַשְּׁמִינִית — חַיַּת שָׂדֶה תִּהְיֶה קִבְרָם. הַתְּשִׁיעִית — רָעָב וָדֶבֶר יַחֲלִיפוּ כִלָּיוֹן. הָעֲשִׂירִית — עֹנֶשׁ חֶרֶב וּמְבוּסָה בְּצָרוֹת, רַעַשׁ וְקוֹלוֹת מַשְׁחִיתִים. אָז אֶתְקַע בַּשּׁוֹפָר מִן הָאֲוִיר וְאֶשְׁלַח אֶת בְּחִירִי, שֶׁבּוֹ כָּל כֹּחִי מִדָּה אַחַת, וְהוּא יִקְרָא לְעַמִּי הַנֶּעֱלָבִים מִן הַגּוֹיִם, מִגּוֹי אֶל גּוֹי. וַאֲנִי אֶשְׂרֹף בָּאֵשׁ אֶת הַמְחָרְפִים אוֹתָם וְאֶת הַשּׁוֹלְטִים בָּהֶם לְעוֹלָם, וְאֶתֵּן אֶת הַמְכַסִּים אוֹתָם לְחֶרְפָּה וּלְבִזָּיוֹן שֶׁל הָעוֹלָם הַזֶּה, כִּי הֲכִינוֹתִי לָהֶם מַאֲכַל אֵשׁ שְׁאוֹל, מְעוֹף בְּלִי הֶפְסֵק בָּאֲוִיר, תַּחַת הָאָרֶץ, בַּתַּחְתִּיּוֹת, וּמְלֵאֵי בֶטֶן תּוֹלַעַת. כִּי יִרְאוּ בָם עֹשֵׂי הַצֶּדֶק, אֲשֶׁר בָּחֲרוּ בִּרְצוֹנִי וְשָׁמְרוּ אֶת מִצְוֹתַי, וְיִשְׂמְחוּ הַשְּׂמֵחִים עַל אֲבַדּוֹן הָאֲנָשִׁים הָעֹזְבִים, הַהֹלְכִים אַחֲרֵי הַפְּסִילִים וְאַחֲרֵי רְצִיחוֹתֵיהֶם; יִהְיוּ נִמַקִּים בְּתוֹךְ מְעֵי הַתּוֹלַעַת הָרָעָה, עֲזָאזֵל, וּנְשׂוּרְפֵי אֵשׁ עַמֵּי עֲזָאזֵל — כִּי חִכִּיתִי לָהֶם שֶׁיָּבֹאוּ אֵלָי, וְלֹא אָבוּ, וַיְכַבְּדוּ זָר, וַיִּתְחַבְּרוּ אֶל אֲשֶׁר לֹא הָיָה חֶלְקָם, וְאֶת הָאָדוֹן הַגִּבּוֹר עָזָבוּ. לָכֵן שְׁמַע אַבְרָהָם וּרְאֵה: זֶה דּוֹרְךָ הַשְּׁבִיעִי יָבֹא עִמְּךָ, וְיֵלְכוּ אֶל אֶרֶץ נָכְרִיָּה, וְיַעֲבִידוּם וִיעַנּוּם כְּשָׁעָה אַחַת שֶׁל הָעוֹלָם הַזֶּה שֶׁלֹּא יִשְׂבָּע. וְגַם אֶת הַגּוֹי אֲשֶׁר יַעֲבֹדוּ, דָּן אָנֹכִי.
