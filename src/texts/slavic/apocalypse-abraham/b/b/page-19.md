וַיֹּאמֶר אֵלַי: הַבֵּט עַתָּה תַּחַת רַגְלֶיךָ אֶל הָרָקִיעַ, וְהָבֵן אֶת הַבְּרִיאָה הַמְצֻיֶּרֶת מֵאָז עַל הָרָקִיעַ הַזֶּה, וְאֶת הַבְּרוּאִים אֲשֶׁר בּוֹ, וְאֶת הָעוֹלָם הַמּוּכָן אַחֲרָיו. וָאַבִּיט תַּחַת רְקִיעַ רַגְלַי, וָאֵרֶא כְּדִמְיוֹן הַשָּׁמַיִם וַאֲשֶׁר עִמּוֹ, וְשָׁם הָאָרֶץ וּפְרִיָּהּ, וּתְנוּעָתָהּ, וְנַפְשָׁהּ, וְכֹחַ בְּנֵי אָדָם שֶׁבָּהּ, וְנַפְשׁוֹת הָרְשָׁעִים שֶׁבָּהּ, וְאֶת מַעֲשֵׂי צִדְקָהּ, וְאֶת תְּהוֹם וְעִנּוּיָהּ, וְאֶת תַּחְתִּיּוֹתֶיהָ אֲשֶׁר בָּהֶן הָאֲבַדּוֹן. וָאֵרֶא שָׁם אֶת הַיָּם וְאִיָּיו וּבְהֶמְתּוֹ וְדָגָיו, וְאֶת הַלִּוְיָתָן וּמֶמְשַׁלְתּוֹ וּמִשְׁכָּבוֹ וּמְאוּרוֹתָיו, וְאֶת הַתֵּבֵל הַשּׁוֹכֶבֶת עָלָיו, וּתְנוּעוֹתָיו, וְהֶהָרֵס מִפָּנָיו. וָאֵרֶא שָׁם נְהָרוֹת וּמְקוֹרוֹתָם וְסִבּוּבֵיהֶם. וָאֵרֶא שָׁם אֶת גַּן עֵדֶן וְאֶת פִּרְיוֹ, וְאֶת הַמַּעְיָן הַמּוֹצִיא נָהָר מִמֶּנּוּ, וְעֵצָיו וּפִרְחֵיהֶם, עֹשֵׂי הַצֶּדֶק, וָאֵרֶא בּוֹ מַאֲכָל וּמְנוּחָה. וָאֵרֶא שָׁם עַם רַב, אֲנָשִׁים וְנָשִׁים וִילָדִים, מַחֲצִיתָם מִצַּד הַתְּמוּנָה הַיְמָנִי וּמַחֲצִיתָם מִצַּד הַשְּׂמָאלִי. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מָה הַתְּמוּנָה הַזֹּאת שֶׁל הַבְּרִיאָה. וַיֹּאמֶר אֵלַי: זֹאת רְצוֹנִי בַּנִּמְצָא בְּמַחְשָׁבָה וְהָיָה טוֹב לְפָנַי; וְאַחַר כֵּן צִוִּיתִי, וְהָיוּ בִדְבָרִי כְּכֹל אֲשֶׁר קָבַעְתִּי לִהְיוֹת, אֲשֶׁר לְפָנֵי כֵן נִרְשְׁמוּ וְעָמְדוּ לְפָנַי הַבְּרוּאִים מֵאָז. וָאֹמַר: אָדוֹן גִּבּוֹר וְקַדְמוֹן, מִי הָעָם בַּתְּמוּנָה הַזֹּאת מִזֶּה וּמִזֶּה. וַיֹּאמֶר אֵלַי: אֵלֶּה אֲשֶׁר מִצַּד שְׂמֹאל, הֲמוֹן עַמִּים אֲשֶׁר הָיוּ מֵאָז, וּמוּכָנִים אַחֲרֶיךָ מֵהֶם לְמִשְׁפָּט וְתִקּוּן, וּמֵהֶם לְנָקָם וּלְאָבְדָן בְּקֵץ הָעוֹלָם; וַאֲשֶׁר מִצַּד יָמִין בַּתְּמוּנָה, אֵלֶּה הָעָם הַמֻּבְדָּל לִי, עַם מִתּוֹךְ הָעַמִּים עִם עֲזָאזֵל — אֵלֶּה אֲשֶׁר הֵכַנְתִּי לְהִוָּלֵד מִמְּךָ וּלְהִקָּרֵא עַמִּי.
