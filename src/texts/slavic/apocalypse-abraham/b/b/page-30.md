וְגַם אֶת הַגּוֹי אֲשֶׁר יַעֲבֹדוּ, דָּן אָנֹכִי.

---

אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: אֶת הָאֵל הַבּוֹרֵא אַתָּה מְבַקֵּשׁ בְּמַחְשֶׁבֶת לִבְּךָ — אֲנִי הוּא. צֵא מֵאֵת תֶּרַח אָבִיךָ וּמִן הַבַּיִת, פֶּן תּוּמַת גַּם אַתָּה בְּחַטֹּאות בֵּית אָבִיךָ. וָאֵצֵא, וַיְהִי כְּצֵאתִי וּבְטֶרֶם הִגַּעְתִּי אֶל פֶּתַח הֶחָצֵר, וְקוֹל רַעַם בָּא וַיִּשְׂרֹף אוֹתוֹ וְאֶת בֵּיתוֹ וְאֵת כָּל אֲשֶׁר בְּבֵיתוֹ עַד הָאָרֶץ, אַרְבָּעִים אַמָּה. אָז בָּא הַקּוֹל אֵלַי פַּעֲמַיִם: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: אֲנִי הוּא, אַל תִּירָא, כִּי אֲנִי לִפְנֵי הָעוֹלָמִים וְאֵל גִּבּוֹר, אֲשֶׁר בָּרָאתִי רִאשׁוֹנָה אֶת אוֹר הָעוֹלָם. וַאֲנִי מָגֵן לְךָ וְעֶזְרְךָ. לֵךְ קַח לִי עֶגְלָה מְשֻׁלֶּשֶׁת וְאַיִל מְשֻׁלָּשׁ וְעֵז מְשֻׁלֶּשֶׁת, וְתֹר וְגוֹזָל, וְהַקְרֵב לִי קָרְבָּן טָהוֹר, וּבַקָּרְבָּן הַהוּא אָקִים לְךָ עוֹלָמִים, וְאַגִּיד לְךָ אֶת הַשְּׁמוּרוֹת וּתְרָאֶה גְּדֹלוֹת אֲשֶׁר לֹא רָאִיתָ, כִּי אָהַבְתָּ לְבַקְּשֵׁנִי וָאֶקְרָא לְךָ אוֹהֲבִי. אַךְ הִשָּׁמְרוּ לְךָ מִכָּל מַאֲכָל הַבָּא מֵאֵשׁ, וּמִשְּׁתוֹת יַיִן, וּמִסּוּךְ שֶׁמֶן, אַרְבָּעִים יוֹם. אָז תַּקְרִיב לִי אֶת הַקָּרְבָּן אֲשֶׁר צִוִּיתִיךָ, בַּמָּקוֹם אֲשֶׁר אַרְאֶה אוֹתְךָ עַל הָהָר הָרָם, וְשָׁם אַרְאֶה לְךָ בִּדְבָרִי אֶת הָעוֹלָמִים, אֶת הַבְּרוּאִים וְאֶת הַהִתְחַדְּשׁוּת, וְאוֹדִיעֲךָ בָהֶם אֶת הַבָּאוֹת עַל עֹשֵׂי הָרַע וְעַל הַצַּדִּיקִים בְּדוֹר בְּנֵי אָדָם. וַיְהִי כְּשָׁמְעִי אֶת הַקּוֹל מְדַבֵּר אֵלַי דְּבָרִים כָּאֵלֶּה, וָאַבִּיט הֵנָּה וָהֵנָּה, וְהִנֵּה אֵין נִשְׁמַת אָדָם, וּכְבָר רוּחִי בָרְחָה, נַפְשִׁי מִמֶּנִּי, וָאֱהִי כְּאֶבֶן וָאֶפֹּל אַפַּיִם אַרְצָה, כִּי לֹא הָיָה עוֹד בִּי כֹּחַ לַעֲמֹד עַל הָאָרֶץ. וּבְעוֹדֶנִּי נֹפֵל עַל פָּנַי אַרְצָה, וָאֶשְׁמַע קוֹל אֹמֵר: לֵךְ יְהוֹאֵל, בְּתִוּוּךְ שְׁמִי הַמְפֹרָשׁ, קוּם אֶת הָאִישׁ הַזֶּה, וְחַזְּקֵהוּ מִן הָרְעָדָה. וַיָּבֹא הַמַּלְאָךְ אֲשֶׁר שֻׁלַּח אֵלַי בִּדְמוּת אִישׁ, וַיַּחֲזֵק בְּיָדִי הַיְמָנִית וַיַּעֲמִידֵנִי עַל רַגְלָי.

קַדְמוֹן וְגִבּוֹר, קָדוֹשׁ אֵל, יָחִיד מוֹשֵׁל, עַצְמִי, בְּלִי כִלָּיוֹן, בְּלִי טֻמְאָה, בְּלִי מוֹלֶדֶת, בְּלִי דֹפִי, בְּלִי טֻמְאָה עוֹד, שָׁלֵם מֵעַצְמְךָ, מֵאִיר מֵעַצְמְךָ, בְּלִי אֵם, בְּלִי אָב, בְּלִי תוֹלְדוֹת, רָם, אִישׁ אֵשׁ, יָשָׁר, אֹהֵב אָדָם, אוֹהֵב, נַדְבָן, נוֹתֵן מַתָּנוֹת, קַנַּאי, אֶרֶךְ אַפַּיִם, וְרַב חֶסֶד — גָּדוֹל, הוּא אֱלֹהַי הָעוֹלָם, גִּבּוֹר, קָדוֹשׁ, צְבָאוֹת, נִכְבָּד מְאֹד, יוֹאֵל. אַתָּה אֲשֶׁר אָהֲבָה נַפְשִׁי, שׁוֹמְרִי לְעוֹלָם, אֵשׁ, מַזְהִיר, מֵאִיר, בַּעַל קוֹל רַעַם, מַרְאֶה בָרָק, רַב עֵינַיִם, מְקַבֵּל תְּפִלַּת מְכַבְּדֶיךָ, וּמַסֵּב פָּנָיו מִתְּפִלַּת הָאוֹחֲזִים בְּךָ בְּעֹשֶׁק הַמַּכְעִיסִים אוֹתְךָ; מַתִּיר אֶת הַנִּמְצָאִים בְּלֹא-שֹׂבַע בֵּין הָרְשָׁעִים, לַמְעֹרְבָּבִים שֶׁל הַתֵּבֵל בָּעוֹלָם הַנִּפְסָד, מְחַדֵּשׁ אֶת עוֹלָם הַצֶּדֶק. אַתָּה מֵאִיר אוֹר לִפְנֵי הָאוֹר הַפְּנִימִי עַל כָּל הַבְּרִיאָה, וּמִפָּנֶיךָ לְהָאִיר יוֹם עַל הָאָרֶץ; וּבִשְׁמֵי מְעוֹנוֹתֶיךָ בְּלִי חֶסְרוֹן, אוֹר אַחֵר, זֹהַר אֲשֶׁר לֹא יְסֻפַּר, מֵאוֹר פָּנֶיךָ. קַבֵּל תְּפִלָּתִי וְהֵעָרֵב בָּהּ, כָּכָה גַם אֶת הַקָּרְבָּן אֲשֶׁר עָשִׂיתָ לְךָ עַל יְדֵי הַמְבַקֵּשׁ אוֹתְךָ. קַבְּלֵנִי הֵיטֵב, וְלַמְּדֵנִי וְהַרְאֵנִי, הוֹדַע לְעַבְדְּךָ כְּכֹל אֲשֶׁר הִבְטַחְתַּנִי.

וַיֹּאמֶר: וְכֵן קָרוֹב לְבָנֵי אָדָם בַּעֲבוּרְךָ, וְאַחֲרֶיךָ בַּעֲבוּר זֶרַע הָעָם הַמֻּבְדָּל, כִּי תִרְאֶה בַתְּמוּנָה אֶת הַכָּבֵד עֲלֵיהֶם. וַאֲנִי אַגִּיד לְךָ מַה יִּהְיֶה וְכַמָּה יִהְיֶה בְּאַחֲרִית הַיָּמִים. הַבֵּט עַתָּה בַּתְּמוּנָה. וָאַבִּיט וָאֵרֶא שָׁם אֵת אֲשֶׁר הָיָה לְפָנַי בַּבְּרִיאָה — וָאֵרֶא אֶת אָדָם וְאֶת חַוָּה אֲשֶׁר עִמּוֹ, וְעִמָּהֶם הָאוֹיֵב הֶעָרוּם, וְאֶת קַיִן פֹּעֵל עָוֶל בְּיַד הָאוֹיֵב, וְאֶת הֶבֶל הַנֶּהֱרָג, וְאֶת הַכִּלָּיוֹן הַמּוּבָא עַל הָרָשָׁע וְהַנָּתוּן לוֹ. וָאֵרֶא שָׁם אֶת הַזְּנוּת וְאֶת הַחוֹמְדִים אוֹתָהּ וְאֶת טֻמְאָתָהּ, וְאֶת קִנְאָתָם וְאֵשׁ כִּלְיוֹנָם בְּתַחְתִּיּוֹת הָאָרֶץ. וָאֵרֶא שָׁם אֶת הַגְּנֵבָה וְאֶת הַנֶּחְפָּזִים אֵלֶיהָ וְאֶת מְזִמּוֹתָהּ וּגְמוּלָהּ — מִשְׁפָּט גָּדוֹל.

וָאֵרֶא שָׁם אֶת דְּמוּת פֶּסֶל הַקִּנְאָה, כְּמַעֲשֵׂה חָרָשׁ עֵצִים כַּאֲשֶׁר עָשָׂה אָבִי, גּוּפוֹ כִּנְחֹשֶׁת, וּלְפָנָיו אִישׁ מִשְׁתַּחֲוֶה לוֹ, וּמִזְבֵּחַ נֶגְדּוֹ, וּנְעָרִים נִשְׁחָטִים עָלָיו לִפְנֵי פְנֵי הַפֶּסֶל. וָאֹמַר: מַה זֶּה הַפֶּסֶל, אוֹ מַה הַמִּזְבֵּחַ, אוֹ מִי הַנִּזְבָּחִים, אוֹ מִי הַזֹּבֵחַ, אוֹ מַה זֶּה הַהֵיכָל אֲשֶׁר אֲנִי רוֹאֶה, נָאֶה בְּחָכְמַת מַעֲשֵׂהוּ וּבְיָפְיוֹ, מִתַּחַת לִדְרִיסַת כְּבוֹדְךָ. וַיֹּאמֶר: שְׁמַע אַבְרָהָם, זֶה אֲשֶׁר רָאִיתָ — הַהֵיכָל וְהַמִּזְבֵּחַ — נָאֶה הוּא לִי, מַחְשֶׁבֶת הַכְּהֻנָּה לְשֵׁם כְּבוֹדִי, אֲשֶׁר בָּהּ יִשְׁכֹּן כָּל קְהַל אֲנָשִׁים, וּמְבוֹא מְלָכִים וּנְבִיאִים, וְכָל קָרְבָּן אֲשֶׁר אָשִׂים לְהֵעָשׂוֹת לִי בְּעַמִּי הַבָּא וּבְזַרְעֲךָ. וְהַגּוּף אֲשֶׁר רָאִיתָ הוּא כַּעֲסִי, אֲשֶׁר בּוֹ יַכְעִיסוּנִי הָעָם הַבָּא מִמְּךָ אֵלָי; וְהָאִישׁ אֲשֶׁר רָאִיתָ נִשְׁחָט, הוּא קָרְבַּן הַמַּכְעִיס — רֶצַח — אֲשֶׁר יִהְיוּ לִי לְעֵדוּת מִשְׁפָּט בְּקֵץ הַתַּכְלִית, לְרָעַת הַבְּרִיאָה.

וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, הָבֵן אֶת אֲשֶׁר אַגִּיד לְךָ, וַעֲנֵנִי כְּכֹל אֲשֶׁר אֶשְׁאָלְךָ: תֶּרַח אָבִיךָ, מַדּוּעַ לֹא שָׁמַע בְּקוֹלְךָ וְלֹא סָר מִשֵּׁדֵי הַפְּסִילִים, עַד אֲשֶׁר אָבַד כָּל בֵּיתוֹ עִמּוֹ. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, אָכֵן לֹא הָיָה בִרְצוֹנוֹ לִשְׁמֹעַ לִי, וְגַם אֲנִי לֹא הָלַכְתִּי אַחֲרֵי מַעֲשָׂיו. וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, כַּאֲשֶׁר חֵלֶק אָבִיךָ בּוֹ, כֵּן חֶלְקְךָ בְךָ, כֵּן גַּם חֵלֶק רְצוֹנִי מוּכָן בִּי, לַיָּמִים הַבָּאִים לִפְנֵיהֶם; לֹא תֵדַע, וְלֹא תִרְאֶה בְעֵינֶיךָ אֶת הָעֲתִידוֹת, כִּי שֶׁל זַרְעֲךָ הֵם. הַבֵּט בַּתְּמוּנָה. וָאַבִּיט וָאֵרֶא, וְהִנֵּה רָעֲדָה הַתְּמוּנָה, וַיָּרָץ מִצִּדָּהּ הַשְּׂמָאלִי עַם נָכְרִי, וַיִּבְזְזוּ אֶת אֲשֶׁר מִצַּד יָמִין, אֲנָשִׁים וְנָשִׁים וִילָדִים; מִקְצָתָם שָׁדְדוּ, וּמִקְצָתָם הֶחֱזִיקוּ לָהֶם. הִנֵּה רָאִיתִי בָּאִים אֲלֵיהֶם אַרְבָּעָה מוֹרָדוֹת, וַיַּצִּיתוּ אֶת הַהֵיכָל בָּאֵשׁ וַיָּבֹזּוּ אֶת הַקֳּדָשִׁים אֲשֶׁר בּוֹ. וָאֹמַר: קַדְמוֹן, אֵלֶּה אֲשֶׁר מִמֶּנִּי לָקַחְתָּ עָם, בּוֹזְזִים אוֹתָם הֲמוֹן עַמִּים — מִקְצָתָם הוֹרְגִים, מִקְצָתָם מַחֲזִיקִים כְּגֵרִים, וְאֶת הַהֵיכָל שׂוֹרְפִים בָּאֵשׁ וְאֶת הַמַּעֲשִׂים הַיָּפִים בּוֹ בּוֹזְזִים וְהוֹרְסִים. קַדְמוֹן וְגִבּוֹר, אִם כֵּן הוּא, לָמָּה מֵעַתָּה פָּרַח לִבִּי, וְלָמָּה כֵן יִהְיֶה עַתָּה.

וְגַם אֶת הַגּוֹי אֲשֶׁר יַעֲבֹדוּ, דָּן אָנֹכִי.
