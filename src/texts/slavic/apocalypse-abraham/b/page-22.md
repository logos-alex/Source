וַיַּעַן וַיֹּאמֶר: שְׁמַע אַבְרָהָם, הָאִישׁ אֲשֶׁר רָאִיתָ נִכְלָם וּמֻכֶּה וְשׁוּב מִשְׁתַּחֲוֶה לוֹ, זֶה מְנוּחַת הַגּוֹיִם, מִזַּרְעֲךָ יִהְיֶה לְעַם בְּאַחֲרִית הַיָּמִים, בַּשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל הָעוֹלָם הַזֶּה בְּגוּפוֹ. 

וּבַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה שֶׁל קֵץ עוֹלָמִי אָקִים אֶת הָאִישׁ הַזֶּה מִזֶּרַע עַמִּי — אֲשֶׁר רָאִיתָ — לוֹ יִדַּמּוּ כֻלָּם, וּבִמְשָׁלִים כְּקוֹרְאִים מִמֶּנִּי, עוֹבְרִים בְּאוֹרוֹתָם. 

וַאֲשֶׁר רָאִיתָ מִצַּד שְׂמֹאל בַּתְּמוּנָה יוֹצְאִים וּמִשְׁתַּחֲוִים לוֹ, מֵהַגּוֹיִם רַבִּים יִבְטְחוּ בּוֹ. 

וַאֲשֶׁר רָאִיתָ מִזַּרְעֲךָ מִצַּד יָמִין, מִקְצָתָם בּוֹשִׁים וּמַכִּים וּמִקְצָתָם מִשְׁתַּחֲוִים לוֹ — רַבִּים מֵהֶם יִכָּשְׁלוּ בּוֹ. 

וְהוּא יְנַסֶּה אֶת הַמִּשְׁתַּחֲוִים לוֹ מִזַּרְעֲךָ, בְּקֵץ הַשָּׁעָה הַשְּׁתֵּים עֶשְׂרֵה, בְּקֵץ הָעוֹלָם הַבָּלוּי. 

בְּטֶרֶם יָחֵל לִצְמֹחַ עוֹלָם הַצֶּדֶק, יָבוֹא מִשְׁפָּטִי עַל הָעַמִּים הָעַרְמוּמִים, עַל זֶרַע עַמְּךָ הַמֻּבְדָּל לִי.

בַּיָּמִים הָהֵם אָבִיא עַל כָּל הַתֵּבֵל עֶשֶׂר מַכּוֹת בְּרָעָה וּמַחֲלָה, וְאֶת אֲנָקַת רוּחָם הַמָּרָה אָבִיא עַל הַדּוֹרוֹת אֲשֶׁר עַל הָאָרֶץ, בַּעֲבוּר קֶצֶף וְהַשְׁחָתַת בְּרוּאֵיהֶם אֲשֶׁר יַכְעִיסוּנִי בָהֶם.
