וַיֹּאמֶר אֵלַי: שְׁמַע אַבְרָהָם, בַּעֲבוּר זַרְעֲךָ הַמַּכְעִיסִים אוֹתִי, יִהְיוּ הַגּוּפוֹת אֲשֶׁר רָאִיתָ וְהַכְּלֵי-הַזֶּמֶר [?] בַּתְּמוּנָה בַּהֵיכָל בְּקִנְאָה, וְכָל אֲשֶׁר רָאִיתָ יִהְיֶה. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יַעַבְרוּ נָא עַתָּה הָרָעוֹת הַחֲמוּרוֹת וְהַמַּעֲשִׂים, פְּקֻדַּת עֹשֵׂי הָרֶשַׁע עַל מַעֲשֵׂי הַצֶּדֶק שֶׁלּוֹ — תוּכַל בֶּאֱמֶת לַעֲשׂוֹת זֹאת. וַיֹּאמֶר אֵלַי: עֵת הַצֶּדֶק תִּפְגַּע, לִפְנֵי דֻגְמַת הַמְּלָכִים הַשּׁוֹפְטִים בְּצֶדֶק, אֲשֶׁר בָּרָאתִי מֵאָז לִמְשֹׁל מִתּוֹכָם, בָּהֶם; וּמֵהֶם יֵצְאוּ אֲנָשִׁים אֲשֶׁר יְמַהֲרוּ אֶת אֲשֶׁר הִגַּדְתִּי לְךָ וְרָאִיתָ. וָאַעַן וָאֹמַר: גִּבּוֹר, שֶׁתְּהִלָּתְךָ בְּכֹחוֹ, תְּהִי תְפִלָּה בְּבַקָּשָׁתִי, כִּי עַל כֵּן הוֹדַעְתָּ לִי וְהֶרְאֵיתָ אוֹתִי כִּי הֶעֱלֵיתָ אוֹתִי אֶל רוּמְךָ, וְלָכֵן הוֹדַעְתָּ לְיְדִידְךָ הַזֶּה כְּכֹל אֲשֶׁר שָׁאַלְתִּי — הֲלְאֹרֶךְ יָמִים יִהְיֶה כְּכֹל אֲשֶׁר רָאִיתִי.
