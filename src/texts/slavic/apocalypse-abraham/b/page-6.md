וּבָרִיסָת, אֲשֶׁר מֵעֵץ עָשׂוּי וּמִכֶּסֶף מְצֻפֶּה, [1] הֲלֹא גַם הוּא לְחִדּוּשׁ הָיָה עוֹמֵד, וְהוּא — בְּטֶרֶם נַעֲשָׂה — נֶעְקַר מִן הָאָרֶץ, גָּדוֹל וְעָצוּם, עִם פְּרָחִים וּפְרִיּוֹת גְּדוּעִים וְעִם חֹמֶר, וּבְחָכְמָתְךָ נַעֲשָׂה אֱלוֹהַּ; וְהִנֵּה כְּבָר יָבַשׁ וְאָבַד: מִן הַגֹּבַהּ נָפַל אַרְצָה, וּמִגְּדֻלָּה בָּא לִקְטַנָּה, וּמַרְאֵה פָנָיו נָמוֹג; וְהוּא נִשְׂרַף בָּאֵשׁ וַיְהִי לְאֵפֶר וְאֵינֶנּוּ עוֹד — וְאַתָּה אוֹמֵר הַיּוֹם אֶעֱשֶׂה אַחֵר וּמָחָר יַעֲשֶׂה אָכְלִי.

> **הערת המהדורה:** הושלם כאן קטע שנשמט מן התרגום הקודם — ״וְהוּא בְּטֶרֶם נַעֲשָׂה נֶעְקַר מִן הָאָרֶץ... וּמַרְאֵה פָנָיו נָמוֹג״ («и еще соушоу емоу прѣже здѣдлнна... и ѥзорћ лицѧ є. во истакловени», עמ' 58 בדפוס) — תיאור מלא של עשיית בריסת: עקירתו מן הארץ עם פרחיו ופריותיו הגדועים, עשייתו אלוה בחכמת אדם, ונפילתו מגובה לקטנה. הקטע קיים גם בנוסח ג' (פורפירייב, עמ' 115) והוא לשון הפליאה; ראו נספח, הערה 10.

הִנְנִי אֹמֵר: הָאֵשׁ נִכְבֶּדֶת יוֹתֵר מִן הַצְּלָמִים הָעֲשׂוּיִים, כִּי גַּם הָעֹשֶׂה אוֹתָהּ נִכְנָע לָהּ, וּמִתְקַלְּסִים בָּהּ הַנּוֹחִים לְאָבְדָם בִּבְעֵרָתָם.

אַךְ נִכְבָּדִים מִמֶּנָּה הַמַּיִם, כִּי גֹּבְרִים עַל הָאֵשׁ וּמַרְוִים אֶת הָאָרֶץ; אַךְ גַּם אוֹתָם לֹא אֶקְרָא אֵל, כִּי הֵם נִכְנָעִים תַּחַת הָאָרֶץ. 

אַךְ נִכְבֶּדֶת מֵהֶם הָאָרֶץ, כִּי גֹּבֶרֶת עַל טֶבַע הַמַּיִם וְרֹב מַיִם — אַךְ גַּם אוֹתָהּ לֹא אֶקְרָא, כִּי הַשֶּׁמֶשׁ מְיַבְּשָׁהּ, וְהִיא נַעֲשֵׂית לְמַעֲשֵׂה אָדָם. 

נִכְבֶּדֶת מִן הָאָרֶץ הַשֶּׁמֶשׁ אֶקְרָא, כִּי בְקַרְנֶיהָ מְאִירָה אֶת כָּל הַתֵּבֵל; אַךְ גַּם אוֹתָהּ לֹא אֶקְרָא, כִּי בְּבוֹא הַלַּיְלָה מַחְשִׁיךְ אוֹתָהּ הַחֹשֶׁךְ. 

וְלֹא אֶת הַיָּרֵחַ וְאֶת הַכּוֹכָבִים אֶקְרָא אֵל, [2] כִּי גַם הֵם בְּעִתָּם בַּלַּיְלָה יַחְשִׁיכוּ אוֹרָם. 

אֲבָל שְׁמַע נָא תֶּרַח אָבִי, וְאַגִּיד לְךָ אֶת הָאֵל אֲשֶׁר עָשָׂה הַכֹּל, [3] וְהוּא הָאֱמֶת: אֲשֶׁר יִצֹּק אֶת הַשָּׁמַיִם, וְהִזְהִיב אֶת הַשֶּׁמֶשׁ וְהִבְהִיר אֶת הַיָּרֵחַ וְעִמּוֹ הַכּוֹכָבִים, אֲשֶׁר הוֹבִישׁ אֶת הָאָרֶץ בְּתוֹךְ מַיִם רַבִּים, וְאוֹתְךָ בְעַצְמְךָ שָׂם בַּחֹמֶר — וְאוֹתוֹ אֲבַקֵּשׁ עַתָּה, בְּמַחְשָׁבָה נָבוֹכָה, אִם יִגָּלֶה לָנוּ הוּא בְּעַצְמוֹ.

וַיְהִי כְּדַבֵּר אַבְרָם כַּדְּבָרִים הָאֵלֶּה אֶל תֶּרַח אָבִיו בַּחֲצַר בֵּיתוֹ, וְ

> **הערה:** הפרק מסתיים באמצע משפט — כפי שנקטע עמ' 59 בדפוס טיכונרבוב; המשך הדברים פותח את פרק ז' (עמ' 60).
