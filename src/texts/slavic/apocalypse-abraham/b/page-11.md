קַדְמוֹן וְגִבּוֹר, קָדוֹשׁ אֵל, יָחִיד מוֹשֵׁל, עַצְמִי, בְּלִי כִלָּיוֹן, בְּלִי טֻמְאָה, בְּלִי מוֹלֶדֶת, בְּלִי דֹפִי, בְּלִי טֻמְאָה עוֹד, שָׁלֵם מֵעַצְמְךָ, מֵאִיר מֵעַצְמְךָ, בְּלִי אֵם, בְּלִי אָב, בְּלִי תוֹלְדוֹת, רָם, אִישׁ אֵשׁ, יָשָׁר, אֹהֵב אָדָם, אוֹהֵב, נַדְבָן, נוֹתֵן מַתָּנוֹת, קַנַּאי, אֶרֶךְ אַפַּיִם, וְרַב חֶסֶד — גָּדוֹל, הוּא אֱלֹהַי הָעוֹלָם, גִּבּוֹר, קָדוֹשׁ, צְבָאוֹת, נִכְבָּד מְאֹד, יוֹאֵל. אַתָּה אֲשֶׁר אָהֲבָה נַפְשִׁי, שׁוֹמְרִי לְעוֹלָם, אֵשׁ, מַזְהִיר, מֵאִיר, בַּעַל קוֹל רַעַם, מַרְאֶה בָרָק, רַב עֵינַיִם, מְקַבֵּל תְּפִלַּת מְכַבְּדֶיךָ, וּמַסֵּב פָּנָיו מִתְּפִלַּת הָאוֹחֲזִים בְּךָ בְּעֹשֶׁק הַמַּכְעִיסִים אוֹתְךָ; מַתִּיר אֶת הַנִּמְצָאִים בְּלֹא-שֹׂבַע בֵּין הָרְשָׁעִים, לַמְעֹרְבָּבִים שֶׁל הַתֵּבֵל בָּעוֹלָם הַנִּפְסָד, מְחַדֵּשׁ אֶת עוֹלָם הַצֶּדֶק. אַתָּה מֵאִיר אוֹר לִפְנֵי הָאוֹר הַפְּנִימִי עַל כָּל הַבְּרִיאָה, וּמִפָּנֶיךָ לְהָאִיר יוֹם עַל הָאָרֶץ; וּבִשְׁמֵי מְעוֹנוֹתֶיךָ בְּלִי חֶסְרוֹן, אוֹר אַחֵר, זֹהַר אֲשֶׁר לֹא יְסֻפַּר, מֵאוֹר פָּנֶיךָ. קַבֵּל תְּפִלָּתִי וְהֵעָרֵב בָּהּ, כָּכָה גַם אֶת הַקָּרְבָּן אֲשֶׁר עָשִׂיתָ לְךָ עַל יְדֵי הַמְבַקֵּשׁ אוֹתְךָ. קַבְּלֵנִי הֵיטֵב, וְלַמְּדֵנִי וְהַרְאֵנִי, הוֹדַע לְעַבְדְּךָ כְּכֹל אֲשֶׁר הִבְטַחְתַּנִי.

וּבְעוֹד הוּא אוֹמֵר אֶת הַשִּׁיר, וּפִי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ עָלָה לְמַעְלָה, וָאֶשְׁמַע קוֹל כִּמְבוּכַת הַיָּם, וְלֹא חָדַל מִבְּעֵרַת הָאֵשׁ הָרַבָּה. וְכַאֲשֶׁר עָלְתָה הָאֵשׁ לְמַעְלָה, וָאֵרֶא תַחַת הָאֵשׁ כִּסֵּא, וּמִן הָאֵשׁ רַבֵּי הָעֵינַיִם סָבִיב, אֹמְרִים שִׁיר תַּחַת הַכִּסֵּא — אַרְבַּע חַיּוֹת אֵשׁ, מַרְאֶה וְקוֹל אֶחָד, כָּל אַחַת אַרְבַּעַת פָּנִים. מַרְאֵה פְנֵיהֶן: אַרְיֵה, שׁוֹר, אָדָם, נֶשֶׁר — אַרְבָּעָה רָאשִׁים עַל גּוּף אֶחָד, כֵּן פְּנֵי הָאַרְבַּע חַיּוֹת. וּלְכָל אַחַת שֵׁשׁ כְּנָפַיִם: מִכְּתֵפֶיהָ, וּמֵחֲצִיָּהּ, וּמִמָּתְנֶיהָ. וּבִכְנָפַיִם אֲשֶׁר מִכְּתֵפֵיהֶן יְכַסּוּ פְּנֵיהֶן, וּבִכְנָפַיִם אֲשֶׁר מִמָּתְנֵיהֶן יְכַסּוּ אֶת רַגְלֵיהֶן, וּבִכְנָפַיִם הָאֶמְצָעִיּוֹת יִפְרְשׂוּ וְעָפוּ יָשָׁר. וַיְהִי כְּכַלּוֹתָם אֶת הַשִּׁיר וַיַּבִּיטוּ אִישׁ אֶל רֵעֵהוּ וַיְאַיְּמוּ אִישׁ בְּרֵעֵהוּ. וַיְהִי כִּרְאוֹת הַמַּלְאָךְ אֲשֶׁר עִמִּי כִּי יְאַיְּמוּ, וַיַּעַזְבֵנִי וַיָּרָץ אֲלֵיהֶם, וַיַּסֵּב אֶת פְּנֵי כָּל חַיָּה מִנֶּגֶד פְּנֵי רֵעוּתָהּ, לְבִלְתִּי רְאוֹת אִישׁ אֶת פְּנֵי רֵעֵהוּ מְאַיְּמוֹת, וַיְלַמְּדֵם שִׁיר הַשָּׁלוֹם הַקַּדְמוֹנִי, וַאֲנִי לְבַדִּי עוֹמֵד וּמַבִּיט. וָאֵרֶא אַחֲרֵי הַחַיּוֹת מֶרְכָּבָה, גַּלְגַּלֵּי אֵשׁ, וְכָל גַּלְגַּל מָלֵא עֵינַיִם סָבִיב, וּמֵעַל הַגַּלְגַּלִּים הַכִּסֵּא אֲשֶׁר רָאִיתִי, מְכֻסֶּה אֵשׁ, וְאֵשׁ סוֹבֵב אוֹתוֹ סָבִיב, וְאוֹר אֲשֶׁר לֹא יְסֻפַּר סוֹבֵב אֶת מַחֲנֵה הָאֵשׁ. וָאֶשְׁמַע אֶת קוֹל קְדֻשָּׁתָם כְּקוֹל אִישׁ אֶחָד, וַיָּבֹא אֵלַי קוֹל מִתּוֹךְ הָאֵשׁ, אֹמֵר: אַבְרָהָם אַבְרָהָם, וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: הַבֵּט אֶל הָרְקִיעִים אֲשֶׁר תַּחַת הָרָקִיעַ אֲשֶׁר עָלָיו אַתָּה מֻצָּב, וּרְאֵה כִּי בְּשׁוּם רָקִיעַ אֵין אַחֵר זוּלָתִי אֲשֶׁר בִּקַּשְׁתָּ וְאֲשֶׁר אֲהֵבְךָ.

וּבְעוֹד הוּא מְדַבֵּר, וְהִנֵּה נִפְתְּחוּ הָרְקִיעִים מִתַּחְתַּי, וָאֵרֶא עַל הָרָקִיעַ הַשְּׁבִיעִי אֲשֶׁר עָמַדְתִּי עָלָיו אֵשׁ פְּרוּסָה, וְאוֹר וָטַל, וַהֲמוֹן מַלְאָכִים, וּגְבוּרַת כָּבוֹד בִּלְתִּי נִרְאֶה מִלְּמַעְלָה — מִלְּבַד הַחַיּוֹת אֲשֶׁר רָאִיתִי, לֹא רָאִיתִי שָׁם אַחֵר. וָאַבִּיט מִגֹּבַהּ עֳמָדִי אֶל הָרָקִיעַ הַשִּׁשִּׁי, וָאֵרֶא שָׁם הֲמוֹן מַלְאָכִים רוּחָנִיִּים בְּלִי גוּף, עֹשִׂים אֶת מִצְוַת מַלְאֲכֵי הָאֵשׁ אֲשֶׁר עַל הָרָקִיעַ הַשְּׁמִינִי, כְּפִי אֲשֶׁר הָיִיתִי עוֹמֵד עַל הַתְּלוּיִים בּוֹ. וְלֹא הָיָה בָּרָקִיעַ הַהוּא כֹּחַ בְּשׁוּם צוּרָה אַחֶרֶת, כִּי אִם מִשְׁפַּחַת הַמַּלְאָכִים הָרוּחָנִיִּים; וַאֲשֶׁר עַל הָרָקִיעַ הַשְּׁבִיעִי — הַכֹּחַ אֲשֶׁר רָאִיתִי — צֻוָּה לָנוּעַ. וָאֵרֶא שָׁם, בָּרָקִיעַ הַחֲמִישִׁי, אֶת כֹּחוֹת הַכּוֹכָבִים מְקַיְּמִים אֶת הַמִּצְוֹת הַמֻּטָּלוֹת עֲלֵיהֶם, וִיסוֹדוֹת הָאָרֶץ שׁוֹמְעִים לָהֶם. וַיֹּאמֶר אֵלַי הַקַּדְמוֹן הַגִּבּוֹר: אַבְרָהָם, אַבְרָהָם. וָאֹמַר הִנֵּנִי. וַיֹּאמֶר: הַבֵּט מִלְּמַעְלָה, וְהִנֵּה תַּחְתֶּיךָ הַכּוֹכָבִים, סְפֹר אוֹתָם וְהַגֵּד לִי מִסְפָּרָם. וָאֹמַר: מָתַי אוּכַל, וַאֲנִי אָדָם — עָפָר וָאֵפֶר. וַיֹּאמֶר לִי: כְּמִסְפַּר הַכּוֹכָבִים וְכֹחָם אֶתֵּן לְזַרְעֲךָ, גּוֹיִם וְעַמִּים מֻבְדָּלִים לִי בְגוֹרָלִי, עִם עֲזָאזֵל. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, יְדַבֵּר נָא עַבְדְּךָ לְפָנֶיךָ, וְאַל יִחַר אַפְּךָ עַל בְּחִירְךָ: הִנֵּה בְּטֶרֶם הֲרִימוֹתַנִי, חֵרְפַנִי עֲזָאזֵל; וְעַתָּה, כְּשֶׁאֵינֶנּוּ לְפָנֶיךָ, תְּכוֹנֵן עִמּוֹ.

וַיֹּאמֶר אֵלַי: הַבֵּט עַתָּה תַּחַת רַגְלֶיךָ אֶל הָרָקִיעַ, וְהָבֵן אֶת הַבְּרִיאָה הַמְצֻיֶּרֶת מֵאָז עַל הָרָקִיעַ הַזֶּה, וְאֶת הַבְּרוּאִים אֲשֶׁר בּוֹ, וְאֶת הָעוֹלָם הַמּוּכָן אַחֲרָיו. וָאַבִּיט תַּחַת רְקִיעַ רַגְלַי, וָאֵרֶא כְּדִמְיוֹן הַשָּׁמַיִם וַאֲשֶׁר עִמּוֹ, וְשָׁם הָאָרֶץ וּפְרִיָּהּ, וּתְנוּעָתָהּ, וְנַפְשָׁהּ, וְכֹחַ בְּנֵי אָדָם שֶׁבָּהּ, וְנַפְשׁוֹת הָרְשָׁעִים שֶׁבָּהּ, וְאֶת מַעֲשֵׂי צִדְקָהּ, וְאֶת תְּהוֹם וְעִנּוּיָהּ, וְאֶת תַּחְתִּיּוֹתֶיהָ אֲשֶׁר בָּהֶן הָאֲבַדּוֹן. וָאֵרֶא שָׁם אֶת הַיָּם וְאִיָּיו וּבְהֶמְתּוֹ וְדָגָיו, וְאֶת הַלִּוְיָתָן וּמֶמְשַׁלְתּוֹ וּמִשְׁכָּבוֹ וּמְאוּרוֹתָיו, וְאֶת הַתֵּבֵל הַשּׁוֹכֶבֶת עָלָיו, וּתְנוּעוֹתָיו, וְהֶהָרֵס מִפָּנָיו. וָאֵרֶא שָׁם נְהָרוֹת וּמְקוֹרוֹתָם וְסִבּוּבֵיהֶם. וָאֵרֶא שָׁם אֶת גַּן עֵדֶן וְאֶת פִּרְיוֹ, וְאֶת הַמַּעְיָן הַמּוֹצִיא נָהָר מִמֶּנּוּ, וְעֵצָיו וּפִרְחֵיהֶם, עֹשֵׂי הַצֶּדֶק, וָאֵרֶא בּוֹ מַאֲכָל וּמְנוּחָה. וָאֵרֶא שָׁם עַם רַב, אֲנָשִׁים וְנָשִׁים וִילָדִים, מַחֲצִיתָם מִצַּד הַתְּמוּנָה הַיְמָנִי וּמַחֲצִיתָם מִצַּד הַשְּׂמָאלִי. וָאֹמַר: קַדְמוֹן וְגִבּוֹר, מָה הַתְּמוּנָה הַזֹּאת שֶׁל הַבְּרִיאָה. וַיֹּאמֶר אֵלַי: זֹאת רְצוֹנִי בַּנִּמְצָא בְּמַחְשָׁבָה וְהָיָה טוֹב לְפָנַי; וְאַחַר כֵּן צִוִּיתִי, וְהָיוּ בִדְבָרִי כְּכֹל אֲשֶׁר קָבַעְתִּי לִהְיוֹת, אֲשֶׁר לְפָנֵי כֵן נִרְשְׁמוּ וְעָמְדוּ לְפָנַי הַבְּרוּאִים מֵאָז. וָאֹמַר: אָדוֹן גִּבּוֹר וְקַדְמוֹן, מִי הָעָם בַּתְּמוּנָה הַזֹּאת מִזֶּה וּמִזֶּה. וַיֹּאמֶר אֵלַי: אֵלֶּה אֲשֶׁר מִצַּד שְׂמֹאל, הֲמוֹן עַמִּים אֲשֶׁר הָיוּ מֵאָז, וּמוּכָנִים אַחֲרֶיךָ מֵהֶם לְמִשְׁפָּט וְתִקּוּן, וּמֵהֶם לְנָקָם וּלְאָבְדָן בְּקֵץ הָעוֹלָם; וַאֲשֶׁר מִצַּד יָמִין בַּתְּמוּנָה, אֵלֶּה הָעָם הַמֻּבְדָּל לִי, עַם מִתּוֹךְ הָעַמִּים עִם עֲזָאזֵל — אֵלֶּה אֲשֶׁר הֵכַנְתִּי לְהִוָּלֵד מִמְּךָ וּלְהִקָּרֵא עַמִּי.
