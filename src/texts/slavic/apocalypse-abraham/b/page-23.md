וְאָז יִוָּתְרוּ מִזַּרְעֲךָ אֲנָשִׁים צַדִּיקִים, בְּמִסְפָּר שְׁמוּרִים לִי, מְמַהֲרִים בִּכְבוֹד שְׁמִי אֶל הַמָּקוֹם הַמּוּכָן לָהֶם מֵאָז; וַאֲשֶׁר רָאִיתָ אֶת הֶחָרָב בַּתְּמוּנָה — יִתְקַיַּם בִּזְבָחִים וּבְמַתָּנוֹת שֶׁל צֶדֶק אֱמֶת בָּעוֹלָם הַצַּדִּיק; וְאֵת מַשְׁחִיתֵיהֶם יַשְׁחִיתוּ וְאֵת מְחָרְפֵיהֶם יַחְרְפוּ, וְהָיוּ רֹקְקִים בְּפָנֵי הַמַּשְׁחִיתִים, בִּרְאוֹתָם אוֹתִי שָׂמֵחַ עִם עַמִּי וּמְקַבֵּל אֶת הַשָׁבִים אֵלַי בִּתְשׁוּבָה. 

רְאֵה אַבְרָהָם אֶת אֲשֶׁר רָאִיתָ וְאֶת אֲשֶׁר שָׁמַעְתָּ וְדַע — כִּי כָל אֵלֶּה גּוֹרָלִי, וְהִנְנִי עִמְּךָ עַד עוֹלָם. 

וּבְעוֹדֶנּוּ מְדַבֵּר, וָאֶמָּצֵא עַל הָאָרֶץ, וָאֹמַר: קַדְמוֹן וְגִבּוֹר, אֵינֶנִּי עוֹד בַּכָּבוֹד אֲשֶׁר הָיִיתִי בּוֹ לְמַעְלָה, וְאָשֶׁר שָׁאַלְתִּי בְּלִבִּי — כָּרַתְהִי נַפְשִׁי; לֹא הֵבַנְתִּי אֶת אֲשֶׁר עִמִּי. 

וַיֹּאמֶר אֵלַי: אֶת מַאֲוַיֵּי לִבְּךָ אַגִּיד לְךָ — הִגַּדְתָּ לָדַעַת אֶת הַנֶּגַע הָעֲשִׂירִי אֲשֶׁר עַל הַגּוֹיִם, אֲשֶׁר הֲכִינוֹתִי וְהִזְמַנְתִּי בְּמִנְיָן שְׁתֵּים עֶשְׂרֵה שָׁעוֹת עַל הָאָרֶץ. 

שְׁמַע אֶת אֲשֶׁר אַגִּיד לְךָ, יִהְיֶה כֵן: הָרִאשׁוֹנָה — צָרָה וּמִדֹּחַק רָב. 

הַשְּׁנִיָּה — שְׂרֵפוֹת רַבּוֹת אֵשׁ בָּעָרִים. 

הַשְּׁלִישִׁית — חֵמָה רַבָּה וְדֶבֶר בַּבְּהֵמָה. 

הָרְבִיעִית — רָעָב בְּתוֹשְׁבֵי הָעוֹלָם. 

הַחֲמִישִׁית — בַּמּוֹשְׁלִים בְּרַעַשׁ וּבְחֶרֶב וָחֵמָה. 

הַשִּׁשִּׁית — בָּרָד וְרֹב שֶׁלֶג. 

הַשְּׁבִיעִית — חַיַּת שָׂדֶה תִּהְיֶה קִבְרָם. 

הַשְּׁמִינִית — רָעָב וָדֶבֶר יַחֲלִיפוּ כִלָּיוֹן. 

הַתְּשִׁיעִית — עֹנֶשׁ חֶרֶב וְהֶרֶג בְּצָרוֹת. 

הָעֲשִׂירִית — רַעַם וְקוֹלוֹת וּרְעָשִׁים לְהַשְׁחָתָה. 

אָז אֶתְקַע בַּשּׁוֹפָר מִן הָאֲוִיר וְאֶשְׁלַח אֶת בְּחִירִי, שֶׁבּוֹ כָּל כֹּחִי מִדָּה אַחַת, וְהוּא יִקְרָא לְעַמִּי הַנֶּעֱלָבִים מִן הַגּוֹיִם, מִגּוֹי אֶל גּוֹי. 

וַאֲנִי אֶשְׂרֹף בָּאֵשׁ אֶת הַמְחָרְפִים אוֹתָם וְאֶת הַשּׁוֹלְטִים בָּהֶם לְעוֹלָם, וְאֶתֵּן אֶת הַמְכַסִּים אוֹתָם לְחֶרְפָּה וּלְבִזָּיוֹן שֶׁל הָעוֹלָם הַזֶּה, כִּי הֲכִינוֹתִי לָהֶם מַאֲכַל אֵשׁ שְׁאוֹל, מְעוֹף בְּלִי הֶפְסֵק בָּאֲוִיר, תַּחַת הָאָרֶץ, בַּתַּחְתִּיּוֹת, וּמְלֵאֵי בֶטֶן תּוֹלַעַת.
