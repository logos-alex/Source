וָאֵרֶא גַם אֲנָשִׁים עֲרֻמִּים, מִצְחָם זֶה מוּל זֶה, וְעֶרְוָתָם, וְנֵזֶק אִישׁ בְּרֵעֵהוּ וּגְמוּלָם. וָאֵרֶא שָׁם אֶת הַתַּאֲוָה, וּבְיָדָהּ רֹאשׁ כָּל עַוְלָה, וְאֶת חִלּוּלָהּ וּפִזּוּרָהּ, נְתוּנָה לְאָבְדָּן. וָאֵרֶא שָׁם אֶת דְּמוּת פֶּסֶל הַקִּנְאָה, כְּמַעֲשֵׂה חָרָשׁ עֵצִים כַּאֲשֶׁר עָשָׂה אָבִי, גּוּפוֹ כִּנְחֹשֶׁת, וּלְפָנָיו אִישׁ מִשְׁתַּחֲוֶה לוֹ, וּמִזְבֵּחַ נֶגְדּוֹ, וּנְעָרִים נִשְׁחָטִים עָלָיו לִפְנֵי פְנֵי הַפֶּסֶל. וָאֹמַר: מַה זֶּה הַפֶּסֶל, אוֹ מַה הַמִּזְבֵּחַ, אוֹ מִי הַנִּזְבָּחִים, אוֹ מִי הַזֹּבֵחַ, אוֹ מַה זֶּה הַהֵיכָל אֲשֶׁר אֲנִי רוֹאֶה, נָאֶה בְּחָכְמַת מַעֲשֵׂהוּ וּבְיָפְיוֹ, מִתַּחַת לִדְרִיסַת כְּבוֹדְךָ. וַיֹּאמֶר: שְׁמַע אַבְרָהָם, זֶה אֲשֶׁר רָאִיתָ — הַהֵיכָל וְהַמִּזְבֵּחַ — נָאֶה הוּא לִי, מַחְשֶׁבֶת הַכְּהֻנָּה לְשֵׁם כְּבוֹדִי, אֲשֶׁר בָּהּ יִשְׁכֹּן כָּל קְהַל אֲנָשִׁים, וּמְבוֹא מְלָכִים וּנְבִיאִים, וְכָל קָרְבָּן אֲשֶׁר אָשִׂים לְהֵעָשׂוֹת לִי בְּעַמִּי הַבָּא וּבְזַרְעֲךָ. וְהַגּוּף אֲשֶׁר רָאִיתָ הוּא כַּעֲסִי, אֲשֶׁר בּוֹ יַכְעִיסוּנִי הָעָם הַבָּא מִמְּךָ אֵלָי; וְהָאִישׁ אֲשֶׁר רָאִיתָ נִשְׁחָט, הוּא קָרְבַּן הַמַּכְעִיס — רֶצַח — אֲשֶׁר יִהְיוּ לִי לְעֵדוּת מִשְׁפָּט בְּקֵץ הַתַּכְלִית, לְרָעַת הַבְּרִיאָה.
